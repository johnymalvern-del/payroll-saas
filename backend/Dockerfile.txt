FROM python:3.12-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_DISABLE_PIP_VERSION_CHECK=1 \
    PIP_DEFAULT_TIMEOUT=120 \
    PIP_RETRIES=10

RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    libpq-dev \
    postgresql-client \
    libpango-1.0-0 \
    libpangoft2-1.0-0 \
    libcairo2 \
    libgdk-pixbuf-2.0-0 \
    libffi-dev \
    libjpeg62-turbo-dev \
    zlib1g-dev \
    shared-mime-info \
    curl \
 && rm -rf /var/lib/apt/lists/*

WORKDIR /app

# Upgrade pip/setuptools/wheel first with retries — this is the step that was
# timing out against pypi.org in Codespaces. Explicit flags + env vars
# (PIP_DEFAULT_TIMEOUT, PIP_RETRIES above) make the network hiccup survivable.
RUN python -m pip install --no-cache-dir --upgrade \
    --timeout 120 \
    --retries 10 \
    pip setuptools wheel

COPY backend/pyproject.toml ./pyproject.toml
COPY backend/app ./app

# Same resilience for the actual project install.
RUN pip install --no-cache-dir \
    --timeout 120 \
    --retries 10 \
    -e ".[dev]"

COPY backend/alembic.ini ./alembic.ini
COPY backend/alembic ./alembic
COPY backend/tests ./tests

EXPOSE 8000
CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
