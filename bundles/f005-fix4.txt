import createClient from "openapi-fetch";
import type { paths } from "./schema";
import { getTenantSlug } from "./tenant";

export type TokenStore = {
  getToken: () => string | null;
  setToken: (token: string | null) => void;
};

export type RefreshFn = () => Promise<string | null>;

// Builds an absolute URL string. new Request("/path") throws in jsdom and
// Node >=18 because Request requires absolute URLs. All callers must use this
// helper before constructing a Request, otherwise fetch is never reached and
// the query fails silently.
export function absoluteUrl(path: string): string {
  const origin =
    typeof window !== "undefined" && window.location?.origin
      ? window.location.origin
      : "http://localhost";
  try {
    return new URL(path, origin).toString();
  } catch {
    return path;
  }
}

export function createWrappedFetch(
  tokenStore: TokenStore,
  refresh: RefreshFn,
  onUnauthorized: () => void,
): (req: Request) => Promise<Response> {
  return async (req) => {
    const original = req.clone();
    const first = new Request(req);
    const slug = getTenantSlug();
    if (slug) first.headers.set("X-Tenant-Slug", slug);
    const token = tokenStore.getToken();
    if (token) first.headers.set("Authorization", `Bearer ${token}`);
    let res = await fetch(first);
    if (res.status !== 401) return res;

    const newToken = await refresh();
    if (!newToken) {
      onUnauthorized();
      return res;
    }
    tokenStore.setToken(newToken);

    const retry = new Request(original);
    if (slug) retry.headers.set("X-Tenant-Slug", slug);
    retry.headers.set("Authorization", `Bearer ${newToken}`);
    res = await fetch(retry);
    if (res.status === 401) onUnauthorized();
    return res;
  };
}

let accessToken: string | null = null;

export function getAccessToken(): string | null {
  return accessToken;
}

export function setAccessToken(token: string | null): void {
  accessToken = token;
}

let refreshPromise: Promise<string | null> | null = null;

export async function refreshAccessToken(): Promise<string | null> {
  if (refreshPromise) return refreshPromise;
  refreshPromise = (async () => {
    try {
      const slug = getTenantSlug();
      const res = await fetch("/api/v1/auth/refresh", {
        method: "POST",
        credentials: "include",
        headers: slug ? { "X-Tenant-Slug": slug } : {},
      });
      if (!res.ok) return null;
      const body = (await res.json().catch(() => null)) as
        | { access_token?: unknown }
        | null;
      return typeof body?.access_token === "string" ? body.access_token : null;
    } catch {
      return null;
    }
  })();
  try {
    return await refreshPromise;
  } finally {
    refreshPromise = null;
  }
}

let onUnauthorizedDefault: (() => void) | null = null;

export function setOnUnauthorized(fn: (() => void) | null): void {
  onUnauthorizedDefault = fn;
}

export const wrappedFetch = createWrappedFetch(
  { getToken: getAccessToken, setToken: setAccessToken },
  refreshAccessToken,
  () => onUnauthorizedDefault?.(),
);

const API_BASE = absoluteUrl("/api/v1");

export const api = createClient<paths>({
  baseUrl: API_BASE,
  credentials: "include",
  fetch: wrappedFetch,
});

export type SheetCell = {
  work_date: string;
  status: string | null;
  worked_minutes: number | null;
  normal_minutes: number | null;
  overtime_minutes: number | null;
  flags: string[];
  open_exception_count: number;
  locked: boolean;
  reviewed: boolean;
};

export type SheetEmployee = {
  employee_id: string;
  employee_code: string;
  employee_name: string;
  site_id: string | null;
  department_id: string | null;
  days: SheetCell[];
};

export type SheetOut = {
  items: SheetEmployee[];
  next_cursor: string | null;
  total?: number | null;
};

export type Ref = { id: string; name: string };

function warnShapeMismatch(data: unknown): void {
  if (!data || typeof data !== "object") {
    console.warn("[attendance] sheet response is not an object", data);
    return;
  }
  const obj = data as Record<string, unknown>;
  if (!Array.isArray(obj.items)) {
    console.warn("[attendance] sheet response missing items array", Object.keys(obj));
    return;
  }
  const emp = obj.items[0];
  if (!emp || typeof emp !== "object") return;
  const wantEmp = ["employee_id", "employee_code", "employee_name", "days"];
  const missEmp = wantEmp.filter((k) => !(k in (emp as Record<string, unknown>)));
  if (missEmp.length > 0) {
    console.warn(
      "[attendance] sheet employee missing keys:",
      missEmp,
      "got:",
      Object.keys(emp as Record<string, unknown>),
    );
  }
  const days = (emp as Record<string, unknown>).days;
  if (Array.isArray(days) && days.length > 0) {
    const c = days[0];
    if (c && typeof c === "object") {
      const wantCell = [
        "work_date",
        "status",
        "worked_minutes",
        "flags",
        "open_exception_count",
        "locked",
        "reviewed",
      ];
      const missCell = wantCell.filter((k) => !(k in (c as Record<string, unknown>)));
      if (missCell.length > 0) {
        console.warn(
          "[attendance] sheet cell missing keys:",
          missCell,
          "got:",
          Object.keys(c as Record<string, unknown>),
        );
      }
    }
  }
}

function normaliseSheetOut(data: unknown): SheetOut {
  const obj = (data ?? {}) as Record<string, unknown>;
  const items = Array.isArray(obj.items) ? (obj.items as SheetEmployee[]) : [];
  const next =
    typeof obj.next_cursor === "string" ? obj.next_cursor : null;
  const total = typeof obj.total === "number" ? obj.total : null;
  return { items, next_cursor: next, total };
}

export async function fetchSheet(params: {
  fromDate: string;
  toDate: string;
  siteId?: string | null;
  branchId?: string | null;
  departmentId?: string | null;
  includeTerminated?: boolean;
  limit?: number;
  cursor?: string | null;
}): Promise<SheetOut> {
  const qs = new URLSearchParams();
  qs.set("from_date", params.fromDate);
  qs.set("to_date", params.toDate);
  if (params.siteId) qs.set("site_id", params.siteId);
  if (params.branchId) qs.set("branch_id", params.branchId);
  if (params.departmentId) qs.set("department_id", params.departmentId);
  if (params.includeTerminated) qs.set("include_terminated", "true");
  qs.set("limit", String(params.limit ?? 50));
  if (params.cursor) qs.set("cursor", params.cursor);
  const res = await wrappedFetch(
    new Request(absoluteUrl(`/api/v1/attendance/sheet?${qs.toString()}`), {
      credentials: "include",
    }),
  );
  if (!res.ok) {
    throw new Error(`sheet fetch failed: ${res.status}`);
  }
  const data = await res.json();
  warnShapeMismatch(data);
  return normaliseSheetOut(data);
}

async function fetchRefs(path: string): Promise<Ref[]> {
  const res = await wrappedFetch(
    new Request(absoluteUrl(`/api/v1${path}?limit=500`), {
      credentials: "include",
    }),
  );
  if (!res.ok) return [];
  const body = (await res.json().catch(() => ({}))) as { items?: unknown };
  return Array.isArray(body.items) ? (body.items as Ref[]) : [];
}

export function fetchBranches(): Promise<Ref[]> {
  return fetchRefs("/branches");
}

export function fetchSites(): Promise<Ref[]> {
  return fetchRefs("/sites");
}

export function fetchDepartments(): Promise<Ref[]> {
  return fetchRefs("/departments");
}

import { useQuery } from "@tanstack/react-query";
import { absoluteUrl, wrappedFetch } from "../api/client";
import { useAuth } from "./AuthContext";

export type Me = {
  id: string;
  username: string;
  role: string;
  employee_id: string | null;
  permissions: string[];
  must_change_password: boolean;
};

export const meQueryKey = ["me"] as const;

export async function fetchMe(): Promise<Me> {
  const res = await wrappedFetch(
    new Request(absoluteUrl("/api/v1/auth/me"), { credentials: "include" }),
  );
  if (!res.ok) {
    throw new Error(`/auth/me failed: ${res.status}`);
  }
  return (await res.json()) as Me;
}

export function useMe() {
  const { status } = useAuth();
  return useQuery<Me, Error>({
    queryKey: meQueryKey,
    queryFn: fetchMe,
    enabled: status === "authenticated",
    staleTime: Infinity,
    retry: 1,
  });
}

function jr(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), {

  });
}

function requestUrl(input: RequestInfo | URL): string {
  if (typeof input === "string") return input;
  if (input instanceof URL) return input.toString();
  if (typeof Request !== "undefined" && input instanceof Request) {
    return input.url;
  }
  return String(input);
}

function setupMock(sheetItems: unknown[]): void {
  global.fetch = vi.fn(async (input: RequestInfo | URL) => {
    const url = requestUrl(input);

      return jr({ items: sheetItems, next_cursor: null });

    if (
      url.includes("/api/v1/branches") ||
      url.includes("/api/v1/sites") ||
      url.includes("/api/v1/departments")

      return jr({ items: [], next_cursor: null });
    }
    return jr({}, 200);
  }) as unknown as typeof fetch;
}

## Important contracts

- `/auth/login` returns `{access_token, refresh_token, token_type, expires_in}` only. No `must_change_password`. UI uses `/auth/me`.
- `/auth/me` returns `{id, username, role, employee_id, permissions[], must_change_password}`.
- Usernames: `normalise_username` strips separators when a digit is present. Use `paone` not `pa-1`.
- Refresh: backend uses `csrf_token` cookie + `csrf_protection` dependency. **Client does not yet send the CSRF header.** Refresh will 403 until fixed. Flagged.
- **`new Request(relativeUrl)` throws in jsdom and Node >=18.** All manual requests go through `absoluteUrl()` in `client.ts`. `openapi-fetch` receives an absolute baseURL too. Never construct a bare relative Request again.
- `wrappedFetch(new Request(absoluteUrl(...)))` is the correct pattern for any request outside the openapi-fetch client.

## Known tool limitation

`apply_bundle.py --frontend` runs Vitest only on files written **inside the bundle**, not on `--also` files. Consequence: if you change a source file and the test that covers it was already committed, you must either include the test file in the bundle or run `npx vitest run <file>` manually. This caught us on F-1b and again on F-2a. Fix candidate for a later tool revision: also run tests that import changed source files (backend `find_dependent_tests` logic ported to TS).
