"""Defensive audit writer for payroll endpoints (Chapter 4).

The control audit table's exact column set is not fully probed yet; this
helper filters kwargs by column name so it works regardless. If a shared
audit helper exists later, wire it in here in one place.
"""
from __future__ import annotations

import uuid
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.orm import Session


def write_audit(
    db: Session,
    user: Any,
    *,
    entity_type: str,
    entity_id: uuid.UUID,
    action: str,
    old_value: dict | None = None,
    new_value: dict | None = None,
    reason: str | None = None,
) -> None:
    from app.db.tenant.models.platform import AuditLog

    cols = {c.name for c in AuditLog.__table__.columns}
    kwargs: dict[str, Any] = {}
    for k, v in (
        ("actor_user_id", getattr(user, "id", None)),
        ("actor_role", getattr(user, "role", None)),
        ("entity_type", entity_type),
        ("entity_id", entity_id),
        ("action", action),
        ("old_value", old_value),
        ("new_value", new_value),
        ("reason", reason),
        ("occurred_at", datetime.now(timezone.utc)),
    ):
        if k in cols:
            kwargs[k] = v
    db.add(AuditLog(**kwargs))

@router.patch("", response_model=PayrollSettingsOut)
def patch_payroll_settings(
    body: PayrollSettingsPatchIn,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_FINALISE))],
) -> PayrollSettingsOut:
    row = db.scalar(select(PayrollSettings).limit(1))
    old_snapshot = None
    if row is None:
        if body.row_version != 0:
            raise AppError(
                "record_changed",
                "Settings do not exist; use row_version=0 to create",
                409,
            )
        row = PayrollSettings(
            id=uuid.uuid4(),
            age_reference_date=None,
            sdl_applies=None,
            net_pay_floor=_DEFAULT_NET_PAY_FLOOR,
            row_version=1,
            updated_by_user_id=getattr(user, "id", None),
        )
        db.add(row)
    else:
        if body.row_version != row.row_version:
            raise AppError(
                "record_changed",
                f"row_version mismatch: sent {body.row_version}, current {row.row_version}",
                409,
            )
        old_snapshot = {
            "age_reference_date": row.age_reference_date,
            "sdl_applies": row.sdl_applies,
            "net_pay_floor": str(row.net_pay_floor),
            "row_version": row.row_version,
        }
        row.row_version += 1
        row.updated_by_user_id = getattr(user, "id", None)

    provided = body.model_fields_set - {"row_version"}
    for field in provided:
        setattr(row, field, getattr(body, field))
    if row.net_pay_floor is None:
        row.net_pay_floor = _DEFAULT_NET_PAY_FLOOR

    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_settings",
        entity_id=row.id,
        action="update",
        old_value=old_snapshot,
        new_value={
            "age_reference_date": row.age_reference_date,
            "sdl_applies": row.sdl_applies,
            "net_pay_floor": str(row.net_pay_floor),
            "row_version": row.row_version,
        },
    )
    db.commit()
    return _to_out(row)

def _encode_uuid_cursor(uid: uuid.UUID) -> str:
    raw = json.dumps({"i": str(uid)}, separators=(",", ":")).encode("utf-8")
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")


def _decode_uuid_cursor(cursor: str) -> uuid.UUID:
    padded = cursor + "=" * (-len(cursor) % 4)
    try:
        raw = base64.urlsafe_b64decode(padded.encode("ascii")).decode("utf-8")
        return uuid.UUID(json.loads(raw)["i"])
    except Exception:
        raise AppError("invalid_cursor", "The pagination cursor is malformed", 400) from None

@router.get("/earning-types", response_model=EarningTypeListOut)
def list_earning_types(
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_PREVIEW))],
    active_only: bool = Query(True),
    limit: int = Query(_DEFAULT_LIMIT, ge=1, le=_MAX_LIMIT),
    cursor: str | None = Query(None),
) -> EarningTypeListOut:
    q = select(EarningType)
    if active_only:
        q = q.where(EarningType.is_active.is_(True))
    if cursor is not None:
        q = q.where(EarningType.id > _decode_uuid_cursor(cursor))
    q = q.order_by(EarningType.id.asc()).limit(limit + 1)
    rows = list(db.scalars(q).all())
    next_cursor = None
    if len(rows) > limit:
        rows = rows[:limit]
        next_cursor = _encode_uuid_cursor(rows[-1].id)
    return EarningTypeListOut(
        items=[EarningTypeOut.model_validate(r) for r in rows],
        next_cursor=next_cursor,
    )

        setattr(row, field, getattr(body, field))
    row.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="deduction_type", entity_id=row.id, action="update",
        old_value=old_snapshot,
        new_value={
            "name": row.name, "category": row.category, "priority": row.priority,
            "is_pre_tax": row.is_pre_tax, "is_active": row.is_active,
            "row_version": row.row_version,
        },
    )
    db.commit()
    return DeductionTypeOut.model_validate(row)


@router.post("/deduction-types/{type_id}/deactivate", response_model=DeductionTypeOut)
def deactivate_deduction_type(
    type_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_FINALISE))],
    row_version: int = Query(...),
) -> DeductionTypeOut:
    row = db.scalar(select(DeductionType).where(DeductionType.id == type_id))
    if row is None:
        raise AppError("not_found", "Deduction type not found", 404)
    if row.is_system:
        raise AppError("system_row", "System deduction types cannot be deactivated via API", 409)
    if row_version != row.row_version:
        raise AppError(
            "record_changed",
            f"row_version mismatch: sent {row_version}, current {row.row_version}",
            409,
        )
    row.is_active = False
    row.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="deduction_type", entity_id=row.id, action="deactivate",
        old_value={"is_active": True}, new_value={"is_active": False},
    )
    db.commit()
    return DeductionTypeOut.model_validate(row)
