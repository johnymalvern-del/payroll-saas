import { beforeEach, describe, expect, it, vi } from "vitest";
import { fireEvent, render, screen, waitFor, within } from "@testing-library/react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { MantineProvider } from "@mantine/core";
import dayjs from "dayjs";
import { AttendanceSheetPage } from "./AttendanceSheetPage";

function jr(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "Content-Type": "application/json" },
  });
}

function requestUrl(input: RequestInfo | URL): string {
  if (typeof input === "string") return input;
  if (input instanceof URL) return input.toString();
  if (typeof Request !== "undefined" && input instanceof Request) return input.url;
  return String(input);
}

function setupMock(opts: {
  sheetItems?: unknown[];
  dailyItems?: unknown[];
  dayPayload?: unknown;
  markRefused?: unknown[];
}): void {
  global.fetch = vi.fn(async (input: RequestInfo | URL) => {
    const url = requestUrl(input);
    if (url.includes("/api/v1/attendance/sheet")) {
      return jr({ items: opts.sheetItems ?? [], next_cursor: null });
    }
    if (url.includes("/api/v1/attendance/daily")) {
      return jr({ items: opts.dailyItems ?? [], next_cursor: null });
    }
    if (url.includes("/api/v1/attendance/day/")) {
      return jr(opts.dayPayload ?? {});
    }
    if (url.includes("/api/v1/attendance/mark_reviewed")) {
      return jr({ reviewed_count: 3, refused_days: opts.markRefused ?? [] });
    }
    if (
      url.includes("/api/v1/branches") ||
      url.includes("/api/v1/sites") ||
      url.includes("/api/v1/departments")
    ) {
      return jr({ items: [], next_cursor: null });
    }
    return jr({}, 200);
  }) as unknown as typeof fetch;
}

function renderPage() {
  const client = new QueryClient({
    defaultOptions: { queries: { retry: false } },
  });
  return render(
    <MantineProvider>
      <QueryClientProvider client={client}>
        <AttendanceSheetPage />
      </QueryClientProvider>
    </MantineProvider>,
  );
}

// Clicks the clickable ancestor of the "worked" badge. The Badge itself sits
// inside a div with role="button" that carries onClick, so we target the
// ancestor to avoid any tooltip-wrapping ambiguity.
function clickWorkedCell(): void {
  const badge = screen.getByText("worked");
  const clickable = badge.closest('[role="button"]') as HTMLElement | null;
  if (!clickable) {
    throw new Error("worked cell is not inside a [role=button] ancestor");
  }
  fireEvent.click(clickable);
}

describe("AttendanceSheetPage monthly view", () => {
  beforeEach(() => setupMock({ sheetItems: [EMPLOYEE] }));

  it("renders title and employee row", async () => {
    renderPage();
    expect(screen.getByText("Attendance")).toBeInTheDocument();
    await waitFor(() => {
      expect(screen.getByText("Ada Lovelace")).toBeInTheDocument();
      expect(screen.getByText("EMP001")).toBeInTheDocument();
    });
  });

  it("shows empty text when sheet returns no employees", async () => {
    setupMock({ sheetItems: [] });
    renderPage();
    await waitFor(() => {
      expect(
        screen.getByText(/no employees match these filters/i),
      ).toBeInTheDocument();
    });
  });

  it("renders a status badge for a worked cell", async () => {
    renderPage();
    await waitFor(() => {
      expect(screen.getByText("worked")).toBeInTheDocument();
    });
  });

  it("shows an error message when sheet fetch fails", async () => {
    global.fetch = vi.fn(async (input: RequestInfo | URL) => {
      const url = requestUrl(input);
      if (url.includes("/api/v1/attendance/sheet")) {
        return jr({ error: { code: "boom" } }, 500);
      }
      return jr({ items: [], next_cursor: null });
    }) as unknown as typeof fetch;
    renderPage();
    await waitFor(() => {
      expect(
        screen.getByText(/could not load the attendance sheet/i),
      ).toBeInTheDocument();
    });
  });

  it("opens the day panel on cell click and shows punches", async () => {
    setupMock({ sheetItems: [EMPLOYEE], dayPayload: DAY });
    renderPage();
    await waitFor(() => expect(screen.getByText("worked")).toBeInTheDocument());
    clickWorkedCell();
    await waitFor(() => {
      expect(screen.getByText("Day detail")).toBeInTheDocument();
    });
    const dialog = screen.getByRole("dialog");
    expect(within(dialog).getByText(/^Punches$/i)).toBeInTheDocument();
  });

  it("day panel never shows an adjustment marker (C2)", async () => {
    setupMock({ sheetItems: [EMPLOYEE], dayPayload: DAY });
    renderPage();
    await waitFor(() => expect(screen.getByText("worked")).toBeInTheDocument());
    clickWorkedCell();
    await waitFor(() => {
      expect(screen.getByText("Day detail")).toBeInTheDocument();
    });
    const dialog = screen.getByRole("dialog");
    expect(/adjust/i.test(dialog.textContent ?? "")).toBe(false);
  });
});

describe("AttendanceSheetPage mark reviewed", () => {
  it("shows refused days table when some days are refused", async () => {
    setupMock({
      sheetItems: [EMPLOYEE],
      markRefused: [
        {
          employee_id: "e1",
          employee_name: "Ada Lovelace",
          work_date: "2025-06-05",
          reason: "open_exceptions",
        },
      ],
    });
    renderPage();
    await waitFor(() => expect(screen.getByText("worked")).toBeInTheDocument());
    // 1) Page button opens the modal.
    fireEvent.click(
      screen.getByRole("button", { name: /mark clean days reviewed/i }),
    );
    // 2) Modal opens; the submit button inside the modal triggers the mutation.
    const dialog = await screen.findByRole("dialog");
    const submitBtn = within(dialog).getByRole("button", {
      name: /mark clean days reviewed/i,
    });
    fireEvent.click(submitBtn);
    // 3) Refused days table appears.
    await waitFor(() => {
      expect(screen.getByText("open_exceptions")).toBeInTheDocument();
    });
  });
});
