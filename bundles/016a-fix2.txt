VALID_TABLE_TYPES = {
    "tax_bracket", "tax_rebate", "tax_threshold", "medical_credit",
    "retirement_limit", "uif", "sdl",
}

REQUIRED_ROW_KEYS = {
    "tax_rebate": {"primary", "secondary_65", "tertiary_75"},
    "tax_threshold": {"below_age_65", "age_65_to_74", "age_75_plus"},
    "medical_credit": {"first_member", "additional_member", "first_dependant"},
    "uif": {"employee_rate", "employer_rate", "monthly_ceiling"},
    "sdl": {"rate", "exemption_threshold_annual"},
    "retirement_limit": {"percentage_of_taxable_income", "annual_cap"},
}

REQUIRED_COLUMNS = {
    "id": "UUID",
    "country_code": "VARCHAR(2)",
    "table_type": "VARCHAR(40)",
    "version": "VARCHAR(40)",
    "tax_year": "VARCHAR(20)",
    "source_document": "VARCHAR(200)",
    "source_url": "VARCHAR(500)",
    "source_date": "DATE",
    "effective_from": "DATE",
    "effective_to": "DATE",
    "is_active": "BOOLEAN DEFAULT TRUE",
    "loaded_by_user_id": "UUID",
    "loaded_at": "TIMESTAMPTZ DEFAULT NOW()",
    "row_version": "INTEGER DEFAULT 1",
}


def _is_int_like(data_type: str) -> bool:
    t = (data_type or "").lower()
    return any(x in t for x in ("int", "numeric", "decimal"))


def bootstrap_schema(engine: Engine) -> None:
    with engine.begin() as conn:
        tables = {r[0] for r in conn.execute(sa.text(
            "SELECT table_name FROM information_schema.tables WHERE table_schema='public'"
        ))}

        if "statutory_table" in tables:
            existing_cols = {row[0]: (row[1] or "").lower() for row in conn.execute(sa.text(
                "SELECT column_name, data_type FROM information_schema.columns "
                "WHERE table_schema='public' AND table_name='statutory_table'"
            ))}

            if _is_int_like(existing_cols.get("version", "")):
                n_main = conn.scalar(sa.text("SELECT COUNT(*) FROM statutory_table")) or 0
                n_row = 0
                n_log = 0
                if "statutory_table_row" in tables:
                    n_row = conn.scalar(sa.text("SELECT COUNT(*) FROM statutory_table_row")) or 0
                if "statutory_table_load_log" in tables:
                    n_log = conn.scalar(sa.text("SELECT COUNT(*) FROM statutory_table_load_log")) or 0
                if (n_main + n_row + n_log) > 0:
                    raise RuntimeError(
                        f"statutory_table has legacy INTEGER version column with data "
                        f"(main={n_main}, row={n_row}, log={n_log}). "
                        "Manual migration required: ALTER COLUMN version TYPE VARCHAR(40) "
                        "USING version::text; then rerun bootstrap_schema."
                    )
                conn.execute(sa.text("DROP TABLE statutory_table CASCADE"))
                conn.execute(sa.text("DROP TABLE IF EXISTS statutory_table_row CASCADE"))
                conn.execute(sa.text("DROP TABLE IF EXISTS statutory_table_load_log CASCADE"))
                tables.discard("statutory_table")
            else:
                for col, typ in REQUIRED_COLUMNS.items():
                    if col not in existing_cols:
                        conn.execute(sa.text(
                            f"ALTER TABLE statutory_table ADD COLUMN {col} {typ}"
                        ))

        if "statutory_table" not in tables:
            conn.execute(sa.text("""
                CREATE TABLE statutory_table (
                    id UUID PRIMARY KEY,
                    country_code VARCHAR(2) NOT NULL,
                    table_type VARCHAR(40) NOT NULL,
                    version VARCHAR(40) NOT NULL,
                    tax_year VARCHAR(20),
                    source_document VARCHAR(200) NOT NULL,
                    source_url VARCHAR(500) NOT NULL,
                    source_date DATE NOT NULL,
                    effective_from DATE NOT NULL,
                    effective_to DATE NOT NULL,
                    is_active BOOLEAN NOT NULL DEFAULT TRUE,
                    loaded_by_user_id UUID,
                    loaded_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
                    row_version INTEGER NOT NULL DEFAULT 1,
                    UNIQUE (country_code, table_type, version)
                )
            """))

        conn.execute(sa.text("""
            CREATE TABLE IF NOT EXISTS statutory_table_row (
                id UUID PRIMARY KEY,
                statutory_table_id UUID NOT NULL REFERENCES statutory_table(id) ON DELETE CASCADE,
                row_key VARCHAR(40) NOT NULL,
                row_data JSONB NOT NULL,
                sort_order INTEGER NOT NULL DEFAULT 0,
                UNIQUE (statutory_table_id, row_key)
            )
        """))

        conn.execute(sa.text("""
            CREATE TABLE IF NOT EXISTS statutory_table_load_log (
                id UUID PRIMARY KEY,
                action VARCHAR(20) NOT NULL,
                statutory_table_id UUID,
                actor_user_id UUID,
                payload JSONB NOT NULL,
                diff_summary JSONB,
                typed_confirmation_matched BOOLEAN NOT NULL DEFAULT FALSE,
                occurred_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
            )
        """))

        conn.execute(sa.text(
            "CREATE INDEX IF NOT EXISTS ix_statutory_table_active "
            "ON statutory_table (country_code, table_type, effective_from, effective_to) "
            "WHERE is_active"
        ))


_bootstrapped_engine_ids: set[int] = set()


def _ensure_bootstrapped(engine: Engine) -> None:
    """Idempotent schema-bootstrap cache keyed on the engine instance."""
    key = id(engine)
    if key in _bootstrapped_engine_ids:
        return
    bootstrap_schema(engine)
    _bootstrapped_engine_ids.add(key)

@dataclass
class ValidationResult:
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.errors


def _parse_date(value: Any, name: str, errors: list[str]) -> date | None:
    if value is None:
        errors.append(f"{name} is required")
        return None
    if isinstance(value, date):
        return value
    try:
        return date.fromisoformat(str(value))
    except (ValueError, TypeError):
        errors.append(f"{name} not a valid ISO date: {value!r}")
        return None


def _parse_decimal(value: Any, name: str, errors: list[str]) -> Decimal | None:
    if value is None:
        errors.append(f"{name} is required")
        return None
    try:
        return Decimal(str(value))
    except (InvalidOperation, TypeError, ValueError):
        errors.append(f"{name} not a valid decimal: {value!r}")
        return None


def validate_document(doc: dict) -> ValidationResult:
    errors: list[str] = []

    for name in ("country_code", "table_type", "version", "source_document",
                 "source_url", "source_date", "effective_from", "effective_to"):
        if not doc.get(name):
            errors.append(f"{name} is required")

    cc = doc.get("country_code")
    if cc and (not isinstance(cc, str) or len(cc) != 2 or not cc.isalpha() or not cc.isupper()):
        errors.append(f"country_code must be 2 uppercase letters: {cc!r}")

    tt = doc.get("table_type")
    if tt and tt not in VALID_TABLE_TYPES:
        errors.append(f"table_type must be one of {sorted(VALID_TABLE_TYPES)}: {tt!r}")

    su = doc.get("source_url")
    if su and not (isinstance(su, str) and su.startswith("https://")):
        errors.append(f"source_url must start with https://: {su!r}")

    ef = _parse_date(doc.get("effective_from"), "effective_from", errors)
    et = _parse_date(doc.get("effective_to"), "effective_to", errors)
    _parse_date(doc.get("source_date"), "source_date", errors)
    if ef and et and et < ef:
        errors.append("effective_to must be >= effective_from")

    rows = doc.get("rows")
    if not isinstance(rows, list) or not rows:
        errors.append("rows must be a non-empty list")
        return ValidationResult(errors=errors)

    if tt == "tax_bracket":
        _validate_tax_brackets(rows, errors)
    elif tt in REQUIRED_ROW_KEYS:
        present = {r.get("row_key") for r in rows if isinstance(r, dict)}
        missing = REQUIRED_ROW_KEYS[tt] - present
        extra = present - REQUIRED_ROW_KEYS[tt]
        if missing:
            errors.append(f"missing required row_keys for {tt}: {sorted(missing)}")
        if extra:
            errors.append(f"unexpected row_keys for {tt}: {sorted(extra)}")

    return ValidationResult(errors=errors)


def _validate_tax_brackets(rows: list, errors: list[str]) -> None:
    parsed: list[tuple[Decimal, Decimal | None]] = []
    for i, row in enumerate(rows):
        if not isinstance(row, dict):
            errors.append(f"bracket row {i} not an object")
            continue
        try:
            lb = Decimal(str(row.get("lower_bound")))
            ub = None if row.get("upper_bound") is None else Decimal(str(row["upper_bound"]))
            rate = Decimal(str(row.get("rate")))
        except (InvalidOperation, TypeError, ValueError):
            errors.append(f"bracket {i}: numeric parse failed")
            continue
        _parse_decimal(row.get("base_tax"), f"bracket {i} base_tax", errors)
        if not (Decimal(0) <= rate <= Decimal(1)):
            errors.append(f"bracket {i}: rate {rate} not in [0, 1]")
        parsed.append((lb, ub))

    if parsed and parsed[0][0] != Decimal(0):
        errors.append(f"first bracket lower_bound must be 0, got {parsed[0][0]}")
    for i in range(len(parsed) - 1):
        if parsed[i][1] != parsed[i + 1][0]:
            errors.append(
                f"brackets not contiguous: [{i}] ends at {parsed[i][1]}, "
                f"[{i+1}] starts at {parsed[i+1][0]}"
            )
    if parsed and parsed[-1][1] is not None:
        errors.append("last bracket upper_bound must be null")


@dataclass
class DiffSummary:
    action: str
    replaced_version: str | None
    rows_added: int
    rows_removed: int
    rows_changed: int

    def to_dict(self) -> dict:
        return {
            "action": self.action,
            "replaced_version": self.replaced_version,
            "rows_added": self.rows_added,
            "rows_removed": self.rows_removed,
            "rows_changed": self.rows_changed,
        }


def _row_key_map(rows: list) -> dict:
    return {r.get("row_key"): r for r in rows if isinstance(r, dict)}


def compute_diff(new_doc: dict, current: dict | None) -> DiffSummary:
    if current is None:
        return DiffSummary("new", None, len(new_doc.get("rows", [])), 0, 0)
    old_map = _row_key_map(current.get("rows", []))
    new_map = _row_key_map(new_doc.get("rows", []))
    added = set(new_map) - set(old_map)
    removed = set(old_map) - set(new_map)
    common = set(old_map) & set(new_map)
    changed = {k for k in common if old_map[k] != new_map[k]}
    return DiffSummary("replaces", current.get("version"), len(added), len(removed), len(changed))


def _fetch_current_active(engine: Engine, country_code: str, table_type: str) -> dict | None:
    with engine.connect() as conn:
        head = conn.execute(sa.text(
            "SELECT id, version FROM statutory_table "
            "WHERE country_code=:cc AND table_type=:tt AND is_active "
            "ORDER BY effective_from DESC LIMIT 1"
        ), {"cc": country_code, "tt": table_type}).first()
        if head is None:
            return None
        rows = conn.execute(sa.text(
            "SELECT row_key, row_data, sort_order FROM statutory_table_row "
            "WHERE statutory_table_id=:tid ORDER BY sort_order"
        ), {"tid": head[0]}).all()
        return {
            "id": str(head[0]),
            "version": head[1],
            "rows": [{"row_key": r[0], **r[1], "sort_order": r[2]} for r in rows],
        }


def apply_document(engine: Engine, doc: dict, *, actor_user_id: str | None, typed_version: str) -> dict:
    res = validate_document(doc)
    if not res.ok:
        raise ValueError("validation failed: " + "; ".join(res.errors))
    if typed_version != doc["version"]:
        raise ValueError(
            f"typed confirmation mismatch: expected {doc['version']!r}, got {typed_version!r}"
        )

    current = _fetch_current_active(engine, doc["country_code"], doc["table_type"])
    diff = compute_diff(doc, current)
    table_id = uuid.uuid4()

    with engine.begin() as conn:
        overlap = conn.execute(sa.text(
            "SELECT version FROM statutory_table "
            "WHERE country_code=:cc AND table_type=:tt AND is_active "
            "AND effective_from <= :et AND effective_to >= :ef"
        ), {
            "cc": doc["country_code"], "tt": doc["table_type"],
            "ef": doc["effective_from"], "et": doc["effective_to"],
        }).first()
        if overlap is not None:
            raise ValueError(
                f"overlaps active version {overlap[0]!r}; end-date that version first"
            )

        conn.execute(sa.text(
            "INSERT INTO statutory_table "
            "(id, country_code, table_type, version, tax_year, source_document, source_url, "
            " source_date, effective_from, effective_to, is_active, loaded_by_user_id) "
            "VALUES (:id, :cc, :tt, :v, :ty, :sd, :su, :sdate, :ef, :et, TRUE, :uid)"
        ), {
            "id": table_id, "cc": doc["country_code"], "tt": doc["table_type"],
            "v": doc["version"], "ty": doc.get("tax_year"),
            "sd": doc["source_document"], "su": doc["source_url"],
            "sdate": doc["source_date"], "ef": doc["effective_from"], "et": doc["effective_to"],
            "uid": actor_user_id,
        })

        for i, row in enumerate(doc["rows"]):
            row_data = {k: v for k, v in row.items() if k not in ("row_key", "sort_order")}
            conn.execute(sa.text(
                "INSERT INTO statutory_table_row "
                "(id, statutory_table_id, row_key, row_data, sort_order) "
                "VALUES (:id, :tid, :rk, CAST(:rd AS JSONB), :so)"
            ), {
                "id": uuid.uuid4(), "tid": table_id, "rk": row["row_key"],
                "rd": json.dumps(row_data, default=str), "so": row.get("sort_order", i),
            })

        conn.execute(sa.text(
            "INSERT INTO statutory_table_load_log "
            "(id, action, statutory_table_id, actor_user_id, payload, diff_summary, "
            " typed_confirmation_matched) "
            "VALUES (:id, 'confirm', :tid, :uid, CAST(:p AS JSONB), CAST(:d AS JSONB), TRUE)"
        ), {
            "id": uuid.uuid4(), "tid": table_id, "uid": actor_user_id,
            "p": json.dumps(doc, default=str),
            "d": json.dumps(diff.to_dict()),
        })

    return {"table_id": str(table_id), "diff": diff.to_dict()}


def end_date(engine: Engine, table_id: str, *, new_end_date: str, actor_user_id: str | None) -> dict:
    with engine.begin() as conn:
        row = conn.execute(sa.text(
            "SELECT effective_from, version FROM statutory_table WHERE id=:id"
        ), {"id": table_id}).first()
        if row is None:
            raise ValueError(f"table {table_id} not found")
        ef, version = row
        et_new = date.fromisoformat(new_end_date)
        if et_new < ef:
            raise ValueError(f"new end-date {et_new} < effective_from {ef}")
        conn.execute(sa.text(
            "UPDATE statutory_table SET effective_to=:et, row_version=row_version+1 WHERE id=:id"
        ), {"et": et_new, "id": table_id})
        conn.execute(sa.text(
            "INSERT INTO statutory_table_load_log "
            "(id, action, statutory_table_id, actor_user_id, payload, typed_confirmation_matched) "
            "VALUES (:id, 'end_date', :tid, :uid, CAST(:p AS JSONB), TRUE)"
        ), {
            "id": uuid.uuid4(), "tid": table_id, "uid": actor_user_id,
            "p": json.dumps({"new_end_date": new_end_date, "version": version}),
        })
    return {"table_id": table_id, "new_end_date": new_end_date}


def get_active_table(engine: Engine, country_code: str, table_type: str, on_date: date) -> dict | None:
    with engine.connect() as conn:
        head = conn.execute(sa.text(
            "SELECT id, version, effective_from, effective_to FROM statutory_table "
            "WHERE country_code=:cc AND table_type=:tt AND is_active "
            "AND effective_from <= :d AND effective_to >= :d "
            "ORDER BY effective_from DESC LIMIT 1"
        ), {"cc": country_code, "tt": table_type, "d": on_date}).first()
        if head is None:
            return None
        rows = conn.execute(sa.text(
            "SELECT row_key, row_data, sort_order FROM statutory_table_row "
            "WHERE statutory_table_id=:tid ORDER BY sort_order"
        ), {"tid": head[0]}).all()
        return {
            "id": str(head[0]), "version": head[1],
            "effective_from": head[2].isoformat(), "effective_to": head[3].isoformat(),
            "rows": [{"row_key": r[0], **r[1], "sort_order": r[2]} for r in rows],
        }


def _default_engine() -> Engine:
    """Return the control-database engine.

    Preference order (matches what tests and the app use):
      1. The engine behind control_session() - same engine fixtures bootstrap.
      2. get_control_engine() if that helper exists.
      3. Fallback to _admin_url("payroll_control" / "control" / "payroll_ctrl").
    """
    try:
        from app.db.control.session import control_session
        with control_session() as s:
            bind = s.get_bind()
        if hasattr(bind, "connect"):
            return bind
        return bind.engine
    except Exception:
        pass
    try:
        from app.db.control.session import get_control_engine  # type: ignore[attr-defined]
        return get_control_engine()
    except (ImportError, AttributeError):
        pass
    from app.services.provisioning import _admin_url
    for name in ("payroll_control", "control", "payroll_ctrl"):
        try:
            eng = sa.create_engine(_admin_url(name), future=True)
            with eng.connect() as c:
                c.execute(sa.text("SELECT 1"))
            return eng
        except Exception:
            continue
    raise RuntimeError(
        "cannot locate control DB; pass --db-url or set CONTROL_DB_URL in the environment"
    )


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(prog="statutory_loader")
    sub = p.add_subparsers(dest="cmd", required=True)

    for name in ("validate", "confirm"):
        sp = sub.add_parser(name)
        sp.add_argument("file")
        sp.add_argument("--db-url")
        if name == "confirm":
            sp.add_argument("--typed-version", required=True)
            sp.add_argument("--actor")

    sp = sub.add_parser("end-date")
    sp.add_argument("--table-id", required=True)
    sp.add_argument("--new-end-date", required=True)
    sp.add_argument("--db-url")
    sp.add_argument("--actor")

    args = p.parse_args(argv)
    engine = sa.create_engine(args.db_url, future=True) if args.db_url else _default_engine()
    bootstrap_schema(engine)

    if args.cmd == "validate":
        doc = json.loads(Path(args.file).read_text(encoding="utf-8"))
        res = validate_document(doc)
        if not res.ok:
            for err in res.errors:
                print(f"ERROR: {err}", file=sys.stderr)
            return 1
        current = _fetch_current_active(engine, doc["country_code"], doc["table_type"])
        diff = compute_diff(doc, current)
        print(f"OK. {diff.action}. {json.dumps(diff.to_dict())}")
        return 0

    if args.cmd == "confirm":
        doc = json.loads(Path(args.file).read_text(encoding="utf-8"))
        try:
            result = apply_document(engine, doc, actor_user_id=args.actor, typed_version=args.typed_version)
        except ValueError as exc:
            print(f"REFUSED: {exc}", file=sys.stderr)
            return 2
        print(f"OK. loaded table_id={result['table_id']} diff={json.dumps(result['diff'])}")
        return 0

    if args.cmd == "end-date":
        try:
            result = end_date(engine, args.table_id, new_end_date=args.new_end_date, actor_user_id=args.actor)
        except ValueError as exc:
            print(f"REFUSED: {exc}", file=sys.stderr)
            return 2
        print(f"OK. {json.dumps(result)}")
        return 0

    return 0


if __name__ == "__main__":
    sys.exit(main())

_COUNTRY_CODE = "ZA"
_PERIODS_PER_YEAR = {"monthly": 12, "weekly": 52, "fortnightly": 26}


def _load_settings(db: Session) -> PayrollSettings:
    settings = db.scalar(select(PayrollSettings).limit(1))
    if settings is None:
        raise AppError(
            "settings_missing",
            "Payroll settings have not been created yet; call PATCH /payroll/settings first",
            409,
        )
    return settings


def _load_control_table(table_type: str, on_date: date) -> list[dict]:
    engine = _default_engine()
    _ensure_bootstrapped(engine)
    row = get_active_table(engine, _COUNTRY_CODE, table_type, on_date)
    if row is None:
        raise AppError(
            "statutory_table_missing",
            f"{_COUNTRY_CODE}/{table_type} has no active version covering {on_date.isoformat()}",
            409,
        )
    return row["rows"]


def _resolve_age(employee: Employee, on_date: date, age_ref: str) -> int:
    if employee.date_of_birth is None:
        raise AppError("missing_date_of_birth", "employee.date_of_birth is null", 409)
    return T.compute_age(employee.date_of_birth, on_date)


def build_pipeline_config(
    db: Session, employee: Employee, pay_date: date,
) -> P.PipelineConfig:
    settings = _load_settings(db)
    if settings.sdl_applies is None:
        raise AppError("sdl_applies_unset", "payroll.sdl_applies is unset", 409)
    if settings.age_reference_date is None:
        raise AppError("age_reference_date_unset", "payroll.age_reference_date is unset", 409)
    if employee.pay_frequency not in _PERIODS_PER_YEAR:
        raise AppError(
            "unsupported_pay_frequency",
            f"pay_frequency {employee.pay_frequency!r} not supported",
            409,
        )

    age_date = pay_date
    age = _resolve_age(employee, age_date, settings.age_reference_date)

    control_engine = _default_engine()
    _ensure_bootstrapped(control_engine)
    try:
        tax_tables = T.load_tax_tables(control_engine, pay_date, country_code=_COUNTRY_CODE)
    except R.PayrollRefusal as exc:
        raise AppError(exc.code, exc.message, exc.http_status) from None

    return P.PipelineConfig(
        tax_engine=T.default_engine(),
        tax_tables=tax_tables,
        uif_rows=_load_control_table("uif", pay_date),
        sdl_rows=_load_control_table("sdl", pay_date),
        retirement_rows=_load_control_table("retirement_limit", pay_date),
        periods_per_year=_PERIODS_PER_YEAR[employee.pay_frequency],
        employee_age=age,
        medical_members=employee.medical_scheme_members or 0,
        uif_exempt=bool(employee.uif_exempt),
        sdl_applies=settings.sdl_applies,
        net_pay_floor=settings.net_pay_floor or Decimal("0.00"),
    )


def _build_earnings(db: Session, employee: Employee, on_date: date) -> list[P.EarningLine]:
    rows = db.scalars(
        select(EmployeeEarning).where(
            EmployeeEarning.employee_id == employee.id,
            EmployeeEarning.is_active.is_(True),
            EmployeeEarning.effective_from <= on_date,
        )
    ).all()
    lines: list[P.EarningLine] = []
    for r in rows:
        if r.effective_to is not None and r.effective_to < on_date:
            continue
        et = db.scalar(select(EarningType).where(EarningType.id == r.earning_type_id))
        if et is None:
            raise AppError(
                "earning_type_missing",
                f"employee_earning {r.id} references missing earning_type",
                500,
            )
        if et.calc_method != "fixed_amount":
            raise AppError(
                "calc_method_unsupported",
                f"earning_type {et.code!r} has calc_method {et.calc_method!r}; "
                f"only 'fixed_amount' is supported in this release",
                409,
            )
        if r.amount is None:
            raise AppError(
                "amount_missing",
                f"employee_earning {r.id} has null amount with fixed_amount calc_method",
                500,
            )
        lines.append(P.EarningLine(
            label=et.name,
            amount=Decimal(r.amount),
            frequency=et.frequency,
            is_taxable=bool(et.is_taxable),
            counts_for_uif=bool(et.counts_for_uif),
            counts_for_sdl=bool(et.counts_for_sdl),
            source_ref={"employee_earning_id": str(r.id), "earning_type_code": et.code},
        ))
    return lines


def _build_deductions(db: Session, employee: Employee, on_date: date) -> list[P.DeductionLine]:
    rows = db.scalars(
        select(EmployeeDeduction).where(
            EmployeeDeduction.employee_id == employee.id,
            EmployeeDeduction.is_active.is_(True),
            EmployeeDeduction.effective_from <= on_date,
        )
    ).all()
    lines: list[P.DeductionLine] = []
    for r in rows:
        if r.effective_to is not None and r.effective_to < on_date:
            continue
        dt = db.scalar(select(DeductionType).where(DeductionType.id == r.deduction_type_id))
        if dt is None:
            raise AppError(
                "deduction_type_missing",
                f"employee_deduction {r.id} references missing deduction_type",
                500,
            )
        lines.append(P.DeductionLine(
            label=dt.name,
            amount=Decimal(r.amount),
            category=dt.category,
            is_pre_tax=bool(dt.is_pre_tax),
            priority=dt.priority,
            source_ref={"employee_deduction_id": str(r.id), "deduction_type_code": dt.code},
        ))
    return lines


def _write_payslip(
    db: Session, run: PayrollRun, pay_period: PayPeriod,
    employee: Employee, draft: P.PayslipDraft, *, status: str,
    prior: Payslip | None,
) -> Payslip:
    payslip = Payslip(
        id=uuid.uuid4(),
        payroll_run_id=run.id,
        pay_period_id=pay_period.id,
        employee_id=employee.id,
        revision=run.revision,
        status=status,
        gross=draft.gross,
        pre_tax_deductions=draft.pre_tax_deductions,
        taxable_income=draft.taxable_income,
        paye=draft.paye,
        uif_employee=draft.uif_employee,
        uif_employer=draft.uif_employer,
        sdl=draft.sdl,
        other_deductions=draft.other_deductions,
        reimbursements=draft.reimbursements,
        net=draft.net,
        employer_costs=draft.employer_costs(),
        snapshot={"calculation_context": draft.calculation_context, "tax_trace": draft.tax_trace},
        calculation_context=draft.calculation_context,
        predecessor_payslip_id=prior.id if prior else None,
    )
    db.add(payslip)
    db.flush()

    if prior is not None and prior.status == "current" and status == "current":
        prior.status = "superseded"

    for i, d in enumerate(draft.deferred_deductions):
        db.add(PayslipLine(
            id=uuid.uuid4(), payslip_id=payslip.id,
            line_type="deduction", step="other", sort_order=1000 + i,
            label=f"Deferred: {d['label']}",
            amount=Decimal(d["amount"]),
            source_ref={"deferred": True, "category": d.get("category")},
        ))
    db.add(PayslipLine(
        id=uuid.uuid4(), payslip_id=payslip.id,
        line_type="tax", step="tax", sort_order=200, label="PAYE",
        amount=draft.paye, source_ref={"method": draft.calculation_context.get("tax_method")},
    ))
    db.add(PayslipLine(
        id=uuid.uuid4(), payslip_id=payslip.id,
        line_type="deduction", step="statutory", sort_order=300, label="UIF (employee)",
        amount=draft.uif_employee, source_ref={"statutory": "uif"},
    ))
    db.add(PayslipLine(
        id=uuid.uuid4(), payslip_id=payslip.id,
        line_type="employer_cost", step="employer_cost", sort_order=800,
        label="UIF (employer)", amount=draft.uif_employer,
        source_ref={"statutory": "uif"},
    ))
    db.add(PayslipLine(
        id=uuid.uuid4(), payslip_id=payslip.id,
        line_type="employer_cost", step="employer_cost", sort_order=810,
        label="SDL", amount=draft.sdl, source_ref={"statutory": "sdl"},
    ))
    db.flush()
    return payslip


def _purge_preview(db: Session, pay_period_id: uuid.UUID) -> None:
    prior = db.scalar(
        select(PayrollRun).where(
            PayrollRun.pay_period_id == pay_period_id,
            PayrollRun.kind == "preview",
        ).order_by(PayrollRun.started_at.desc())
    )
    if prior is None:
        return
    db.execute(
        PayslipLine.__table__.delete().where(
            PayslipLine.payslip_id.in_(
                select(Payslip.id).where(Payslip.payroll_run_id == prior.id)
            )
        )
    )
    db.execute(Payslip.__table__.delete().where(Payslip.payroll_run_id == prior.id))
    db.execute(PayrollRun.__table__.delete().where(PayrollRun.id == prior.id))
    db.flush()


def _in_scope_employees(db: Session) -> list[Employee]:
    return list(db.scalars(
        select(Employee).where(Employee.status != "terminated")
    ).all())


def _formula_warnings(db: Session) -> list[str]:
    settings = _load_settings(db)
    acks = settings.formula_acks or {}
    return [code for code in FORMULA_REGISTRY if code not in acks]


def run_preview(db: Session, user, pay_period: PayPeriod) -> PayrollRun:
    if pay_period.status == "locked":
        raise AppError("period_locked", "Cannot preview a finalised period; reopen it first", 409)

    _purge_preview(db, pay_period.id)
    ensure_system_earning_types(db)

    prior_final_max = db.scalar(
        select(Payslip.revision).where(Payslip.pay_period_id == pay_period.id)
        .order_by(Payslip.revision.desc())
    ) or 0
    rev = int(prior_final_max) + 1

    run = PayrollRun(
        id=uuid.uuid4(),
        pay_period_id=pay_period.id,
        kind="preview",
        revision=rev,
        status="running",
        rules_snapshot={},
        started_by_user_id=getattr(user, "id", None),
    )
    db.add(run)
    db.flush()

    warnings = _formula_warnings(db)
    run.rules_snapshot = {
        "preview": True,
        "unacknowledged_formulas": warnings,
        "employee_count": 0,
    }

    employees = _in_scope_employees(db)
    for emp in employees:
        config = build_pipeline_config(db, emp, pay_period.pay_date)
        earnings = _build_earnings(db, emp, pay_period.pay_date)
        deductions = _build_deductions(db, emp, pay_period.pay_date)
        draft = P.run_pipeline(config, earnings, deductions, [])
        if draft.refused:
            run.status = "failed"
            run.error_code = draft.refusal_code
            run.error_detail = f"employee {emp.employee_code}: {draft.refusal_message}"
            run.completed_at = datetime.now(timezone.utc)
            db.flush()
            raise AppError(draft.refusal_code or "compute_refused", draft.refusal_message or "", 409)
        _write_payslip(db, run, pay_period, emp, draft, status="preview", prior=None)
        run.rules_snapshot["employee_count"] = run.rules_snapshot["employee_count"] + 1

    run.status = "completed"
    run.completed_at = datetime.now(timezone.utc)
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_run", entity_id=run.id, action="preview",
        new_value={"pay_period_id": str(pay_period.id), "employee_count": run.rules_snapshot["employee_count"]},
    )
    return run


def finalise(db: Session, user, pay_period: PayPeriod) -> PayrollRun:
    if pay_period.status == "locked":
        raise AppError("period_locked", "Period is already finalised", 409)

    warnings = _formula_warnings(db)
    if warnings:
        raise AppError(
            "formula_unacknowledged",
            f"cannot finalise until all formulas are acknowledged; missing: {warnings}",
            409,
        )

    _purge_preview(db, pay_period.id)
    ensure_system_earning_types(db)

    prior_final_max = db.scalar(
        select(Payslip.revision).where(Payslip.pay_period_id == pay_period.id)
        .order_by(Payslip.revision.desc())
    ) or 0
    rev = int(prior_final_max) + 1

    run = PayrollRun(
        id=uuid.uuid4(),
        pay_period_id=pay_period.id,
        kind="final",
        revision=rev,
        status="running",
        rules_snapshot={},
        started_by_user_id=getattr(user, "id", None),
    )
    db.add(run)
    db.flush()

    employees = _in_scope_employees(db)
    run.rules_snapshot = {"final": True, "employee_count": 0, "revision": rev}

    for emp in employees:
        prior = db.scalar(
            select(Payslip).where(
                Payslip.pay_period_id == pay_period.id,
                Payslip.employee_id == emp.id,
                Payslip.status == "current",
            )
        )
        config = build_pipeline_config(db, emp, pay_period.pay_date)
        earnings = _build_earnings(db, emp, pay_period.pay_date)
        deductions = _build_deductions(db, emp, pay_period.pay_date)
        draft = P.run_pipeline(config, earnings, deductions, [])
        if draft.refused:
            run.status = "failed"
            run.error_code = draft.refusal_code
            run.error_detail = f"employee {emp.employee_code}: {draft.refusal_message}"
            run.completed_at = datetime.now(timezone.utc)
            db.flush()
            raise AppError(draft.refusal_code or "compute_refused", draft.refusal_message or "", 409)
        _write_payslip(db, run, pay_period, emp, draft, status="current", prior=prior)
        run.rules_snapshot["employee_count"] = run.rules_snapshot["employee_count"] + 1

    pay_period.status = "locked"
    pay_period.row_version += 1
    run.status = "completed"
    run.completed_at = datetime.now(timezone.utc)
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_run", entity_id=run.id, action="finalise",
        new_value={"pay_period_id": str(pay_period.id), "revision": rev,
                   "employee_count": run.rules_snapshot["employee_count"]},
    )
    return run


def reopen(db: Session, user, pay_period: PayPeriod, reason: str) -> PeriodReopenLog:
    if pay_period.status != "locked":
        raise AppError("period_not_locked", "Only a locked period can be reopened", 409)

    later_locked = db.scalar(
        select(PayPeriod).where(
            PayPeriod.status == "locked",
            PayPeriod.pay_date > pay_period.pay_date,
        ).limit(1)
    )
    if later_locked is not None:
        raise AppError(
            "later_period_locked",
            f"reopen later locked period {later_locked.period_key!r} first",
            409,
        )

    prior_rev = db.scalar(
        select(Payslip.revision).where(Payslip.pay_period_id == pay_period.id)
        .order_by(Payslip.revision.desc())
    ) or 0
    new_rev = int(prior_rev) + 1

    log = PeriodReopenLog(
        id=uuid.uuid4(),
        pay_period_id=pay_period.id,
        reopened_by_user_id=getattr(user, "id", None),
        reason=reason,
        prior_revision=int(prior_rev),
        new_revision=new_rev,
    )
    db.add(log)
    pay_period.status = "open"
    pay_period.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="pay_period", entity_id=pay_period.id, action="reopen",
        old_value={"status": "locked"}, new_value={"status": "open", "reason": reason},
    )
    return log
