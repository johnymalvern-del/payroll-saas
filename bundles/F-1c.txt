export const DEFAULT_SITE_TZ = "Africa/Johannesburg";

let currentSiteTz: string = DEFAULT_SITE_TZ;

export function getSiteTz(): string {
  return currentSiteTz;
}

export function setSiteTz(tz: string | null | undefined): void {
  const trimmed = (tz ?? "").trim();
  currentSiteTz = trimmed.length > 0 ? trimmed : DEFAULT_SITE_TZ;
}

import { useEffect } from "react";
import { useQuery } from "@tanstack/react-query";
import { absoluteUrl, wrappedFetch } from "../api/client";
import { setSiteTz } from "../lib/config";
import { useAuth } from "./AuthContext";

export type Me = {
  id: string;
  username: string;
  role: string;
  employee_id: string | null;
  permissions: string[];
  must_change_password: boolean;
  default_timezone: string | null;
};

export const meQueryKey = ["me"] as const;

export async function fetchMe(): Promise<Me> {
  const res = await wrappedFetch(
    new Request(absoluteUrl("/api/v1/auth/me"), { credentials: "include" }),
  );
  if (!res.ok) {
    throw new Error(`/auth/me failed: ${res.status}`);
  }
  return (await res.json()) as Me;
}

export function applyMeSiteTz(me: Me | null | undefined): void {
  setSiteTz(me?.default_timezone ?? null);
}

export function useMe() {
  const { status } = useAuth();
  const query = useQuery<Me, Error>({
    queryKey: meQueryKey,
    queryFn: fetchMe,
    enabled: status === "authenticated",
    staleTime: Infinity,
    retry: 1,
  });

  useEffect(() => {
    if (query.data) {
      applyMeSiteTz(query.data);
    }
    if (status === "unauthenticated") {
      setSiteTz(null);
    }
  }, [query.data, status]);

  return query;
}

import dayjs from "dayjs";
import utc from "dayjs/plugin/utc";
import timezone from "dayjs/plugin/timezone";
import { getSiteTz } from "./config";
import { strings } from "./strings";

dayjs.extend(utc);
dayjs.extend(timezone);

export function suggestionSummary(
  sf: Record<string, unknown>,
): string | null {
  const outcome = typeof sf.outcome === "string" ? sf.outcome : null;
  if (!outcome) return null;
  const tin =
    typeof sf.time_in === "string"
      ? dayjs(sf.time_in).tz(getSiteTz()).format("HH:mm")
      : null;
  const tout =
    typeof sf.time_out === "string"
      ? dayjs(sf.time_out).tz(getSiteTz()).format("HH:mm")
      : null;
  if (outcome === "worked" && tin && tout) {
    return `${strings.exceptions.suggestWorked} ${tin}\u2013${tout}`;
  }
  return outcome.replace(/_/g, " ");
}

import { afterEach, describe, expect, it } from "vitest";
import { DEFAULT_SITE_TZ, getSiteTz, setSiteTz } from "../config";
import { applyMeSiteTz, type Me } from "../../auth/useMe";

const BASE_ME: Me = {
  id: "u1",
  username: "u",
  role: "employee",
  employee_id: null,
  permissions: [],
  must_change_password: false,
  default_timezone: "Europe/London",
};

afterEach(() => {
  setSiteTz(null);
});

describe("site timezone", () => {
  it("defaults to DEFAULT_SITE_TZ", () => {
    expect(getSiteTz()).toBe(DEFAULT_SITE_TZ);
  });

  it("setSiteTz sets a non-empty tz", () => {
    setSiteTz("Europe/London");
    expect(getSiteTz()).toBe("Europe/London");
  });

  it("setSiteTz(null) resets to default", () => {
    setSiteTz("Europe/London");
    setSiteTz(null);
    expect(getSiteTz()).toBe(DEFAULT_SITE_TZ);
  });

  it("applyMeSiteTz reads default_timezone", () => {
    applyMeSiteTz(BASE_ME);
    expect(getSiteTz()).toBe("Europe/London");
  });

  it("applyMeSiteTz(null) resets", () => {
    setSiteTz("Europe/London");
    applyMeSiteTz(null);
    expect(getSiteTz()).toBe(DEFAULT_SITE_TZ);
  });

  it("applyMeSiteTz with null default_timezone resets", () => {
    setSiteTz("Europe/London");
    applyMeSiteTz({ ...BASE_ME, default_timezone: null });
    expect(getSiteTz()).toBe(DEFAULT_SITE_TZ);
  });
});
