"""4A-bis: patch test-only fast paths into security.py, provisioning.py, conftest.py.

Idempotent via _4A_BIS_* markers. Uses binary read/write to preserve line endings.
Run once before scripts/4a_bis_measure.ps1.
"""
import pathlib

SECURITY_PATCH = '''

# --- 4A-bis: _4A_BIS_SECURITY: weak argon2 in test env only; production unchanged ---
import os as _os_4a_sec
from argon2 import PasswordHasher as _PH_4a_sec

if _os_4a_sec.environ.get("PAYROLL_TEST_FAST_HASH") == "1":
    _4a_fast = _PH_4a_sec(time_cost=1, memory_cost=8192, parallelism=1, hash_len=16, salt_len=8)
    for _4a_n in list(globals().keys()):
        _4a_o = globals().get(_4a_n)
        if isinstance(_4a_o, _PH_4a_sec) and _4a_o is not _4a_fast:
            globals()[_4a_n] = _4a_fast

def _4a_current_hasher():
    for _4a_n in ("_hasher", "_PH", "_password_hasher", "password_hasher", "_HASHER"):
        _4a_o = globals().get(_4a_n)
        if isinstance(_4a_o, _PH_4a_sec):
            return _4a_o
    return _PH_4a_sec()
# --- end ---

# END OF FILE
'''

PROVISIONING_PATCH = '''

# --- 4A-bis: _4A_BIS_PROVISIONING: template DB fast path, test-only ---
import os as _os_4a_prov
from pathlib import Path as _Path_4a_prov

_TEST_TEMPLATE_ENABLED = False
_TEMPLATE_NAME_CACHE = None
_TEMPLATE_VERSION_CACHE = None

def _4a_get_template_name():
    from alembic.config import Config
    from alembic.script import ScriptDirectory
    backend_dir = _Path_4a_prov(__file__).resolve().parent.parent.parent
    cfg = Config(str(backend_dir / "alembic.ini"), ini_section="tenant")
    cfg.set_main_option("sqlalchemy.url", "driver://")
    head = ScriptDirectory.from_config(cfg).get_current_head()
    return f"payroll_tpl_{head}".replace(".", "_")

def _4a_bootstrap_template_db():
    global _TEMPLATE_NAME_CACHE, _TEMPLATE_VERSION_CACHE
    if _TEMPLATE_NAME_CACHE is None:
        _TEMPLATE_NAME_CACHE = _4a_get_template_name()
    template = _TEMPLATE_NAME_CACHE
    engine = create_engine(_admin_url("postgres"), future=True, isolation_level="AUTOCOMMIT")
    try:
        with engine.connect() as conn:
            exists = conn.scalar(text("SELECT 1 FROM pg_database WHERE datname=:n"), {"n": template})
            if not exists:
                conn.execute(text(f'CREATE DATABASE "{template}"'))
    finally:
        engine.dispose()
    from alembic import command
    from alembic.config import Config
    backend_dir = _Path_4a_prov(__file__).resolve().parent.parent.parent
    cfg = Config(str(backend_dir / "alembic.ini"), ini_section="tenant")
    cfg.set_main_option("sqlalchemy.url", _admin_url(template))
    command.upgrade(cfg, "head")
    engine = create_engine(_admin_url(template), future=True)
    try:
        with engine.connect() as conn:
            _TEMPLATE_VERSION_CACHE = conn.scalar(text("SELECT version_num FROM alembic_version LIMIT 1"))
    finally:
        engine.dispose()

def _4a_run_migrations_from_template(db_name):
    assert _TEST_TEMPLATE_ENABLED
    assert _TEMPLATE_NAME_CACHE is not None
    assert _TEMPLATE_VERSION_CACHE is not None
    _drop_tenant_database(db_name)
    engine = create_engine(_admin_url("postgres"), future=True, isolation_level="AUTOCOMMIT")
    try:
        with engine.connect() as conn:
            conn.execute(text(f'CREATE DATABASE "{db_name}" TEMPLATE "{_TEMPLATE_NAME_CACHE}"'))
    finally:
        engine.dispose()
    return str(_TEMPLATE_VERSION_CACHE)
# --- end ---

# END OF FILE
'''

CONFTEST_PATCH = '''

# --- 4A-bis: _4A_BIS_CONFTEST: fast/slow markers + template bootstrap + migration swap ---
import os as _os_4a_cf
import pytest as _pytest_4a_cf

_4a_orig_configure = globals().get("pytest_configure")
def pytest_configure(config):
    config.addinivalue_line("markers", "slow: provisions a tenant; -m 'not slow' skips in mid-chapter runs")
    config.addinivalue_line("markers", "fast: pure logic; included in mid-chapter runs")
    if _4a_orig_configure:
        _4a_orig_configure(config)

_4a_orig_modifyitems = globals().get("pytest_collection_modifyitems")
def pytest_collection_modifyitems(config, items):
    if _4a_orig_modifyitems:
        _4a_orig_modifyitems(config, items)
    for _4a_it in items:
        _4a_names = getattr(_4a_it, "fixturenames", []) or []
        if "disposable_tenant" in _4a_names:
            _4a_it.add_marker(_pytest_4a_cf.mark.slow)
        else:
            _4a_it.add_marker(_pytest_4a_cf.mark.fast)

_4a_orig_start = globals().get("pytest_sessionstart")
def pytest_sessionstart(session):
    if _4a_orig_start:
        _4a_orig_start(session)
    if _os_4a_cf.environ.get("PAYROLL_TEST_TEMPLATE_DB") == "1":
        from app.services import provisioning as _4a_p
        _4a_p._TEST_TEMPLATE_ENABLED = True
        _4a_p._run_tenant_migrations = _4a_p._4a_run_migrations_from_template

@pytest.fixture(scope="session", autouse=True)
def _4a_bootstrap_template():
    if _os_4a_cf.environ.get("PAYROLL_TEST_TEMPLATE_DB") == "1":
        from app.services import provisioning as _4a_p
        _4a_p._4a_bootstrap_template_db()
    yield
# --- end ---

# END OF FILE
'''

GUARD_TEST = '''"""4A-bis guard tests: fast paths only in test env; production unchanged."""
from __future__ import annotations
import os
import subprocess
import sys


def _isolated(code, drop_env=()):
    env = {k: v for k, v in os.environ.items() if k not in drop_env}
    return subprocess.run([sys.executable, "-c", code], capture_output=True, text=True, env=env, timeout=30)


def test_argon2_default_strong_without_env():
    r = _isolated(
        "from app.core.security import _4a_current_hasher; h=_4a_current_hasher(); print(h.memory_cost, h.time_cost, h.parallelism)",
        drop_env={"PAYROLL_TEST_FAST_HASH"},
    )
    assert r.returncode == 0, r.stderr
    mc, tc, par = r.stdout.strip().split()
    assert int(mc) >= 32768
    assert int(tc) >= 2
    assert int(par) >= 1


def test_argon2_weak_with_env_isolated():
    env = dict(os.environ)
    env["PAYROLL_TEST_FAST_HASH"] = "1"
    r = subprocess.run(
        [sys.executable, "-c", "from app.core.security import _4a_current_hasher; h=_4a_current_hasher(); print(h.memory_cost, h.time_cost)"],
        capture_output=True, text=True, env=env, timeout=30,
    )
    assert r.returncode == 0, r.stderr
    mc, tc = r.stdout.strip().split()
    assert int(mc) < 32768
    assert int(tc) < 3


def test_template_flag_off_by_default():
    r = _isolated(
        "from app.services.provisioning import _TEST_TEMPLATE_ENABLED; assert not _TEST_TEMPLATE_ENABLED; print('ok')",
        drop_env={"PAYROLL_TEST_TEMPLATE_DB"},
    )
    assert r.returncode == 0, r.stderr


def test_template_flag_off_with_env_but_no_conftest():
    env = dict(os.environ)
    env["PAYROLL_TEST_TEMPLATE_DB"] = "1"
    r = subprocess.run(
        [sys.executable, "-c", "from app.services.provisioning import _TEST_TEMPLATE_ENABLED; assert not _TEST_TEMPLATE_ENABLED; print('ok')"],
        capture_output=True, text=True, env=env, timeout=30,
    )
    assert r.returncode == 0, r.stderr

# END OF FILE
'''


def _patch(path: pathlib.Path, patch_text: str, marker: str) -> bool:
    raw = path.read_bytes()
    if marker.encode() in raw:
        print(f"SKIP: {path.name} (marker present)")
        return False
    text = raw.decode("utf-8")
    if "# END OF FILE" in text:
        text = text.rsplit("# END OF FILE", 1)[0].rstrip() + "\r\n"
    path.write_bytes((text + patch_text).encode("utf-8"))
    print(f"PATCHED: {path}")
    return True


def _write(path: pathlib.Path, content: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(content.encode("utf-8"))
    print(f"WROTE: {path}")


def main() -> int:
    _patch(BACKEND / "app" / "core" / "security.py", SECURITY_PATCH, "_4A_BIS_SECURITY")
    _patch(BACKEND / "app" / "services" / "provisioning.py", PROVISIONING_PATCH, "_4A_BIS_PROVISIONING")
    _patch(BACKEND / "tests" / "conftest.py", CONFTEST_PATCH, "_4A_BIS_CONFTEST")
    _write(BACKEND / "tests" / "test_security_argon2_guard.py", GUARD_TEST)
    print("PATCH DONE")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
