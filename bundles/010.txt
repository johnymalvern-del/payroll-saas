"""Patch people.py (add two missing columns) and conftest.py (fix fast-path enable).

Both patches are idempotent via _4A_BIS_010 marker.
"""
import pathlib
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
BACKEND = ROOT / "backend"


def patch_people() -> int:
    p = BACKEND / "app" / "db" / "tenant" / "models" / "people.py"
    text = p.read_text(encoding="utf-8")
    if "_4A_BIS_010" in text:
        print(f"SKIP: {p.name} already patched")
        return 0
    lines = text.splitlines(keepends=True)
    inserted = False
    for i, line in enumerate(lines):
        if "uif_exempt:" in line and "Mapped" in line:
            indent = len(line) - len(line.lstrip(" "))
            new_lines = [
                " " * indent
                + "contracted_weekly_hours: Mapped[Decimal | None] = mapped_column("
                + "Numeric(5, 2), nullable=True)\n",
                " " * indent
                + "medical_scheme_members: Mapped[int | None] = mapped_column("
                + "Integer, nullable=True)\n",
            ]
            lines[i + 1 : i + 1] = new_lines
            inserted = True
            break
    if not inserted:
        print(f"STOP: {p.name}: uif_exempt line not found")
        return 1
    text = "".join(lines)
    if "# END OF FILE" in text:
        text = text.rsplit("# END OF FILE", 1)[0].rstrip() + "\n"
    text += "\n# _4A_BIS_010 people.py patch\n# END OF FILE\n"
    p.write_text(text, encoding="utf-8")
    print(f"PATCHED: {p.name}")
    return 0


def patch_conftest() -> int:
    p = BACKEND / "tests" / "conftest.py"
    text = p.read_text(encoding="utf-8")
    if "_4A_BIS_010" in text:
        print(f"SKIP: {p.name} already patched")
        return 0
    old = "        _4a_p._4a_bootstrap_template_db()"
    new = (
        "        _4a_p._TEST_TEMPLATE_ENABLED = True\n"
        "        _4a_p._run_tenant_migrations = _4a_p._4a_run_migrations_from_template\n"
        "        _4a_p._4a_bootstrap_template_db()"
    )
    if old not in text:
        print(f"STOP: {p.name}: fixture body pattern not found")
        return 1
    text = text.replace(old, new, 1)
    if "# END OF FILE" in text:
        text = text.rsplit("# END OF FILE", 1)[0].rstrip() + "\n"
    text += "\n# _4A_BIS_010 conftest.py patch\n# END OF FILE\n"
    p.write_text(text, encoding="utf-8")
    print(f"PATCHED: {p.name}")
    return 0


def main() -> int:
    rc = 0
    rc |= patch_people()
    rc |= patch_conftest()
    return rc


if __name__ == "__main__":
    sys.exit(main())

"""Diagnostic: is the test fast path active?

Skips when env var is not set (BEFORE run). Asserts swap state when env
var is set (AFTER run). Failure message names the exact breakage.
"""
from __future__ import annotations

import os

import pytest


def test_fastpath_state():
    if os.environ.get("PAYROLL_TEST_TEMPLATE_DB") != "1":
        pytest.skip("PAYROLL_TEST_TEMPLATE_DB not set")
    from app.services import provisioning
    assert provisioning._TEST_TEMPLATE_ENABLED, (
        "_TEST_TEMPLATE_ENABLED is False: the swap never fired. "
        "Did the conftest fixture run before disposable_tenant?"
    )
    fn = provisioning._run_tenant_migrations
    assert getattr(fn, "__name__", "") == "_4a_run_migrations_from_template", (
        f"_run_tenant_migrations is {getattr(fn, '__name__', fn)!r}, "
        f"expected _4a_run_migrations_from_template"
    )
    assert provisioning._TEMPLATE_NAME_CACHE is not None, (
        "template name never cached: bootstrap fixture did not run"
    )
    assert provisioning._TEMPLATE_VERSION_CACHE is not None, (
        "template version never cached: bootstrap fixture did not finish"
    )
