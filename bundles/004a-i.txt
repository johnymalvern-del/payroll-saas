"""Payroll earnings-generation formulas (Chapter 4, Bundle 004a-i).

Each gen_* function is pure: no DB, no clock, no side effects. Inputs are
explicit and, where a statutory multiplier is needed, the parameter has NO
default — the pipeline must supply it from an acknowledged source. This
respects the "no invented statutory values" rule while letting the functions
be tested today with clearly-fake fixture values.

FORMULA_REGISTRY lists every formula. Each must be acknowledged by the
payroll admin (F8) before first finalise; readiness endpoint surfaces which
are outstanding.
"""
from __future__ import annotations

from datetime import date
from decimal import Decimal, ROUND_HALF_UP


VALID_PREMIUM_MODES = ("full", "none", "hourly_equivalent")


FORMULA_REGISTRY: dict[str, str] = {
    "gen_base_salary":
        "Full-period base monthly salary",
    "gen_part_period_joiner_leaver":
        "Prorated base salary for mid-period hire or termination (calendar-day fraction)",
    "gen_unpaid_absence":
        "Daily rate x unpaid absence days",
    "gen_salaried_premiums":
        "Salaried premium pay per shift premium mode (full/none/hourly_equivalent)",
    "gen_paid_leave":
        "Daily rate x paid leave days",
    "gen_public_holiday_unworked":
        "Unworked public holiday pay per country policy (requires explicit multiplier)",
}


def _q(amount: Decimal) -> Decimal:
    return amount.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


def gen_base_salary(*, monthly_amount: Decimal) -> Decimal:
    """Full-period base salary. Monthly amount quantised to 2 dp half-up."""
    return _q(monthly_amount)


def gen_part_period_joiner_leaver(
    *,
    monthly_amount: Decimal,
    hired_on: date | None,
    terminated_on: date | None,
    period_start: date,
    period_end: date,
) -> Decimal:
    """Calendar-day proration for joiner/leaver within [period_start, period_end]."""
    full_start = max(period_start, hired_on) if hired_on else period_start
    full_end = min(period_end, terminated_on) if terminated_on else period_end
    if full_end < full_start:
        return Decimal("0.00")
    days_present = (full_end - full_start).days + 1
    days_in_period = (period_end - period_start).days + 1
    return _q(monthly_amount * Decimal(days_present) / Decimal(days_in_period))


def gen_unpaid_absence(
    *,
    daily_rate: Decimal,
    unpaid_days: Decimal,
) -> Decimal:
    """Daily rate x unpaid absence days."""
    return _q(daily_rate * unpaid_days)


def gen_salaried_premiums(
    *,
    salaried_premium_mode: str,
    hourly_rate: Decimal,
    premium_hours: Decimal,
    premium_multiplier: Decimal | None = None,
) -> Decimal:
    """Salaried premium pay.

    - 'full': salary already covers it -> 0
    - 'none': never any premium -> 0
    - 'hourly_equivalent': hourly_rate * premium_hours * premium_multiplier
      premium_multiplier has no default; the pipeline must supply it.
    """
    if salaried_premium_mode in ("full", "none"):
        return Decimal("0.00")
    if salaried_premium_mode == "hourly_equivalent":
        if premium_multiplier is None:
            raise ValueError(
                "salaried_premium_mode='hourly_equivalent' requires "
                "premium_multiplier (no default by design)"
            )
        return _q(hourly_rate * premium_hours * premium_multiplier)
    raise ValueError(f"unknown salaried_premium_mode: {salaried_premium_mode!r}")


def gen_paid_leave(
    *,
    daily_rate: Decimal,
    paid_leave_days: Decimal,
) -> Decimal:
    """Daily rate x paid leave days."""
    return _q(daily_rate * paid_leave_days)


def gen_public_holiday_unworked(
    *,
    daily_rate: Decimal,
    holiday_days: Decimal,
    public_holiday_pay_multiplier: Decimal | None = None,
) -> Decimal:
    """Unworked public holiday pay.

    The statutory multiplier varies by jurisdiction and cannot be baked in.
    Caller must supply public_holiday_pay_multiplier; None raises.
    """
    if public_holiday_pay_multiplier is None:
        raise ValueError(
            "gen_public_holiday_unworked requires public_holiday_pay_multiplier "
            "(no default by design)"
        )
    return _q(daily_rate * holiday_days * public_holiday_pay_multiplier)


def hourly_rate_from_monthly(
    *,
    monthly_salary: Decimal,
    contracted_weekly_hours: Decimal,
) -> Decimal:
    """D4 formula (subject to accountant confirmation).

    monthly_hours = contracted_weekly_hours * 52 / 12
    hourly_rate   = monthly_salary / monthly_hours
    Returned to 4 dp half-up.
    """
    if contracted_weekly_hours <= 0:
        raise ValueError("contracted_weekly_hours must be positive")
    monthly_hours = contracted_weekly_hours * Decimal(52) / Decimal(12)
    return (monthly_salary / monthly_hours).quantize(
        Decimal("0.0001"), rounding=ROUND_HALF_UP
    )

"""Formula readiness + acknowledgement (Chapter 4, Bundle 004a-i).

F8: the readiness screen warns until each formula is acknowledged by the
payroll admin; the first finalise per tenant refuses if any formula is
unacknowledged. Acks are stored in payroll_settings.formula_acks as JSONB.
"""
from __future__ import annotations

import uuid
from datetime import datetime, timezone
from decimal import Decimal
from typing import Annotated, Any

from fastapi import APIRouter, Depends
from sqlalchemy import select

from app.api.deps import TenantDbDep
from app.api.v1.payroll_formulas_schemas import (
    AcknowledgeOut,
    FormulaInfo,
    ReadinessOut,
)
from app.core.errors import AppError
from app.core.permissions import Permission, require_permission
from app.db.tenant.models.payroll_settings import PayrollSettings
from app.services.payroll_audit import write_audit
from app.services.payroll_formulas import FORMULA_REGISTRY

router = APIRouter(prefix="/payroll-formulas", tags=["payroll-formulas"])


@router.get("/readiness", response_model=ReadinessOut)
def get_readiness(
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_PREVIEW))],
) -> ReadinessOut:
    settings = db.scalar(select(PayrollSettings).limit(1))
    acks = (settings.formula_acks or {}) if settings is not None else {}
    items: list[FormulaInfo] = []
    for code, description in FORMULA_REGISTRY.items():
        rec = acks.get(code) or {}
        items.append(FormulaInfo(
            code=code,
            description=description,
            verified=code in acks,
            acknowledged_at=rec.get("at"),
            acknowledged_by=rec.get("by"),
        ))
    return ReadinessOut(items=items, all_acknowledged=all(i.verified for i in items))


@router.post("/{code}/acknowledge", response_model=AcknowledgeOut)
def acknowledge_formula(
    code: str,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_FINALISE))],
) -> AcknowledgeOut:
    if code not in FORMULA_REGISTRY:
        raise AppError("unknown_formula", f"Formula {code!r} not registered", 404)
    settings = db.scalar(select(PayrollSettings).limit(1))
    if settings is None:
        settings = PayrollSettings(
            id=uuid.uuid4(),
            net_pay_floor=Decimal("0.00"),
            formula_acks={},
            row_version=1,
            updated_by_user_id=getattr(user, "id", None),
        )
        db.add(settings)
        db.flush()
    old_acks = dict(settings.formula_acks or {})
    now = datetime.now(timezone.utc)
    new_acks = dict(old_acks)
    new_acks[code] = {
        "at": now.isoformat(),
        "by": str(getattr(user, "id", "")) or None,
    }
    settings.formula_acks = new_acks
    settings.row_version += 1
    settings.updated_by_user_id = getattr(user, "id", None)
    write_audit(
        db, user,
        entity_type="payroll_formula_ack",
        entity_id=settings.id,
        action="acknowledge",
        old_value={"acked": sorted(old_acks.keys())},
        new_value={"acked": sorted(new_acks.keys()), "code": code},
    )
    db.commit()
    return AcknowledgeOut(
        code=code, acknowledged=True, acknowledged_at=now,
    )

    engine = get_tenant_engine(disposable_tenant.tenant_db_name)
    session = Session(engine)
    try:
        yield session, engine
    finally:
        session.close()
        engine.dispose()


def _admin():
    return SimpleNamespace(role="payroll_admin", id=uuid.uuid4(), employee_id=None)


def test_readiness_empty(tenant_session):
    session, _ = tenant_session
    out = get_readiness(db=session, user=_admin())
    assert len(out.items) == 6
    assert all(not i.verified for i in out.items)
    assert out.all_acknowledged is False


def test_acknowledge_creates_settings_when_absent(tenant_session):
    session, _ = tenant_session
    assert session.query(PayrollSettings).count() == 0
    ack = acknowledge_formula(
        code="gen_base_salary", db=session, user=_admin(),
    )
    assert ack.code == "gen_base_salary"
    assert ack.acknowledged is True
    rows = session.query(PayrollSettings).all()
    assert len(rows) == 1
    assert "gen_base_salary" in rows[0].formula_acks


def test_acknowledge_updates_existing(tenant_session):
    session, _ = tenant_session
    session.add(PayrollSettings(
        id=uuid.uuid4(),
        net_pay_floor=__import__("decimal").Decimal("0.00"),
        formula_acks={},
        row_version=1,
    ))
    session.commit()
    acknowledge_formula(code="gen_base_salary", db=session, user=_admin())
    acknowledge_formula(code="gen_unpaid_absence", db=session, user=_admin())
    row = session.query(PayrollSettings).one()
    assert set(row.formula_acks.keys()) == {"gen_base_salary", "gen_unpaid_absence"}
    assert row.row_version == 3


def test_acknowledge_unknown_formula_raises(tenant_session):
    session, _ = tenant_session
    with pytest.raises(AppError) as exc_info:
        acknowledge_formula(code="not_a_real_formula", db=session, user=_admin())
    assert exc_info.value.status_code == 404
    assert exc_info.value.code == "unknown_formula"


def test_all_six_can_be_acknowledged_and_readiness_flips(tenant_session):
    session, _ = tenant_session
    admin = _admin()
    for code in FORMULA_REGISTRY:
        acknowledge_formula(code=code, db=session, user=admin)
    out = get_readiness(db=session, user=_admin())
    assert out.all_acknowledged is True
    assert all(i.verified for i in out.items)
