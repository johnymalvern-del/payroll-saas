function warnShapeMismatch(data: unknown): void {
  if (!data || typeof data !== "object") {
    console.warn("[attendance] sheet response is not an object", data);
    return;
  }
  const obj = data as Record<string, unknown>;
  if (!Array.isArray(obj.items)) {
    console.warn("[attendance] sheet response missing items array", Object.keys(obj));
    return;
  }
  const emp = obj.items[0];
  if (!emp || typeof emp !== "object") return;
  const wantEmp = ["employee_id", "employee_code", "employee_name", "days"];
  const missEmp = wantEmp.filter((k) => !(k in (emp as Record<string, unknown>)));
  if (missEmp.length > 0) {
    console.warn(
      "[attendance] sheet employee missing keys:",
      missEmp,
      "got:",
      Object.keys(emp as Record<string, unknown>),
    );
  }
  const days = (emp as Record<string, unknown>).days;
  if (Array.isArray(days) && days.length > 0) {
    const c = days[0];
    if (c && typeof c === "object") {
      const wantCell = [
        "work_date",
        "status",
        "worked_minutes",
        "flags",
        "open_exception_count",
        "locked",
        "reviewed",
      ];
      const missCell = wantCell.filter((k) => !(k in (c as Record<string, unknown>)));
      if (missCell.length > 0) {
        console.warn(
          "[attendance] sheet cell missing keys:",
          missCell,
          "got:",
          Object.keys(c as Record<string, unknown>),
        );
      }
    }
  }
}

export async function fetchSheet(params: {
  fromDate: string;
  toDate: string;
  siteId?: string | null;
  branchId?: string | null;
  departmentId?: string | null;
  includeTerminated?: boolean;
  limit?: number;
  cursor?: string | null;
}): Promise<SheetOut> {
  const qs = new URLSearchParams();
  qs.set("from_date", params.fromDate);
  qs.set("to_date", params.toDate);
  if (params.siteId) qs.set("site_id", params.siteId);
  if (params.branchId) qs.set("branch_id", params.branchId);
  if (params.departmentId) qs.set("department_id", params.departmentId);
  if (params.includeTerminated) qs.set("include_terminated", "true");
  qs.set("limit", String(params.limit ?? 50));
  if (params.cursor) qs.set("cursor", params.cursor);
  const res = await wrappedFetch(
    new Request(`/api/v1/attendance/sheet?${qs.toString()}`, {
      credentials: "include",
    }),
  );
  if (!res.ok) {
    throw new Error(`sheet fetch failed: ${res.status}`);
  }
  const data = await res.json();
  warnShapeMismatch(data);
  return data as SheetOut;
}

async function fetchRefs(path: string): Promise<Ref[]> {
  const res = await wrappedFetch(
    new Request(`/api/v1${path}?limit=500`, { credentials: "include" }),
  );
  if (!res.ok) return [];
  const body = (await res.json()) as { items?: Ref[] };
  return body.items ?? [];
}

export function fetchBranches(): Promise<Ref[]> {
  return fetchRefs("/branches");
}

export function fetchSites(): Promise<Ref[]> {
  return fetchRefs("/sites");
}

export function fetchDepartments(): Promise<Ref[]> {
  return fetchRefs("/departments");
}

import dayjs from "dayjs";
import type { SheetCell, SheetEmployee } from "../../api/attendance";

type Props = {
  fromDate: string;
  toDate: string;
  employees: SheetEmployee[];
  attendanceStatus: string[];

function buildDays(fromDate: string, toDate: string): string[] {
  if (!fromDate || !toDate) return [];
  const start = dayjs(fromDate);
  const end = dayjs(toDate);
  const days: string[] = [];
  let cur = start;
  while (cur.isBefore(end) || cur.isSame(end, "day")) {
    days.push(cur.format("YYYY-MM-DD"));
    cur = cur.add(1, "day");
  }
  return days;
}

function cellFor(employee: SheetEmployee, date: string): SheetCell | null {
  return employee.days.find((c) => c.work_date === date) ?? null;
}

function statusColor(status: string | null): string {
  switch (status) {
    case "worked":
      return "green";
    case "leave":
      return "blue";
    case "sick":
      return "orange";
    case "absent":
      return "red";
    case "day_off":
      return "gray";
    default:
      return "gray";
  }
}

function statusLabel(status: string | null): string {
  if (!status) return "\u2014";
  return status.replace("_", " ");
}

function CellView({ cell }: { cell: SheetCell | null }) {
  if (!cell) {
    return (
      <Text size="xs" c="dimmed">
        {"\u2014"}
      </Text>
    );
  }
  const lines: string[] = [statusLabel(cell.status)];
  if (cell.worked_minutes !== null && cell.worked_minutes !== undefined) {
    lines.push(`${cell.worked_minutes} worked`);
  }
  if (cell.open_exception_count > 0) {
    lines.push(`${cell.open_exception_count} open exception(s)`);
  }
  if (cell.flags && cell.flags.length > 0) {
    lines.push(`flags: ${cell.flags.join(", ")}`);
  }
  if (cell.locked) lines.push("locked");
  if (cell.reviewed) lines.push("reviewed");

  return (
    <Tooltip label={lines.join(" \u00b7 ")} withinPortal>
      <Group gap={4} wrap="nowrap" justify="center">
        {cell.status ? (
          <Badge size="xs" color={statusColor(cell.status)} variant="light">
            {statusLabel(cell.status)}
          </Badge>
        ) : (
          <Text size="xs" c="dimmed">
            {"\u2014"}
          </Text>
        )}
        {cell.open_exception_count > 0 ? (
          <Text size="xs" c="red" fw={700}>
            !
          </Text>
        ) : null}
      </Group>
    </Tooltip>
  );
}

export function MonthlyGrid({ fromDate, toDate, employees, attendanceStatus }: Props) {
  const days = buildDays(fromDate, toDate);
  const filterSet = attendanceStatus.length > 0 ? new Set(attendanceStatus) : null;

  const rows = employees
    .map((emp) => {
      const cells = days.map((d) => cellFor(emp, d));
      if (filterSet) {
        const anyMatch = cells.some(
          (c) => c && c.status && filterSet.has(c.status),
        );
        if (!anyMatch) return null;
      }
      return { emp, cells };
    })
    .filter(
      (r): r is { emp: SheetEmployee; cells: (SheetCell | null)[] } => r !== null,
    );

  if (rows.length === 0) {
    return <Text c="dimmed">{strings.attendance.empty}</Text>;
  }

  return (
    <ScrollArea type="auto" offsetScrollbars>
      <Table
        striped
        withTableBorder
        withColumnBorders
        stickyHeader
        highlightOnHover
      >
        <Table.Thead>
          <Table.Tr>
            <Table.Th
              style={{
                minWidth: 200,
                position: "sticky",
                left: 0,
                background: "white",
                zIndex: 2,
              }}
            >
              {strings.attendance.employee}
            </Table.Th>
            {days.map((d) => (
              <Table.Th key={d} style={{ minWidth: 70, textAlign: "center" }}>
                {dayjs(d).format("DD")}
              </Table.Th>
            ))}
          </Table.Tr>
        </Table.Thead>
        <Table.Tbody>
          {rows.map(({ emp, cells }) => (
            <Table.Tr key={emp.employee_id}>
              <Table.Td
                style={{
                  position: "sticky",
                  left: 0,
                  background: "white",
                  zIndex: 1,
                }}
              >
                <Text size="sm" fw={500}>
                  {emp.employee_name}
                </Text>
                <Text size="xs" c="dimmed">
                  {emp.employee_code}
                </Text>
              </Table.Td>
              {cells.map((c, i) => (
                <Table.Td key={days[i]} style={{ textAlign: "center" }}>
                  <CellView cell={c} />
                </Table.Td>
              ))}
            </Table.Tr>
          ))}
        </Table.Tbody>
      </Table>
    </ScrollArea>
  );
}

export function AttendanceSheetPage() {
  const [fromDate, setFromDate] = useState<Date | null>(
    dayjs().startOf("month").toDate(),
  );
  const [toDate, setToDate] = useState<Date | null>(
    dayjs().endOf("month").toDate(),
  );
  const [branchId, setBranchId] = useState<string | null>(null);
  const [siteId, setSiteId] = useState<string | null>(null);
  const [departmentId, setDepartmentId] = useState<string | null>(null);
  const [attendanceStatus, setAttendanceStatus] = useState<string[]>([]);
  const [includeTerminated, setIncludeTerminated] = useState(false);
  const [cursor, setCursor] = useState<string | null>(null);

  const fromStr = useMemo(
    () => (fromDate ? dayjs(fromDate).format("YYYY-MM-DD") : ""),
    [fromDate],
  );
  const toStr = useMemo(
    () => (toDate ? dayjs(toDate).format("YYYY-MM-DD") : ""),
    [toDate],
  );
  const rangeOk = useMemo(() => {
    if (!fromDate || !toDate) return false;
    const diff = dayjs(toDate).diff(dayjs(fromDate), "day");
    return diff >= 0 && diff <= 30;
  }, [fromDate, toDate]);

  const branches = useQuery({
    queryKey: ["refs", "branches"],
    queryFn: fetchBranches,
  });
  const sites = useQuery({
    queryKey: ["refs", "sites"],
    queryFn: fetchSites,
  });
  const departments = useQuery({
    queryKey: ["refs", "departments"],
    queryFn: fetchDepartments,
  });

    queryKey: [
      "attendance",
      "sheet",
      fromStr,
      toStr,
      branchId,
      siteId,
      departmentId,
      includeTerminated,
      cursor,

    queryFn: () =>
      fetchSheet({
        fromDate: fromStr,
        toDate: toStr,
        branchId,
        siteId,
        departmentId,
        includeTerminated,
        limit: PAGE_SIZE,
        cursor,
      }),
    enabled: rangeOk,
  });

  const onReset = () => {
    setFromDate(dayjs().startOf("month").toDate());
    setToDate(dayjs().endOf("month").toDate());

    setDepartmentId(null);

          onFromDate={(d) => {
            setFromDate(d);
            setCursor(null);
          }}
          onToDate={(d) => {
            setToDate(d);
            setCursor(null);
          }}
          onBranch={(v) => {
            setBranchId(v);
            setCursor(null);
          }}
          onSite={(v) => {
            setSiteId(v);

          }}

## Important contracts

- /auth/login returns `{access_token, refresh_token, token_type, expires_in}` only. No must_change_password. UI uses /auth/me.
- /auth/me returns `{id, username, role, employee_id, permissions[], must_change_password}`.
- Usernames: normalise_username strips separators when a digit is present. Use `paone` not `pa-1`.
- Refresh uses `csrf_token` cookie + `csrf_protection`. Client does NOT yet send the CSRF header. Refresh will 403 until fixed. Flagged.
- Attendance sheet: `GET /api/v1/attendance/sheet?from_date=&to_date=&site_id=&branch_id=&department_id=&include_terminated=&limit=&cursor=` → `{items: SheetEmployee[], next_cursor}` where SheetEmployee = `{employee_id, employee_code, employee_name, site_id, department_id, days: SheetCell[]}` and SheetCell = `{work_date, status, worked_minutes, normal_minutes, overtime_minutes, flags, open_exception_count, locked, reviewed}`. Client logs a console.warn if the shape doesn't match at runtime. Field names assumed from the backend spec; verify on first live run.

## Verified gaps

## Decisions

- Attendance sheet uses `wrappedFetch` + manual type cast (not `api.GET`) until `npm run api:types` is run against a live backend. When the schema is generated, F-2a's `api/attendance.ts` types become authoritative.
- Attendance status chip is a client-side post-filter (`attendanceStatus` in page state) layered over the sheet response.
- Employee-status filter is `includeTerminated` only (backend supports that).
- Sheet cell tooltip surfaces minutes, flags, open exception count, locked, reviewed — colour is never the only signal (status text + tooltip).
- Pagination: "first page" (clears cursor) and "next page" (uses next_cursor). No "previous" — cursor is forward-only.
- No row virtualisation yet (20 employees × 31 days = 620 cells, fine at this size).
- No day panel yet — F-2b.
