"""Fix people.py with builtin-safe annotations.

Uses `float | None` (builtin) instead of `Decimal | None` and `sa.Numeric` /
`sa.Integer` (or bare names) so no new imports are needed. Byte-safe write.
"""
import ast
import pathlib
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
BACKEND = ROOT / "backend"


def _read_norm(path):
    text = path.read_bytes().decode("utf-8")
    return text.replace("\r\n", "\n").replace("\r", "\n")


def _write_norm(path, text):
    ast.parse(text)  # raises before writing if broken
    path.write_bytes(text.encode("utf-8"))


def patch_people():
    p = BACKEND / "app" / "db" / "tenant" / "models" / "people.py"
    text = _read_norm(p)
    if "_4A_BIS_011" in text:
        print(f"SKIP: {p.name} already patched")
        return 0

    head = text.split("\nclass ")[0]
    has_sa = "import sqlalchemy as sa" in head

    if has_sa:
        num_expr = "sa.Numeric(5, 2)"
        int_expr = "sa.Integer"
    elif "Numeric" in head and "Integer" in head:
        num_expr = "Numeric(5, 2)"
        int_expr = "Integer"
    else:
        print(f"STOP: {p.name}: neither 'import sqlalchemy as sa' nor "
              f"'Numeric'+'Integer' imports found in header")
        print("HEADER (first 30 lines):")
        for i, line in enumerate(head.splitlines()[:30], 1):
            print(f"  {i:3d}| {line}")
        return 1

    marker = "    uif_exempt:"
    idx = text.find(marker)
    if idx == -1:
        print(f"STOP: {p.name}: 'uif_exempt:' line not found")
        return 1
    eol = text.find("\n", idx)
    if eol == -1:
        eol = len(text) - 1
    insert_at = eol + 1

    new_lines = (
        f"    contracted_weekly_hours: Mapped[float | None] = mapped_column("
        f"{num_expr}, nullable=True)\n"
        f"    medical_scheme_members: Mapped[int | None] = mapped_column("
        f"{int_expr}, nullable=True)\n"
    )
    new_text = (
        text[:insert_at] + new_lines + text[insert_at:]
        + "\n# _4A_BIS_011 people\n"
    )
    _write_norm(p, new_text)
    print(f"PATCHED: {p.name} (using {num_expr} / {int_expr})")
    return 0


def main():
    return patch_people()


if __name__ == "__main__":
    sys.exit(main())
