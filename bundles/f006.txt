function warn(where: string, data: unknown, want: string[]): void {
  if (!data || typeof data !== "object") {
    console.warn(`[attendance] ${where}: not an object`, data);
    return;
  }
  const miss = want.filter((k) => !(k in (data as Record<string, unknown>)));
  if (miss.length > 0) {
    console.warn(`[attendance] ${where} missing keys:`, miss, "got:",
      Object.keys(data as Record<string, unknown>));
  }
}

export async function fetchSheet(params: {
  fromDate: string;
  toDate: string;
  siteId?: string | null;
  branchId?: string | null;
  departmentId?: string | null;
  includeTerminated?: boolean;
  limit?: number;
  cursor?: string | null;
}): Promise<SheetOut> {
  const qs = new URLSearchParams();
  qs.set("from_date", params.fromDate);
  qs.set("to_date", params.toDate);
  if (params.siteId) qs.set("site_id", params.siteId);
  if (params.branchId) qs.set("branch_id", params.branchId);
  if (params.departmentId) qs.set("department_id", params.departmentId);
  if (params.includeTerminated) qs.set("include_terminated", "true");
  qs.set("limit", String(params.limit ?? 50));
  if (params.cursor) qs.set("cursor", params.cursor);
  const res = await wrappedFetch(
    new Request(absoluteUrl(`/api/v1/attendance/sheet?${qs.toString()}`), {
      credentials: "include",
    }),
  );
  if (!res.ok) {
    throw new Error(`sheet fetch failed: ${res.status}`);
  }
  const data = await res.json();
  warn("sheet response", data, ["items"]);
  return normaliseSheetOut(data);
}

export async function markReviewed(req: MarkReviewedRequest): Promise<MarkReviewedOut> {
  const body: Record<string, unknown> = {
    from_date: req.fromDate,
    to_date: req.toDate,
  };
  if (req.siteId) body.site_id = req.siteId;
  if (req.branchId) body.branch_id = req.branchId;
  if (req.departmentId) body.department_id = req.departmentId;
  if (req.includeTerminated) body.include_terminated = true;
  const res = await wrappedFetch(
    new Request(absoluteUrl("/api/v1/attendance/mark_reviewed"), {
      method: "POST",
      credentials: "include",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    }),
  );
  if (!res.ok) {
    throw new Error(`mark reviewed failed: ${res.status}`);
  }
  const data = (await res.json().catch(() => ({}))) as Record<string, unknown>;
  const refused = Array.isArray(data.refused_days)
    ? (data.refused_days as RefusedDay[])
    : [];
  const count =
    typeof data.reviewed_count === "number" ? data.reviewed_count : 0;
  return { reviewed_count: count, refused_days: refused };
}

  const body = (await res.json().catch(() => ({}))) as { items?: unknown };
  return Array.isArray(body.items) ? (body.items as Ref[]) : [];
}

import { Badge, Group, ScrollArea, Table, Text, Tooltip } from "@mantine/core";
import dayjs from "dayjs";
import type { SheetCell, SheetEmployee } from "../../api/attendance";
import { strings } from "../../lib/strings";

type Props = {
  fromDate: string;
  toDate: string;
  employees: SheetEmployee[];
  attendanceStatus: string[];
  onCellClick?: (employeeId: string, workDate: string) => void;
};

function buildDays(fromDate: string, toDate: string): string[] {
  if (!fromDate || !toDate) return [];
  const start = dayjs(fromDate);
  const end = dayjs(toDate);
  const days: string[] = [];
  let cur = start;
  while (cur.isBefore(end) || cur.isSame(end, "day")) {
    days.push(cur.format("YYYY-MM-DD"));
    cur = cur.add(1, "day");
  }
  return days;
}

function cellFor(employee: SheetEmployee, date: string): SheetCell | null {
  return employee.days.find((c) => c.work_date === date) ?? null;
}

function statusColor(status: string | null): string {
  switch (status) {
    case "worked":
      return "green";
    case "leave":
      return "blue";
    case "sick":
      return "orange";
    case "absent":
      return "red";
    case "day_off":
      return "gray";
    default:
      return "gray";
  }
}

function statusLabel(status: string | null): string {
  if (!status) return "\u2014";
  return status.replace("_", " ");
}

function CellView({
  cell,
  employeeId,
  workDate,
  onCellClick,
}: {
  cell: SheetCell | null;
  employeeId: string;
  workDate: string;
  onCellClick?: (employeeId: string, workDate: string) => void;
}) {
  if (!cell) {
    return (
      <Text size="xs" c="dimmed">
        {"\u2014"}
      </Text>
    );
  }
  const lines: string[] = [statusLabel(cell.status)];
  if (cell.worked_minutes !== null && cell.worked_minutes !== undefined) {
    lines.push(`${cell.worked_minutes} worked`);
  }
  if (cell.open_exception_count > 0) {
    lines.push(`${cell.open_exception_count} open exception(s)`);
  }
  if (cell.flags && cell.flags.length > 0) {
    lines.push(`flags: ${cell.flags.join(", ")}`);
  }
  if (cell.locked) lines.push("locked");
  if (cell.reviewed) lines.push("reviewed");

  const content = (
    <Group gap={4} wrap="nowrap" justify="center">
      {cell.status ? (
        <Badge size="xs" color={statusColor(cell.status)} variant="light">
          {statusLabel(cell.status)}
        </Badge>
      ) : (
        <Text size="xs" c="dimmed">
          {"\u2014"}
        </Text>
      )}
      {cell.open_exception_count > 0 ? (
        <Text size="xs" c="red" fw={700}>
          !
        </Text>
      ) : null}
    </Group>
  );

  return (
    <Tooltip label={lines.join(" \u00b7 ")} withinPortal>
      <div
        role={onCellClick ? "button" : undefined}
        tabIndex={onCellClick ? 0 : undefined}
        onClick={() => onCellClick?.(employeeId, workDate)}
        onKeyDown={(e) => {
          if (!onCellClick) return;
          if (e.key === "Enter" || e.key === " ") {
            e.preventDefault();
            onCellClick(employeeId, workDate);
          }
        }}
        style={{ cursor: onCellClick ? "pointer" : "default" }}
      >
        {content}
      </div>
    </Tooltip>
  );
}

export function MonthlyGrid({
  fromDate,
  toDate,
  employees,
  attendanceStatus,
  onCellClick,
}: Props) {
  const days = buildDays(fromDate, toDate);
  const filterSet = attendanceStatus.length > 0 ? new Set(attendanceStatus) : null;

  const rows = employees
    .map((emp) => {
      const cells = days.map((d) => cellFor(emp, d));
      if (filterSet) {
        const anyMatch = cells.some(
          (c) => c && c.status && filterSet.has(c.status),
        );
        if (!anyMatch) return null;
      }
      return { emp, cells };
    })
    .filter(
      (r): r is { emp: SheetEmployee; cells: (SheetCell | null)[] } => r !== null,
    );

  if (rows.length === 0) {
    return <Text c="dimmed">{strings.attendance.empty}</Text>;
  }

  return (
    <ScrollArea type="auto" offsetScrollbars>
      <Table striped withTableBorder withColumnBorders stickyHeader highlightOnHover>
        <Table.Thead>
          <Table.Tr>
            <Table.Th
              style={{
                minWidth: 200,
                position: "sticky",
                left: 0,
                background: "white",

  branchId: string | null;

export function MarkReviewedModal({
  opened,
  fromDate,
  toDate,
  siteId,
  branchId,
  departmentId,
  includeTerminated,
  onClose,
}: Props) {
  const [result, setResult] = useState<MarkReviewedOut | null>(null);
  const mutation = useMutation({
    mutationFn: () =>
      markReviewed({
        fromDate,
        toDate,
        siteId,
        branchId,
        departmentId,
        includeTerminated,
      }),

  });

    setResult(null);
    mutation.reset();
    onClose();

      onClose={close}

type View = "monthly" | "daily";

export function AttendanceSheetPage() {
  const [view, setView] = useState<View>("monthly");
  const [fromDate, setFromDate] = useState<Date | null>(
    dayjs().startOf("month").toDate(),
  );
  const [toDate, setToDate] = useState<Date | null>(
    dayjs().endOf("month").toDate(),
  );
  const [dailyDate, setDailyDate] = useState<Date | null>(dayjs().toDate());
  const [branchId, setBranchId] = useState<string | null>(null);
  const [siteId, setSiteId] = useState<string | null>(null);
  const [departmentId, setDepartmentId] = useState<string | null>(null);
  const [attendanceStatus, setAttendanceStatus] = useState<string[]>([]);
  const [includeTerminated, setIncludeTerminated] = useState(false);
  const [cursor, setCursor] = useState<string | null>(null);
  const [selectedDay, setSelectedDay] = useState<{ employeeId: string; workDate: string } | null>(null);
  const [markOpen, setMarkOpen] = useState(false);

  const fromStr = useMemo(
    () => (fromDate ? dayjs(fromDate).format("YYYY-MM-DD") : ""),
    [fromDate],
  );
  const toStr = useMemo(
    () => (toDate ? dayjs(toDate).format("YYYY-MM-DD") : ""),
    [toDate],
  );
  const dailyStr = useMemo(
    () => (dailyDate ? dayjs(dailyDate).format("YYYY-MM-DD") : ""),
    [dailyDate],
  );
  const rangeOk = useMemo(() => {
    if (!fromDate || !toDate) return false;
    const diff = dayjs(toDate).diff(dayjs(fromDate), "day");
    return diff >= 0 && diff <= 30;
  }, [fromDate, toDate]);

  const branches = useQuery({ queryKey: ["refs", "branches"], queryFn: fetchBranches });
  const sites = useQuery({ queryKey: ["refs", "sites"], queryFn: fetchSites });
  const departments = useQuery({ queryKey: ["refs", "departments"], queryFn: fetchDepartments });

  const sheet = useQuery({
    queryKey: ["attendance", "sheet", fromStr, toStr, branchId, siteId, departmentId, includeTerminated, cursor],
    queryFn: () =>
      fetchSheet({ fromDate: fromStr, toDate: toStr, branchId, siteId, departmentId, includeTerminated, limit: PAGE_SIZE, cursor }),
    enabled: view === "monthly" && rangeOk,
  });

  const daily = useQuery({
    queryKey: ["attendance", "daily", dailyStr, branchId, siteId, departmentId, includeTerminated, cursor],
    queryFn: () =>
      fetchDaily({ date: dailyStr, branchId, siteId, departmentId, includeTerminated, limit: PAGE_SIZE, cursor }),
    enabled: view === "daily" && !!dailyStr,
  });

  const onReset = () => {
    setFromDate(dayjs().startOf("month").toDate());
    setToDate(dayjs().endOf("month").toDate());
    setDailyDate(dayjs().toDate());
    setBranchId(null);
    setSiteId(null);
    setDepartmentId(null);
    setAttendanceStatus([]);
    setIncludeTerminated(false);
    setCursor(null);
  };

  const openDay = (employeeId: string, workDate: string) => {
    setSelectedDay({ employeeId, workDate });
  };

  return (
    <Stack gap="md">
      <Group justify="space-between" align="center">
        <Title order={2}>{strings.attendance.title}</Title>
        <SegmentedControl
          value={view}
          onChange={(v) => {
            setView(v as View);
            setCursor(null);
          }}
          data={[
            { label: strings.attendance.viewMonthly, value: "monthly" },
            { label: strings.attendance.viewDaily, value: "daily" },
          ]}
        />
      </Group>

      <Paper withBorder p="md">
        <FilterBar
          fromDate={fromDate}
          toDate={toDate}
          branchId={branchId}
          siteId={siteId}
          departmentId={departmentId}
          attendanceStatus={attendanceStatus}
          includeTerminated={includeTerminated}
          branches={branches.data ?? []}
          sites={sites.data ?? []}
          departments={departments.data ?? []}
          onFromDate={(d) => { setFromDate(d); setCursor(null); }}
          onToDate={(d) => { setToDate(d); setCursor(null); }}
          onBranch={(v) => { setBranchId(v); setCursor(null); }}
          onSite={(v) => { setSiteId(v); setCursor(null); }}
          onDepartment={(v) => { setDepartmentId(v); setCursor(null); }}
          onAttendanceStatus={setAttendanceStatus}
          onIncludeTerminated={(v) => { setIncludeTerminated(v); setCursor(null); }}
          onReset={onReset}
        />
        {view === "monthly" && !rangeOk ? (
          <Text c="red" size="sm" mt="sm">
            {strings.attendance.rangeTooWide}
          </Text>
        ) : null}
      </Paper>

      <Paper withBorder p="md">
        {view === "monthly" ? (
          sheet.isLoading ? (
            <Group justify="center" p="xl"><Loader /></Group>
          ) : sheet.isError ? (
            <Text c="red">{strings.attendance.error}</Text>
          ) : (
            <MonthlyGrid
              fromDate={fromStr}
              toDate={toStr}
              employees={sheet.data?.items ?? []}
              attendanceStatus={attendanceStatus}
              onCellClick={openDay}
            />
          )
        ) : daily.isLoading ? (
          <Group justify="center" p="xl"><Loader /></Group>
        ) : daily.isError ? (
          <Text c="red">{strings.attendance.dailyError}</Text>
        ) : (
          <DailyList
            rows={daily.data?.items ?? []}
            onRowClick={(empId, date) => openDay(empId, date)}
          />
        )}
      </Paper>

      <Group justify="space-between" wrap="wrap">
        <Button variant="default" onClick={() => setMarkOpen(true)} disabled={view !== "monthly" || !rangeOk}>
          {strings.attendance.markReviewed.submit}
        </Button>
        <Group gap="sm">
          <Button variant="default" disabled={!cursor} onClick={() => setCursor(null)}>
            {strings.attendance.firstPage}
          </Button>
          <Button
            variant="default"
            disabled={view === "monthly" ? !sheet.data?.next_cursor : !daily.data?.next_cursor}

            {strings.attendance.nextPage}

            <label>
              <Text size="sm" fw={500}>{strings.attendance.dailyDate}</Text>
              <input

                  setCursor(null);

        />

        branchId={branchId}
        departmentId={departmentId}
        includeTerminated={includeTerminated}
        onClose={() => setMarkOpen(false)}
      />
    </Stack>
  );
}

function jr(body: unknown, status = 200): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "Content-Type": "application/json" },
  });
}

function requestUrl(input: RequestInfo | URL): string {
  if (typeof input === "string") return input;
  if (input instanceof URL) return input.toString();
  if (typeof Request !== "undefined" && input instanceof Request) return input.url;
  return String(input);
}

function setupMock(opts: {
  sheetItems?: unknown[];
  dailyItems?: unknown[];
  dayPayload?: unknown;
  markRefused?: unknown[];
}): void {
  global.fetch = vi.fn(async (input: RequestInfo | URL) => {
    const url = requestUrl(input);
    if (url.includes("/api/v1/attendance/sheet")) {
      return jr({ items: opts.sheetItems ?? [], next_cursor: null });
    }
    if (url.includes("/api/v1/attendance/daily")) {
      return jr({ items: opts.dailyItems ?? [], next_cursor: null });
    }
    if (url.includes("/api/v1/attendance/day/")) {

      url.includes("/api/v1/sites") ||
      url.includes("/api/v1/departments")
    ) {
      return jr({ items: [], next_cursor: null });

}

## Important contracts

- `/auth/login` returns `{access_token, refresh_token, token_type, expires_in}` only.
- `/auth/me` returns `{id, username, role, employee_id, permissions[], must_change_password}`.
- Usernames: `normalise_username` strips separators when a digit is present.
- Refresh: CSRF header still missing on client (flagged since F-1a).
- **`new Request(relativeUrl)` throws in jsdom/Node**. All manual requests go through `absoluteUrl()` in `client.ts`.
- Attendance sheet/daily/day/mark_reviewed field shapes assumed and normalised defensively (see `attendance.ts` and `warn` in it). Verified against real backend not yet.
- `GET /attendance/day/{employee_id}/{work_date}` assumed per 2B spec. May differ; if it 404s, F-2b-day-panel-fix replaces `fetchDay` with `fetchPunches` from `/punches`.
- `POST /attendance/mark_reviewed` assumed to return `{reviewed_count, refused_days: [...]}`. May differ; if it 422s or shape doesn't match, one fix bundle adjusts.
