import createClient from "openapi-fetch";
import type { paths } from "./schema";
import { getTenantSlug } from "./tenant";

export type TokenStore = {
  getToken: () => string | null;
  setToken: (token: string | null) => void;
};

export type RefreshFn = () => Promise<string | null>;

export function createWrappedFetch(
  tokenStore: TokenStore,
  refresh: RefreshFn,
  onUnauthorized: () => void,
): (req: Request) => Promise<Response> {
  return async (req) => {
    const original = req.clone();
    const first = new Request(req);
    const slug = getTenantSlug();
    if (slug) first.headers.set("X-Tenant-Slug", slug);
    const token = tokenStore.getToken();
    if (token) first.headers.set("Authorization", `Bearer ${token}`);
    let res = await fetch(first);
    if (res.status !== 401) return res;

    const newToken = await refresh();
    if (!newToken) {
      onUnauthorized();
      return res;
    }
    tokenStore.setToken(newToken);

    const retry = new Request(original);
    if (slug) retry.headers.set("X-Tenant-Slug", slug);
    retry.headers.set("Authorization", `Bearer ${newToken}`);
    res = await fetch(retry);
    if (res.status === 401) onUnauthorized();
    return res;
  };
}

let accessToken: string | null = null;

  return accessToken;
}

export function setAccessToken(token: string | null): void {
  accessToken = token;
}

let refreshPromise: Promise<string | null> | null = null;

export function refreshAccessToken(): Promise<string | null> {
  if (refreshPromise) return refreshPromise;
  refreshPromise = (async () => {
    try {
      const slug = getTenantSlug();
      const res = await fetch("/api/v1/auth/refresh", {
        method: "POST",
        credentials: "include",
        headers: slug ? { "X-Tenant-Slug": slug } : {},
      });
      if (!res.ok) return null;
      const body = (await res.json().catch(() => null)) as
        | { access_token?: unknown }
        | null;
      return typeof body?.access_token === "string" ? body.access_token : null;
    } catch {
      return null;
    }
  })();
  try {
    return await refreshPromise;
  } finally {
    refreshPromise = null;
  }
}

import { parseApiError, type ApiError } from "../api/errors";
import { getTenantSlug } from "../api/tenant";

const BASE = "/api/v1";

function tenantHeaders(): Record<string, string> {
  const slug = getTenantSlug();
  return slug ? { "X-Tenant-Slug": slug } : {};
}

export type LoginResult = {
  accessToken: string;
  mustChangePassword: boolean;
};

export class AuthRequestError extends Error {
  readonly apiError: ApiError;
  constructor(err: ApiError) {
    super(err.message);
    this.apiError = err;
  }
}

export async function loginRequest(
  identifier: string,
  password: string,
): Promise<LoginResult> {
  const res = await fetch(`${BASE}/auth/login`, {
    method: "POST",
    credentials: "include",
    headers: { "Content-Type": "application/json", ...tenantHeaders() },
    body: JSON.stringify({ identifier, password }),
  });
  if (!res.ok) {
    throw new AuthRequestError(await parseApiError(res));
  }
  const body = (await res.json().catch(() => null)) as
    | { access_token?: unknown; must_change_password?: unknown }
    | null;
  const token = body?.access_token;
  if (typeof token !== "string") {
    throw new AuthRequestError({
      code: "bad_response",
      message: "Login response missing access_token",
      status: 200,
    });
  }
  return {
    accessToken: token,
    mustChangePassword: body?.must_change_password === true,
  };
}

export async function logoutRequest(): Promise<void> {
  try {
    await fetch(`${BASE}/auth/logout`, {
      method: "POST",
      credentials: "include",
      headers: tenantHeaders(),
    });
  } catch {
    // network error on logout is not fatal; local state clears regardless
  }
}

export async function changePasswordRequest(params: {
  currentPassword: string;
  newPassword: string;
}): Promise<void> {
  const res = await fetch(`${BASE}/auth/change-password`, {
    method: "POST",
    credentials: "include",
    headers: { "Content-Type": "application/json", ...tenantHeaders() },
    body: JSON.stringify({
      current_password: params.currentPassword,
      new_password: params.newPassword,
    }),
  });
  if (!res.ok) {
    throw new AuthRequestError(await parseApiError(res));
  }
}

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from "react";
import { setAccessToken, setOnUnauthorized } from "../api/client";
import {
  AuthRequestError,
  changePasswordRequest,
  loginRequest,
  logoutRequest,
} from "./api";

export type AuthState = {
  status: "authenticated" | "unauthenticated";
  mustChangePassword: boolean;
  loginError: string | null;
  busy: boolean;
};

export type AuthContextValue = AuthState & {
  login: (identifier: string, password: string) => Promise<boolean>;
  logout: () => Promise<void>;
  changePassword: (current: string, next: string) => Promise<boolean>;
};

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<AuthState>({
    status: "unauthenticated",
    mustChangePassword: false,
    loginError: null,
    busy: false,
  });

  useEffect(() => {
    setOnUnauthorized(() => {
      setAccessToken(null);
      setState({
        status: "unauthenticated",
        mustChangePassword: false,
        loginError: "Session expired. Please sign in again.",
        busy: false,
      });
    });
    return () => setOnUnauthorized(null);
  }, []);

  const login = useCallback(
    async (identifier: string, password: string): Promise<boolean> => {
      setState((s) => ({ ...s, busy: true, loginError: null }));
      try {
        const result = await loginRequest(identifier, password);
        setAccessToken(result.accessToken);
        setState({
          status: "authenticated",
          mustChangePassword: result.mustChangePassword,
          loginError: null,
          busy: false,
        });
        return true;
      } catch (err) {
        const message =
          err instanceof AuthRequestError
            ? err.apiError.message
            : "Sign-in failed";
        setState({
          status: "unauthenticated",
          mustChangePassword: false,
          loginError: message,
          busy: false,
        });
        return false;
      }
    },
    [],
  );

  const logout = useCallback(async (): Promise<void> => {
    await logoutRequest();
    setAccessToken(null);
    setState({
      status: "unauthenticated",
      mustChangePassword: false,
      loginError: null,
      busy: false,
    });
  }, []);

  const changePassword = useCallback(
    async (current: string, next: string): Promise<boolean> => {
      setState((s) => ({ ...s, busy: true, loginError: null }));
      try {
        await changePasswordRequest({ currentPassword: current, newPassword: next });
        setState((s) => ({ ...s, mustChangePassword: false, busy: false }));
        return true;
      } catch (err) {
        const message =
          err instanceof AuthRequestError
            ? err.apiError.message
            : "Change password failed";
        setState((s) => ({ ...s, busy: false, loginError: message }));
        return false;
      }
    },
    [],
  );

  return (
    <AuthContext.Provider value={{ ...state, login, logout, changePassword }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

export function ProtectedRoute({ children }: { children: ReactNode }) {
  const { status, mustChangePassword } = useAuth();
  const location = useLocation();

  if (status === "unauthenticated") {
    return <Navigate to="/login" replace state={{ from: location.pathname }} />;
  }
  if (
    status === "authenticated" &&
    mustChangePassword &&
    location.pathname !== "/change-password"
  ) {
    return <Navigate to="/change-password" replace />;
  }
  return <>{children}</>;
}
