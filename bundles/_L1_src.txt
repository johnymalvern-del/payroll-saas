"""Redaction helper for audit-log values.

Every create, update or deactivate writes an audit_log row with old_value and
new_value. Sensitive fields must never appear there in plaintext. This module
centralises the redaction so every write path uses the same rule.

Sensitive keys are matched case-insensitively. The list is:

    national_id, national_id_encrypted
    bank_account_number, bank_account_number_encrypted
    mfa_secret, mfa_secret_encrypted
    password, password_hash, current_password, new_password
    token, access_token, refresh_token, jti, jti_hash

Nested dicts and lists are walked recursively. Nothing else is redacted:
tax_number, paye_reference, uif_reference, sdl_reference, coida_reference,
email, mobile, first_name, last_name all remain because they are either
business identifiers or already-excluded-not-secret identity fields, and a
post-hoc audit investigation needs to see them.
"""
from __future__ import annotations

from typing import Any


REDACTED = "[redacted]"

_SENSITIVE_KEYS: frozenset[str] = frozenset({
    "national_id",
    "national_id_encrypted",
    "bank_account_number",
    "bank_account_number_encrypted",
    "mfa_secret",
    "mfa_secret_encrypted",
    "password",
    "password_hash",
    "current_password",
    "new_password",
    "token",
    "access_token",
    "refresh_token",
    "jti",
    "jti_hash",
})


def _is_sensitive(key: str) -> bool:
    return key.lower() in _SENSITIVE_KEYS


def redact_for_audit(value: Any) -> Any:
    """Return a redacted copy of value (dict, list, or scalar)."""
    if value is None:
        return None
    if isinstance(value, dict):
        out: dict[Any, Any] = {}
        for k, v in value.items():
            if isinstance(k, str) and _is_sensitive(k):
                out[k] = REDACTED
            else:
                out[k] = redact_for_audit(v)
        return out
    if isinstance(value, (list, tuple)):
        return [redact_for_audit(v) for v in value]
    return value

"""FastAPI dependency helpers.

- `ControlDbDep`   : a control-database session (platform admin, registration)
- `TenantDep`      : a resolved TenantContext (raises if the request has no tenant)
- `TenantDbDep`    : a tenant-database session, scoped to the resolved tenant
- `StorageDep`     : file storage backend (local in dev, S3 in prod)
- `JobRunnerDep`   : job-runner backend (in-process in dev, RQ in prod)
- `ScannerDep`     : virus-scanner backend (no-op in dev, ClamAV in prod)

The tenant context travels as a value (TenantContext), not a ContextVar.
This is required because FastAPI runs sync dependencies and sync endpoints
on different anyio worker threads, and context variables do not cross that
boundary.
"""
from __future__ import annotations

from collections.abc import Iterator
from typing import Annotated

from fastapi import Depends, HTTPException, Request, status
from sqlalchemy.orm import Session

from app.core.errors import AppError
from app.core.logging import get_logger
from app.core.tenant_routing import (
    TenantContext,
    TenantError,
    TenantInactive,
    TenantNotProvisioned,
    TenantNotFound,
    TenantNotSpecified,
    TenantSchemaBehind,
    resolve_slug,
    tenant_context,
)
from app.db.control.session import control_session
from app.db.tenant.session import tenant_session

log = get_logger("app.api.deps")


def _control_db() -> Iterator[Session]:
    with control_session() as session:
        yield session


ControlDbDep = Annotated[Session, Depends(_control_db)]


def _translate_tenant_error(exc: TenantError) -> AppError:
    if isinstance(exc, TenantNotSpecified):
        return AppError(
            "tenant_missing",
            "Tenant not specified (provide subdomain or X-Tenant-Slug)",
            status.HTTP_400_BAD_REQUEST,
        )
    if isinstance(exc, TenantNotFound):
        return AppError(
            "tenant_not_found", f"Tenant '{exc.slug}' not found", status.HTTP_404_NOT_FOUND
        )
    if isinstance(exc, TenantInactive):
        return AppError(
            "tenant_inactive",
            f"Tenant '{exc.slug}' is {exc.status}",
            status.HTTP_403_FORBIDDEN,
        )
    if isinstance(exc, TenantNotProvisioned):
        return AppError(
            "tenant_not_provisioned",
            f"Tenant '{exc.slug}' is not yet provisioned",
            status.HTTP_503_SERVICE_UNAVAILABLE,
        )
    if isinstance(exc, TenantSchemaBehind):
        return AppError(
            "tenant_schema_behind",
            f"Tenant '{exc.slug}' requires a migration ({exc.current!r} -> {exc.expected!r})",
            status.HTTP_503_SERVICE_UNAVAILABLE,
        )
    return AppError(
        "tenant_error", str(exc), status.HTTP_500_INTERNAL_SERVER_ERROR
    )


def _tenant(request: Request) -> Iterator[TenantContext]:
    slug = resolve_slug(request)
    if slug is None:
        raise _translate_tenant_error(TenantNotSpecified())
    try:
        with tenant_context(slug) as ctx:
            yield ctx
    except TenantError as exc:
        raise _translate_tenant_error(exc) from exc


TenantDep = Annotated[TenantContext, Depends(_tenant)]


def _tenant_db(ctx: TenantDep) -> Iterator[Session]:
    with tenant_session(ctx.tenant_db_name) as session:
        yield session


TenantDbDep = Annotated[Session, Depends(_tenant_db)]


def ensure_no_tenant(request: Request) -> None:
    """Raise if the request is carrying a tenant. Used by platform endpoints."""
    if resolve_slug(request) is not None:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="This endpoint must not be called for a specific tenant",
        )
