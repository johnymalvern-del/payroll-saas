"""Minimal PDF renderer for payslips (Chapter 4, Bundle 004b-b).

No external dependency (no reportlab). Renders a single-page text PDF
containing the standard payslip fields, which is sufficient for a proof-of-
concept and a mechanical correctness check. Upgrading to a nicer layout is
a follow-up that swaps this module only.

Storage via StorageDep is deferred (documented in STATE.md at 016c). The
endpoint returns bytes on demand.
"""
from __future__ import annotations

from decimal import Decimal
from typing import Any


def _esc(s: str) -> str:
    return s.replace("\\", "\\\\").replace("(", "\\(").replace(")", "\\)")


def _fmt(v: Any) -> str:
    if v is None:
        return ""
    if isinstance(v, Decimal):
        return f"{v:.2f}"
    return str(v)


def _payslip_text_lines(payslip: Any, employee: Any, pay_period: Any) -> list[str]:
    code = getattr(employee, "employee_code", "") or ""
    name = " ".join(
        x for x in (getattr(employee, "first_name", "") or "", getattr(employee, "last_name", "") or "")
        if x
    ).strip()
    period = getattr(pay_period, "period_key", "") or ""
    pay_date = getattr(pay_period, "pay_date", "")
    revision = getattr(payslip, "revision", 1)
    status = getattr(payslip, "status", "")

    lines = [
        f"PAYSLIP  {period}  ({pay_date})",
        f"Employee: {code}  {name}".rstrip(),
        f"Revision: {revision}  Status: {status}",
        "",
        f"Gross                 {_fmt(getattr(payslip, 'gross', None))}",
        f"Pre-tax deductions    {_fmt(getattr(payslip, 'pre_tax_deductions', None))}",
        f"Taxable income        {_fmt(getattr(payslip, 'taxable_income', None))}",
        f"PAYE                  {_fmt(getattr(payslip, 'paye', None))}",
        f"UIF (employee)        {_fmt(getattr(payslip, 'uif_employee', None))}",
        f"Other deductions      {_fmt(getattr(payslip, 'other_deductions', None))}",
        f"Reimbursements        {_fmt(getattr(payslip, 'reimbursements', None))}",
        f"NET PAY               {_fmt(getattr(payslip, 'net', None))}",
        "",
        "Employer costs",
        f"UIF (employer)        {_fmt(getattr(payslip, 'uif_employer', None))}",
        f"SDL                   {_fmt(getattr(payslip, 'sdl', None))}",
    ]
    return lines


def _render_minimal_pdf(lines: list[str]) -> bytes:
    """Construct a valid single-page PDF (Courier, 10pt) from text lines."""
    content = ["BT", "/F1 10 Tf", "14 TL", "50 780 Td"]
    for i, line in enumerate(lines):
        if i > 0:
            content.append("T*")
        content.append(f"({_esc(line)}) Tj")
    content.append("ET")
    stream = "\n".join(content).encode("latin-1", errors="replace")

    objs: list[bytes] = []
    objs.append(b"<< /Type /Catalog /Pages 2 0 R >>")
    objs.append(b"<< /Type /Pages /Kids [3 0 R] /Count 1 >>")
    objs.append(
        b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595 842] "
        b"/Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>"
    )
    objs.append(
        b"<< /Length " + str(len(stream)).encode() + b" >>\nstream\n" + stream + b"\nendstream"
    )
    objs.append(b"<< /Type /Font /Subtype /Type1 /BaseFont /Courier >>")

    out = bytearray()
    out += b"%PDF-1.4\n"
    offsets = [0]
    for i, o in enumerate(objs, start=1):
        offsets.append(len(out))
        out += f"{i} 0 obj\n".encode()
        out += o
        out += b"\nendobj\n"
    xref_offset = len(out)
    out += f"xref\n0 {len(objs) + 1}\n".encode()
    out += b"0000000000 65535 f \n"
    for off in offsets[1:]:
        out += f"{off:010d} 00000 n \n".encode()
    out += b"trailer\n"
    out += f"<< /Size {len(objs) + 1} /Root 1 0 R >>\n".encode()
    out += b"startxref\n"
    out += f"{xref_offset}\n".encode()
    out += b"%%EOF\n"
    return bytes(out)


def render_payslip_pdf(payslip: Any, employee: Any, pay_period: Any) -> bytes:
    return _render_minimal_pdf(_payslip_text_lines(payslip, employee, pay_period))

"""Payslip read endpoints (Chapter 4, Bundle 004b-b).

Four endpoints:
  GET /payroll/payslips                      list (excludes preview rows)
  GET /payroll/payslips/{id}                 get one (scope applied)
  GET /payroll/payslips/{id}/explain         header + lines + snapshot
  GET /payroll/payslips/{id}/pdf             bytes on demand

Permission gates on PAYSLIP_VIEW (granted to all five roles per the current
matrix). Scope enforcement via apply_scope / assert_in_scope decides which
rows a manager or auditor may see; per-F14 role refinements are a follow-up.

Storage is not used: PDF is rendered on demand and returned as bytes.
"""

router = APIRouter(prefix="/payroll/payslips", tags=["payroll-payslips"])

_DEFAULT_LIMIT = 100
_MAX_LIMIT = 500


def _encode_cursor(uid: uuid.UUID) -> str:
    raw = json.dumps({"i": str(uid)}, separators=(",", ":")).encode("utf-8")
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")


def _decode_cursor(cursor: str) -> uuid.UUID:
    padded = cursor + "=" * (-len(cursor) % 4)
    try:
        raw = base64.urlsafe_b64decode(padded.encode("ascii")).decode("utf-8")
        return uuid.UUID(json.loads(raw)["i"])
    except Exception:
        raise AppError("invalid_cursor", "The pagination cursor is malformed", 400) from None


def _load_payslip_scoped(db, user, payslip_id: uuid.UUID) -> Payslip:
    row = db.scalar(select(Payslip).where(Payslip.id == payslip_id))
    if row is None or row.status == "preview":
        raise AppError("not_found", "Payslip not found", 404)
    try:
        assert_in_scope(user, row.employee_id, db)
    except ScopeViolation:
        raise AppError("not_found", "Payslip not found", 404)
    return row


@router.get("", response_model=PayslipListOut)
def list_payslips(
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYSLIP_VIEW))],
    employee_id: uuid.UUID | None = Query(None),
    pay_period_id: uuid.UUID | None = Query(None),
    status: str | None = Query("current"),
    limit: int = Query(_DEFAULT_LIMIT, ge=1, le=_MAX_LIMIT),
    cursor: str | None = Query(None),
) -> PayslipListOut:
    q = select(Payslip)
    if status is not None:
        q = q.where(Payslip.status == status)
    q = q.where(Payslip.status != "preview")  # previews never surfaced here
    if employee_id is not None:
        q = q.where(Payslip.employee_id == employee_id)
    if pay_period_id is not None:
        q = q.where(Payslip.pay_period_id == pay_period_id)
    q = apply_scope(q, user, db, model=Payslip)
    if cursor is not None:
        q = q.where(Payslip.id > _decode_cursor(cursor))
    q = q.order_by(Payslip.id.asc()).limit(limit + 1)
    rows = list(db.scalars(q).all())
    next_cursor = None
    if len(rows) > limit:
        rows = rows[:limit]
        next_cursor = _encode_cursor(rows[-1].id)
    return PayslipListOut(
        items=[PayslipOut.model_validate(r) for r in rows],
        next_cursor=next_cursor,
    )


@router.get("/{payslip_id}", response_model=PayslipOut)
def get_payslip(
    payslip_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYSLIP_VIEW))],
) -> PayslipOut:
    row = _load_payslip_scoped(db, user, payslip_id)
    return PayslipOut.model_validate(row)


@router.get("/{payslip_id}/explain", response_model=PayslipExplainOut)
def explain_payslip(
    payslip_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYSLIP_VIEW))],
) -> PayslipExplainOut:
    row = _load_payslip_scoped(db, user, payslip_id)
    lines = list(db.scalars(
        select(PayslipLine).where(PayslipLine.payslip_id == row.id)
        .order_by(PayslipLine.sort_order.asc())
    ).all())
    return PayslipExplainOut(
        payslip=PayslipOut.model_validate(row),
        lines=[PayslipLineOut.model_validate(l) for l in lines],
        snapshot=row.snapshot or {},
        calculation_context=row.calculation_context or {},
    )


@router.get("/{payslip_id}/pdf")
def get_payslip_pdf(
    payslip_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYSLIP_VIEW))],
) -> Response:
    row = _load_payslip_scoped(db, user, payslip_id)
    employee = db.scalar(select(Employee).where(Employee.id == row.employee_id))
    pay_period = db.scalar(select(PayPeriod).where(PayPeriod.id == row.pay_period_id))
    if employee is None or pay_period is None:
        raise AppError("not_found", "Payslip references missing employee or period", 500)
    pdf_bytes = render_payslip_pdf(row, employee, pay_period)
    filename = f"payslip-{employee.employee_code}-{pay_period.period_key}.pdf"
    return Response(
        content=pdf_bytes,
        media_type="application/pdf",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )

"""Minimal PDF renderer (Chapter 4, Bundle 004b-b).

Pure function test - no DB.

from __future__ import annotations

import uuid
from datetime import date, datetime, timezone
from decimal import Decimal

from app.services.payroll_pdf import render_payslip_pdf


def _fake_payslip(**_overrides):
    base = dict(
        id=uuid.uuid4(),
        revision=1,
        status="current",
        gross=Decimal("1000.00"),
        pre_tax_deductions=Decimal("0.00"),
        taxable_income=Decimal("1000.00"),
        paye=Decimal("179.17"),
        uif_employee=Decimal("5.00"),
        uif_employer=Decimal("5.00"),
        sdl=Decimal("10.00"),
        other_deductions=Decimal("0.00"),
        reimbursements=Decimal("0.00"),
        net=Decimal("815.83"),
    )
    base.update(_overrides)
    return type("P", (), base)


def _fake_employee():
    return type("E", (), {
        "employee_code": "EMP-001",
        "first_name": "Ada",
        "last_name": "Lovelace",
    })


def _fake_period():
    return type("PP", (), {
        "period_key": "2025-06-M",
        "pay_date": date(2025, 6, 30),
    })


def test_pdf_bytes_start_with_header():
    pdf = render_payslip_pdf(_fake_payslip(), _fake_employee(), _fake_period())
    assert pdf[:5] == b"%PDF-"
    assert pdf.rstrip().endswith(b"%%EOF")


def test_pdf_contains_key_fields():
    pdf = render_payslip_pdf(_fake_payslip(), _fake_employee(), _fake_period())
    text = pdf.decode("latin-1")
    assert "EMP-001" in text
    assert "Ada Lovelace" in text
    assert "2025-06-M" in text
    assert "1000.00" in text
    assert "179.17" in text
    assert "815.83" in text
    assert "NET PAY" in text


def test_pdf_escapes_parentheses():
    # A gross of 1000.00 renders as "1000.00" - parentheses only in labels
    pdf = render_payslip_pdf(_fake_payslip(), _fake_employee(), _fake_period())
    # Balanced parentheses in the raw file
    text = pdf.decode("latin-1")
    assert text.count("(") == text.count(")")

def _route_map():
    from fastapi.routing import APIRoute
    from app.main import app
    out: dict[tuple[str, str], APIRoute] = {}
    for r in app.routes:
        if not isinstance(r, APIRoute):
            continue
        for m in r.methods:
            if m in ("HEAD", "OPTIONS"):
                continue
            out[(m, r.path)] = r
    return out


def _extract_permissions(route) -> set[Permission]:
    found: set[Permission] = set()
    stack = [route.dependant]
    while stack:
        d = stack.pop()
        fn = getattr(d, "call", None)
        if fn is not None and getattr(fn, "__closure__", None):
            for cell in fn.__closure__:
                try:
                    v = cell.cell_contents
                except ValueError:
                    continue
                if isinstance(v, Permission):
                    found.add(v)
        stack.extend(getattr(d, "dependencies", []) or [])
    return found


_IDS = [f"{m}-{p}-{perm.value}" for m, p, perm in EXPECTED_ROUTES]


@pytest.mark.parametrize("method,path,expected_perm", EXPECTED_ROUTES, ids=_IDS)
def test_route_registered(method: str, path: str, expected_perm: Permission) -> None:
    routes = _route_map()
    key = (method, path)
    assert key in routes, (
        f"Route {method} {path} not registered. "
        f"Available payroll routes: {sorted(k for k in routes if 'payroll' in k[1])}"
    )


@pytest.mark.parametrize("method,path,expected_perm", EXPECTED_ROUTES, ids=_IDS)
def test_route_has_expected_permission(method: str, path: str, expected_perm: Permission) -> None:
    routes = _route_map()
    route = routes.get((method, path))
    assert route is not None, f"Route {method} {path} not found"
    perms = _extract_permissions(route)
    assert expected_perm in perms, (
        f"{method} {path}: expected permission {expected_perm.value!r}, "
        f"found {sorted(p.value for p in perms)}"
    )


def test_payroll_routes_all_have_a_permission() -> None:
    routes = _route_map()
    offenders = []
    for (method, path), route in routes.items():
        if not path.startswith("/api/v1/payroll"):
            continue
        if not _extract_permissions(route):
            offenders.append(f"{method} {path}")
    assert not offenders, f"payroll routes missing require_permission: {offenders}"
