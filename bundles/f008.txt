export async function fetchExceptions(params: {
  status?: string;
  type?: string;
  fromDate?: string;
  toDate?: string;
  limit?: number;
  cursor?: string | null;
}): Promise<ExceptionList> {
  const qs = new URLSearchParams();
  if (params.status) qs.set("status", params.status);
  if (params.type) qs.set("type", params.type);
  if (params.fromDate) qs.set("from_date", params.fromDate);
  if (params.toDate) qs.set("to_date", params.toDate);
  qs.set("limit", String(params.limit ?? 100));
  if (params.cursor) qs.set("cursor", params.cursor);
  const res = await wrappedFetch(
    new Request(absoluteUrl(`/api/v1/attendance/exceptions?${qs.toString()}`), {
      credentials: "include",
    }),
  );
  if (!res.ok) throw new Error(`http_${res.status}`);
  const data = (await res.json()) as Record<string, unknown>;
  return {
    items: Array.isArray(data.items) ? (data.items as ExceptionRow[]) : [],
    next_cursor:
      typeof data.next_cursor === "string" ? data.next_cursor : null,
    total: typeof data.total === "number" ? data.total : null,
  };
}

export async function resolveException(

  body: ResolveRequest,
): Promise<ResolveResponse> {

        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      },
    ),
  );
  if (!res.ok) {
    const detail = (await res.json().catch(() => ({}))) as {
      error?: { code?: string };
    };
    throw new Error(detail.error?.code ?? `http_${res.status}`);
  }
  return (await res.json()) as ResolveResponse;
}

  if (!value) return "";
  const d = dayjs(value);
  return d.isValid() ? d.format("HH:mm") : "";
}

export function AdjustmentForm({ workDate, initial, onSubmit, submitLabel, busy }: Props) {
  const [outcome, setOutcome] = useState<AdjustmentOutcome>(
    initial?.outcome ?? "worked",
  );
  const [timeIn, setTimeIn] = useState<string>(toTimeInput(initial?.time_in));
  const [timeOut, setTimeOut] = useState<string>(toTimeInput(initial?.time_out));
  const [reasonCode, setReasonCode] = useState<string>(initial?.reason_code ?? "");
  const [note, setNote] = useState<string>(initial?.note ?? "");
  const [error, setError] = useState<string | null>(null);

  const requiresTimes = outcome === "worked";
  const reasonIsOther = reasonCode.trim().toLowerCase() === "other";

  const submit = () => {
    setError(null);
    if (!reasonCode.trim()) {
      setError(strings.adjustments.reasonRequired);
      return;
    }
    if (requiresTimes && (!timeIn || !timeOut)) {
      setError(strings.adjustments.timesRequired);
      return;
    }
    if (requiresTimes && timeIn && timeOut) {
      const tin = dayjs(`${workDate}T${timeIn}:00`);
      const tout = dayjs(`${workDate}T${timeOut}:00`);
      if (!tin.isValid() || !tout.isValid() || !tout.isAfter(tin)) {
        setError(strings.adjustments.timeOrder);
        return;
      }
    }
    if (reasonIsOther && note.trim().length < 10) {
      setError(strings.adjustments.noteRequiredForOther);
      return;
    }
    onSubmit({
      outcome,
      time_in: requiresTimes && timeIn
        ? dayjs(`${workDate}T${timeIn}:00`).toISOString()
        : null,
      time_out: requiresTimes && timeOut
        ? dayjs(`${workDate}T${timeOut}:00`).toISOString()
        : null,
      reason_code: reasonCode.trim(),
      note: note.trim() ? note.trim() : null,
    });
  };

  return (
    <Stack gap="sm">
      {error ? <Alert color="red">{error}</Alert> : null}
      <Select
        label={strings.adjustments.outcome}

export function ExceptionsPage() {

    queryKey: ["refs", "employees"],
    queryFn: () => fetchEmployees(),
    staleTime: 5 * 60 * 1000,
  });

  const exceptions = useQuery({
    queryKey: ["exceptions", statusFilter, typeFilter.join(",")],
    queryFn: () =>
      fetchExceptions({
        status: statusFilter,

      }),

    mutationFn: (vars: { id: string; body: ResolveRequest }) =>

      setServerError(mapErrorCode(code));

            data={STATUS_OPTIONS}

            label={strings.exceptions.typeFilter}
            data={TYPE_OPTIONS}
            value={typeFilter}
            onChange={setTypeFilter}
            placeholder={strings.exceptions.allTypes}
