"""Fix medical_scheme_members NOT NULL mismatch.

Model declares nullable=True but migration 0017 made it NOT NULL DEFAULT 0.
Change to nullable=False with default=0 so ORM inserts 0 instead of NULL.
Byte-safe write. Idempotent via _4A_BIS_012 marker.
"""
import ast
import pathlib
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
BACKEND = ROOT / "backend"


def main() -> int:
    p = BACKEND / "app" / "db" / "tenant" / "models" / "people.py"
    text = p.read_bytes().decode("utf-8").replace("\r\n", "\n").replace("\r", "\n")
    if "_4A_BIS_012" in text:
        print(f"SKIP: {p.name} already patched")
        return 0
    lines = text.split("\n")
    found = False
    for i, line in enumerate(lines):
        if "medical_scheme_members" in line and "mapped_column" in line:
            stripped = line.lstrip()
            indent = line[: len(line) - len(stripped)]
            int_ref = "sa.Integer" if "sa.Integer" in line else "Integer"
            lines[i] = (
                f"{indent}medical_scheme_members: Mapped[int] = mapped_column("
                f"{int_ref}, nullable=False, default=0)"
            )
            found = True
            break
    if not found:
        print(f"STOP: {p.name}: medical_scheme_members line not found")
        for line in text.split("\n"):
            if "medical_scheme_members" in line:
                print(f"  current: {line!r}")
        return 1
    new_text = "\n".join(lines) + "\n# _4A_BIS_012 people\n"
    ast.parse(new_text)
    p.write_bytes(new_text.encode("utf-8"))
    print(f"PATCHED: {p.name}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

# END OF FILE
