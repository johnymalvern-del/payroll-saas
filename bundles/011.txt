"""Byte-safe patch: add 2 columns to people.py, add swap lines to conftest fixture.

Replaces the broken patch_010.py. Uses read_bytes/write_bytes so Windows
never applies newline translation. Normalizes all endings to \\n first.
Idempotent via _4A_BIS_010B markers.
"""
import ast
import pathlib
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
BACKEND = ROOT / "backend"


def _read_norm(path):
    raw = path.read_bytes()
    text = raw.decode("utf-8")
    return text.replace("\r\n", "\n").replace("\r", "\n")


def _write_norm(path, text):
    ast.parse(text)  # raises before writing if broken
    path.write_bytes(text.encode("utf-8"))


def patch_people():
    p = BACKEND / "app" / "db" / "tenant" / "models" / "people.py"
    text = _read_norm(p)
    if "_4A_BIS_010B" in text:
        print(f"SKIP: {p.name} already patched")
        return 0
    marker = "    uif_exempt:"
    idx = text.find(marker)
    if idx == -1:
        print(f"STOP: {p.name}: 'uif_exempt:' line not found")
        return 1
    eol = text.find("\n", idx)
    if eol == -1:
        eol = len(text) - 1
    insert_at = eol + 1
    new_lines = (
        "    contracted_weekly_hours: Mapped[Decimal | None] = mapped_column("
        "Numeric(5, 2), nullable=True)\n"
        "    medical_scheme_members: Mapped[int | None] = mapped_column("
        "Integer, nullable=True)\n"
    )
    new_text = (
        text[:insert_at] + new_lines + text[insert_at:]
        + "\n# _4A_BIS_010B people\n"
    )
    _write_norm(p, new_text)
    print(f"PATCHED: {p.name}")
    return 0


def patch_conftest():
    p = BACKEND / "tests" / "conftest.py"
    text = _read_norm(p)
    if "_4A_BIS_010B" in text:
        print(f"SKIP: {p.name} already patched")
        return 0
    marker = "        _4a_p._4a_bootstrap_template_db()"
    idx = text.find(marker)
    if idx == -1:
        print(f"STOP: {p.name}: fixture body pattern not found")
        return 1
    new_lines = (
        "        _4a_p._TEST_TEMPLATE_ENABLED = True\n"
        "        _4a_p._run_tenant_migrations = _4a_p._4a_run_migrations_from_template\n"
    )
    new_text = (
        text[:idx] + new_lines + text[idx:]
        + "\n# _4A_BIS_010B conftest\n"
    )
    _write_norm(p, new_text)
    print(f"PATCHED: {p.name}")
    return 0


def main():
    rc = patch_people()
    rc |= patch_conftest()
    return rc


if __name__ == "__main__":
    sys.exit(main())
