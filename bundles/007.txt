=== FILE: scripts/4a_bis_patch_v3.py ===
"""4A-bis v3: fix v2's verify_password arg-order bug.

v2 overrode verify_password with signature (password_hash, password) while
the original is (password, stored_hash). That inverted the args into argon2.

v3 does not override hash_password or verify_password at all — their existing
bodies already call _hasher.verify(hash, password) with the right order. v3
just (a) reassigns module-level PasswordHasher instances to a weak one when
PAYROLL_TEST_FAST_HASH=1 at import time, (b) provides a correct
_4a_current_hasher(), (c) leaves the originals intact otherwise.

Idempotent via _4A_BIS_SECURITY_V3 marker.
"""
import pathlib

ROOT = pathlib.Path(__file__).resolve().parent.parent
BACKEND = ROOT / "backend"

PATCH = '''

# --- 4A-bis v3: _4A_BIS_SECURITY_V3: fix v2 arg-order bug ---
import os as _os_v3
from argon2 import PasswordHasher as _PH_v3

_4a_fast_v3 = _PH_v3(time_cost=1, memory_cost=8192, parallelism=1, hash_len=16, salt_len=8)

_v3_prev_hash = globals().get("hash_password")
_v3_prev_verify = globals().get("verify_password")

if _os_v3.environ.get("PAYROLL_TEST_FAST_HASH") == "1":
    for _n_v3 in list(globals().keys()):
        _v_v3 = globals().get(_n_v3)
        if isinstance(_v_v3, _PH_v3) and _v_v3 is not _4a_fast_v3:
            globals()[_n_v3] = _4a_fast_v3

    def hash_password(password):  # type: ignore[no-redef]
        return _4a_fast_v3.hash(password)

    def verify_password(password, stored_hash):  # type: ignore[no-redef]
        try:
            return _4a_fast_v3.verify(stored_hash, password)
        except Exception:
            return False

def _4a_current_hasher():
    if _os_v3.environ.get("PAYROLL_TEST_FAST_HASH") == "1":
        return _4a_fast_v3
    for _n_v3 in ("_hasher", "_PH", "_password_hasher", "password_hasher", "_HASHER"):
        _v_v3 = globals().get(_n_v3)
        if isinstance(_v_v3, _PH_v3):
            return _v_v3
    return _PH_v3()
# --- end 4A-bis v3 ---

# END OF FILE
'''


def main() -> int:
    path = BACKEND / "app" / "core" / "security.py"
    raw = path.read_bytes()
    if b"_4A_BIS_SECURITY_V3" in raw:
        print(f"SKIP: {path.name} (v3 marker present)")
        return 0
    text = raw.decode("utf-8")
    if "# END OF FILE" in text:
        text = text.rsplit("# END OF FILE", 1)[0].rstrip() + "\r\n"
    path.write_bytes((text + PATCH).encode("utf-8"))
    print(f"PATCHED v3: {path}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

# END OF FILE

=== FILE: backend/tests/test_security_argon2_guard.py ===
"""4A-bis guard tests: fast paths only in test env; production unchanged.

Weak-hash tests skip when v3 marker is absent (i.e. patch not yet applied).
v2's broken overrides are shadowed by v3's correct ones at import time.
"""
from __future__ import annotations

import os
import pathlib
import subprocess
import sys

import pytest


_SECURITY_PATH = pathlib.Path(__file__).resolve().parent.parent / "app" / "core" / "security.py"


def _v3_applied() -> bool:
    try:
        return "_4A_BIS_SECURITY_V3" in _SECURITY_PATH.read_text(encoding="utf-8")
    except OSError:
        return False


def _isolated(code, drop_env=()):
    env = {k: v for k, v in os.environ.items() if k not in drop_env}
    return subprocess.run(
        [sys.executable, "-c", code],
        capture_output=True, text=True, env=env, timeout=30,
    )


def test_argon2_default_strong_without_env():
    """Production defaults unchanged: >= 32768 memory_cost, >= 2 time_cost."""
    r = _isolated(
        "from app.core import security; "
        "h = next((v for k, v in vars(security).items() if isinstance(v, __import__('argon2').PasswordHasher)), None); "
        "import argon2; "
        "h = h or argon2.PasswordHasher(); "
        "print(h.memory_cost, h.time_cost)",
        drop_env={"PAYROLL_TEST_FAST_HASH"},
    )
    assert r.returncode == 0, r.stderr
    mc, tc = r.stdout.strip().split()
    assert int(mc) >= 32768, f"memory_cost too low: {mc}"
    assert int(tc) >= 2, f"time_cost too low: {tc}"


@pytest.mark.skipif(not _v3_applied(), reason="v3 patch not applied yet")
def test_argon2_weak_with_env_isolated():
    env = dict(os.environ)
    env["PAYROLL_TEST_FAST_HASH"] = "1"
    r = subprocess.run(
        [sys.executable, "-c", "from app.core.security import _4a_current_hasher; h=_4a_current_hasher(); print(h.memory_cost, h.time_cost)"],
        capture_output=True, text=True, env=env, timeout=30,
    )
    assert r.returncode == 0, r.stderr
    mc, tc = r.stdout.strip().split()
    assert int(mc) < 32768, f"expected weak, got {mc}"
    assert int(tc) < 3, f"expected weak, got {tc}"


@pytest.mark.skipif(not _v3_applied(), reason="v3 patch not applied yet")
def test_hash_password_weak_when_env_set():
    """End-to-end: hash_password uses weak params when env is set."""
    env = dict(os.environ)
    env["PAYROLL_TEST_FAST_HASH"] = "1"
    code = (
        "from app.core.security import hash_password; "
        "h = hash_password('test'); "
        "print(h[:60])"
    )
    r = subprocess.run(
        [sys.executable, "-c", code],
        capture_output=True, text=True, env=env, timeout=30,
    )
    assert r.returncode == 0, r.stderr
    prefix = r.stdout.strip()
    assert "m=8192" in prefix, f"hash_password used strong params: {prefix!r}"


def test_hash_password_strong_without_env():
    """Production hash_password uses strong params (m=65536)."""
    r = _isolated(
        "from app.core.security import hash_password; h = hash_password('test'); print(h[:60])",
        drop_env={"PAYROLL_TEST_FAST_HASH"},
    )
    assert r.returncode == 0, r.stderr
    prefix = r.stdout.strip()
    assert "m=65536" in prefix, f"hash_password not strong: {prefix!r}"


def test_template_flag_off_by_default():
    r = _isolated(
        "from app.services.provisioning import _TEST_TEMPLATE_ENABLED; assert not _TEST_TEMPLATE_ENABLED; print('ok')",
        drop_env={"PAYROLL_TEST_TEMPLATE_DB"},
    )
    assert r.returncode == 0, r.stderr


def test_template_flag_off_with_env_but_no_conftest():
    env = dict(os.environ)
    env["PAYROLL_TEST_TEMPLATE_DB"] = "1"
    r = subprocess.run(
        [sys.executable, "-c", "from app.services.provisioning import _TEST_TEMPLATE_ENABLED; assert not _TEST_TEMPLATE_ENABLED; print('ok')"],
        capture_output=True, text=True, env=env, timeout=30,
    )
    assert r.returncode == 0, r.stderr

$ErrorActionPreference = 'Stop'
$root = 'C:\xampp\htdocs\Payroll'
$backend = Join-Path $root 'backend'

Push-Location $root
python scripts\4a_bis_patch.py
$rc = $LASTEXITCODE
Pop-Location
if ($rc -ne 0) { throw "STOP: v1 patch failed" }

Write-Host ""
Write-Host "=== step 1b: apply v3 (fix v2 arg order) ==="
Push-Location $root
python scripts\4a_bis_patch_v3.py

Pop-Location
if ($rc -ne 0) { throw "STOP: v3 patch failed" }

Write-Host ""
Write-Host "=== step 2: BEFORE (env unset) ==="
Push-Location $backend
Remove-Item Env:PAYROLL_TEST_TEMPLATE_DB -ErrorAction SilentlyContinue

$swb = [System.Diagnostics.Stopwatch]::StartNew()

$beforeExit = $LASTEXITCODE
$swb.Stop()
Pop-Location
$beforeSec = [int]$swb.Elapsed.TotalSeconds
$beforeCount = if ($beforeOut -match '(\d+) passed') { $matches[1] } else { "0" }
Write-Host "BEFORE: ${beforeSec}s; passed: $beforeCount; exit: $beforeExit"

Write-Host ""
Write-Host "=== step 3: AFTER #1 (bootstrap) ==="
Push-Location $backend
$env:PAYROLL_TEST_TEMPLATE_DB = '1'
$env:PAYROLL_TEST_FAST_HASH = '1'
$swa1 = [System.Diagnostics.Stopwatch]::StartNew()
$after1Out = python -m pytest tests/test_adjustments.py tests/test_sheet.py -q --no-header 2>&1
$after1Exit = $LASTEXITCODE
$swa1.Stop()
Pop-Location
$after1Sec = [int]$swa1.Elapsed.TotalSeconds
$after1Count = if ($after1Out -match '(\d+) passed') { $matches[1] } else { "0" }
Write-Host "AFTER #1: ${after1Sec}s; passed: $after1Count; exit: $after1Exit"

Write-Host ""
Write-Host "=== step 4: AFTER #2 (steady) ==="
Push-Location $backend
$swa2 = [System.Diagnostics.Stopwatch]::StartNew()
$after2Out = python -m pytest tests/test_adjustments.py tests/test_sheet.py -q --no-header 2>&1
$after2Exit = $LASTEXITCODE
$swa2.Stop()
Pop-Location
$after2Sec = [int]$swa2.Elapsed.TotalSeconds
$after2Count = if ($after2Out -match '(\d+) passed') { $matches[1] } else { "0" }
Write-Host "AFTER #2: ${after2Sec}s; passed: $after2Count; exit: $after2Exit"

Write-Host ""
Write-Host "=== step 5: guard tests (env cleared) ==="
Push-Location $backend
Remove-Item Env:PAYROLL_TEST_TEMPLATE_DB -ErrorAction SilentlyContinue
Remove-Item Env:PAYROLL_TEST_FAST_HASH -ErrorAction SilentlyContinue
$guardOut = python -m pytest tests/test_security_argon2_guard.py -q --no-header 2>&1
$guardExit = $LASTEXITCODE
Pop-Location
Write-Host "guard exit: $guardExit"
$guardOut | Select-Object -Last 8 | ForEach-Object { Write-Host $_ }

Write-Host ""
Write-Host "=== step 6: SUMMARY ==="
if ([int]$beforeCount -gt 0 -and [int]$after2Count -gt 0) {
    $perTestBefore = [math]::Round([double]$beforeSec / [double]$beforeCount, 2)
    $perTestAfter = [math]::Round([double]$after2Sec / [double]$after2Count, 2)
    $perTestSaved = [math]::Round($perTestBefore - $perTestAfter, 2)
    $estFull = [int](2435 - ($perTestSaved * 701))
    if ($estFull -lt 300) { $estFull = 300 }
    Write-Host ("BEFORE: ${beforeSec}s ($beforeCount tests); per-test ${perTestBefore}s")
    Write-Host ("AFTER #1: ${after1Sec}s")
    Write-Host ("AFTER #2: ${after2Sec}s ($after2Count tests); per-test ${perTestAfter}s")
    Write-Host ("Per-test savings: ${perTestSaved}s")
    Write-Host ("Estimated full-suite: ~${estFull}s (~$([int]($estFull/60)) min)")
} else {
    Write-Host "Cannot compute summary (some runs failed to parse pass count)."
    Write-Host "BEFORE: ${beforeSec}s exit=$beforeExit"
    Write-Host "AFTER #1: ${after1Sec}s exit=$after1Exit"
    Write-Host "AFTER #2: ${after2Sec}s exit=$after2Exit"
}

Write-Host ""
Write-Host "=== step 7: drift diagnostic ==="
Push-Location $backend
$driftOut = python -m pytest tests/test_schema_drift.py -x -vv --tb=long --no-header 2>&1

Pop-Location
$driftOut | Out-File -FilePath (Join-Path $root 'drift_diag.out') -Encoding utf8
Write-Host "drift exit: $driftExit; saved drift_diag.out"

$driftTestPath = Join-Path $backend 'tests\test_schema_drift.py'
if (Test-Path $driftTestPath) {
    Get-Content $driftTestPath -Raw | Out-File -FilePath (Join-Path $root 'drift_source.out') -Encoding utf8
    Write-Host "saved drift_source.out ($((Get-Item (Join-Path $root 'drift_source.out')).Length) bytes)"
}

Write-Host ""
Write-Host "=== step 8: git commit patched files ==="
Push-Location $root
git add backend/app/core/security.py backend/tests/test_security_argon2_guard.py

Write-Host ""
Write-Host "============= PASTE THIS BLOCK BACK ============="
