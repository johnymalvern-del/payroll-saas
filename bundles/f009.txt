import dayjs from "dayjs";

dayjs.extend(timezone);

export function suggestionSummary(
  sf: Record<string, unknown>,
): string | null {
  const outcome = typeof sf.outcome === "string" ? sf.outcome : null;
  if (!outcome) return null;
  const tin =
    typeof sf.time_in === "string"
      ? dayjs(sf.time_in).tz(DEFAULT_SITE_TZ).format("HH:mm")
      : null;
  const tout =
    typeof sf.time_out === "string"
      ? dayjs(sf.time_out).tz(DEFAULT_SITE_TZ).format("HH:mm")
      : null;
  if (outcome === "worked" && tin && tout) {
    return `${strings.exceptions.suggestWorked} ${tin}\u2013${tout}`;
  }
  return outcome.replace(/_/g, " ");
}

import { absoluteUrl, wrappedFetch } from "./client";

export type EmployeeRef = {
  id: string;
  employee_code: string;
  first_name: string;
  last_name: string;
};

const PAGE_LIMIT = 500;
const MAX_PAGES = 20;

export async function fetchEmployees(): Promise<EmployeeRef[]> {
  const all: EmployeeRef[] = [];
  let cursor: string | null = null;
  let pages = 0;
  for (;;) {
    if (pages >= MAX_PAGES) {
      console.warn(`fetchEmployees: stopped at MAX_PAGES=${MAX_PAGES}`);
      break;
    }
    const qs = new URLSearchParams();
    qs.set("limit", String(PAGE_LIMIT));
    if (cursor) qs.set("cursor", cursor);
    const res = await wrappedFetch(
      new Request(absoluteUrl(`/api/v1/employees?${qs.toString()}`), {
        credentials: "include",
      }),
    );
    if (!res.ok) {
      throw new Error(`http_${res.status}`);
    }
    const body = (await res.json().catch(() => ({}))) as {
      items?: unknown;
      next_cursor?: unknown;
    };
    if (Array.isArray(body.items)) all.push(...(body.items as EmployeeRef[]));
    cursor = typeof body.next_cursor === "string" ? body.next_cursor : null;
    pages += 1;
    if (!cursor) break;
  }
  return all;
}

function toTimeInput(value: string | null | undefined): string {
  if (!value) return "";
  const d = dayjs(value).tz(DEFAULT_SITE_TZ);
  return d.isValid() ? d.format("HH:mm") : "";
}

export function AdjustmentForm({
  workDate,
  initial,
  onSubmit,
  submitLabel,
  busy,
}: Props) {
  const [outcome, setOutcome] = useState<AdjustmentOutcome>(
    initial?.outcome ?? "worked",
  );
  const [timeIn, setTimeIn] = useState<string>(toTimeInput(initial?.time_in));
  const [timeOut, setTimeOut] = useState<string>(
    toTimeInput(initial?.time_out),
  );
  const [reasonCode, setReasonCode] = useState<string>(
    initial?.reason_code ?? "",
  );
  const [note, setNote] = useState<string>(initial?.note ?? "");
  const [error, setError] = useState<string | null>(null);

  const requiresTimes = outcome === "worked";
  const reasonIsOther = reasonCode.trim().toLowerCase() === "other";

  const submit = () => {
    setError(null);
    if (!reasonCode.trim()) {
      setError(strings.adjustments.reasonRequired);
      return;
    }
    if (requiresTimes && (!timeIn || !timeOut)) {
      setError(strings.adjustments.timesRequired);
      return;
    }
    if (requiresTimes && timeIn && timeOut) {
      const tin = dayjs.tz(`${workDate}T${timeIn}:00`, DEFAULT_SITE_TZ);
      const tout = dayjs.tz(`${workDate}T${timeOut}:00`, DEFAULT_SITE_TZ);
      if (!tin.isValid() || !tout.isValid() || !tout.isAfter(tin)) {
        setError(strings.adjustments.timeOrder);
        return;
      }
    }
    if (reasonIsOther && note.trim().length < 10) {
      setError(strings.adjustments.noteRequiredForOther);
      return;
    }
    onSubmit({
      outcome,
      time_in:
        requiresTimes && timeIn
          ? dayjs.tz(`${workDate}T${timeIn}:00`, DEFAULT_SITE_TZ).toISOString()
          : null,
      time_out:
        requiresTimes && timeOut
          ? dayjs
              .tz(`${workDate}T${timeOut}:00`, DEFAULT_SITE_TZ)
              .toISOString()
          : null,
      reason_code: reasonCode.trim(),
      note: note.trim() ? note.trim() : null,
    });
  };

  return (
    <Stack gap="sm">
      {error ? <Alert color="red">{error}</Alert> : null}
      <Select
        label={strings.adjustments.outcome}
        data={ADJUSTMENT_OUTCOMES.map((o) => ({
          value: o,
          label: o.replace(/_/g, " "),
        }))}
        value={outcome}
        onChange={(v) => v && setOutcome(v as AdjustmentOutcome)}
        allowDeselect={false}
        data-testid="adj-outcome"
      />
      {requiresTimes ? (
        <Group grow>
          <TextInput
            label={strings.adjustments.timeIn}
            type="time"
            value={timeIn}
            onChange={(e) => setTimeIn(e.currentTarget.value)}
            data-testid="adj-time-in"
          />
          <TextInput
            label={strings.adjustments.timeOut}
            type="time"
            value={timeOut}
            onChange={(e) => setTimeOut(e.currentTarget.value)}
            data-testid="adj-time-out"

  { label: strings.exceptions.statusDismissed, value: "dismissed" },

export function ExceptionsPage() {
  const qc = useQueryClient();
  const [statusFilter, setStatusFilter] = useState<string>("open");
  const [typeFilter, setTypeFilter] = useState<string[]>([]);
  const [modal, setModal] = useState<ModalState>(null);
  const [serverError, setServerError] = useState<string | null>(null);

  const employees = useQuery({
    queryKey: ["refs", "employees"],
    queryFn: () => fetchEmployees(),
    staleTime: 5 * 60 * 1000,
  });

  const exceptions = useQuery({
    queryKey: ["exceptions", statusFilter],
    queryFn: () => fetchExceptions({ status: statusFilter }),
  });

  const resolve = useMutation({
    mutationFn: (vars: { id: string; body: ResolveRequest }) =>
      resolveException(vars.id, vars.body),
    onSuccess: () => {
      setModal(null);
      setServerError(null);
      qc.invalidateQueries({ queryKey: ["exceptions"] });
    },
    onError: (err) => {
      const code = err instanceof Error ? err.message : "unknown_error";
      setServerError(mapErrorCode(code));
    },
  });

  const empMap = new Map<string, { name: string; code: string }>();
  for (const e of employees.data ?? []) {
    empMap.set(e.id, {
      name: `${e.first_name} ${e.last_name}`.trim(),
      code: e.employee_code,
    });
  }

  const applySuggestion = (row: ExceptionRow) => {
    setServerError(null);
    const summary =
      suggestionSummary(row.suggested_fix) ?? exceptionTypeLabel(row.type);
    if (!window.confirm(strings.exceptions.applyConfirm(summary))) return;
    resolve.mutate({
      id: row.id,
      body: { action: "apply_suggestion", custom_adjustment: null, note: null },
    });
  };

  const rawItems = exceptions.data?.items ?? [];
  const items =
    typeFilter.length === 0
      ? rawItems
      : rawItems.filter((r) => typeFilter.includes(r.type));

  return (
    <Stack gap="md">
      <Title order={2}>{strings.exceptions.title}</Title>
      {employees.isError ? (
        <Alert color="red" data-testid="employees-error">
          <Group justify="space-between" align="center">
            <Text size="sm">{strings.exceptions.employeesError}</Text>
            <Button
              size="xs"
              variant="default"
              onClick={() => employees.refetch()}
              data-testid="employees-retry"
            >
              {strings.exceptions.employeesRetry}
            </Button>
          </Group>
        </Alert>

        <Alert color="red" data-testid="direct-error">
          {serverError}
        </Alert>
      ) : null}
      <Paper withBorder p="md">
        <Group gap="sm" wrap="wrap" align="end">
          <SegmentedControl
            value={statusFilter}
            onChange={(v) => setStatusFilter(v)}
            data={STATUS_OPTIONS}
          />
          <MultiSelect
            label={strings.exceptions.typeFilter}
            data={TYPE_OPTIONS}
            value={typeFilter}
            onChange={setTypeFilter}
            placeholder={strings.exceptions.allTypes}
            description={strings.exceptions.multiTypeNote}
            clearable
          />
        </Group>
      </Paper>
      <Paper withBorder p="md">
        {exceptions.isPending ? (
          <Group justify="center" p="xl">
            <Loader />
          </Group>
        ) : exceptions.isError ? (
          <Text c="red">{strings.exceptions.error}</Text>
        ) : items.length === 0 ? (
          <Text c="dimmed">{strings.exceptions.empty}</Text>
        ) : (
          <Stack gap="sm">
            {items.map((row) => {
              const info = empMap.get(row.employee_id);

                <ExceptionCard
                  key={row.id}
                  exception={row}
                  employeeName={info?.name ?? row.employee_id.slice(0, 8)}
                  employeeCode={info?.code ?? ""}
                  onApplySuggestion={() => applySuggestion(row)}
                  onCustom={() => setModal({ exception: row, action: "custom" })}
                  onDismiss={() =>
                    setModal({ exception: row, action: "dismiss" })
                  }
                  busy={resolve.isPending}
                />
              );
            })}
          </Stack>
        )}
      </Paper>
      <ResolveModal
        opened={modal !== null}
        exception={modal?.exception ?? null}
        initialAction={modal?.action ?? "custom"}
        onClose={() => {
          setModal(null);
          setServerError(null);
        }}
        onResolve={(req) => {
          if (!modal) return;
          resolve.mutate({ id: modal.exception.id, body: req });
        }}
        busy={resolve.isPending}
        serverError={serverError}
      />
    </Stack>
  );
}
