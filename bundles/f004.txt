import { useQuery } from "@tanstack/react-query";
import { wrappedFetch } from "../api/client";
import { useAuth } from "./AuthContext";

export type Me = {
  id: string;
  username: string;
  role: string;
  employee_id: string | null;
  permissions: string[];
  must_change_password: boolean;
};

export const meQueryKey = ["me"] as const;

export async function fetchMe(): Promise<Me> {
  const res = await wrappedFetch(
    new Request("/api/v1/auth/me", { credentials: "include" }),
  );
  if (!res.ok) {
    throw new Error(`/auth/me failed: ${res.status}`);
  }
  return (await res.json()) as Me;
}

export function useMe() {
  const { status } = useAuth();
  return useQuery<Me, Error>({
    queryKey: meQueryKey,
    queryFn: fetchMe,
    enabled: status === "authenticated",
    staleTime: Infinity,
    retry: 1,
  });
}

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from "react";
import { setAccessToken, setOnUnauthorized } from "../api/client";
import { queryClient } from "../lib/queryClient";
import { meQueryKey } from "./useMe";
import {
  AuthRequestError,
  changePasswordRequest,
  loginRequest,
  logoutRequest,
} from "./api";

export type AuthState = {
  status: "authenticated" | "unauthenticated";
  loginError: string | null;
  busy: boolean;
};

export type AuthContextValue = AuthState & {
  login: (identifier: string, password: string) => Promise<boolean>;
  logout: () => Promise<void>;
  changePassword: (current: string, next: string) => Promise<boolean>;
};

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<AuthState>({
    status: "unauthenticated",
    loginError: null,
    busy: false,
  });

  useEffect(() => {
    setOnUnauthorized(() => {
      setAccessToken(null);
      queryClient.clear();
      setState({
        status: "unauthenticated",
        loginError: "Session expired. Please sign in again.",
        busy: false,
      });
    });
    return () => setOnUnauthorized(null);
  }, []);

  const login = useCallback(
    async (identifier: string, password: string): Promise<boolean> => {
      setState((s) => ({ ...s, busy: true, loginError: null }));
      try {
        const result = await loginRequest(identifier, password);
        setAccessToken(result.accessToken);
        queryClient.removeQueries({ queryKey: meQueryKey });
        setState({
          status: "authenticated",
          loginError: null,
          busy: false,
        });
        return true;
      } catch (err) {
        const message =
          err instanceof AuthRequestError
            ? err.apiError.message
            : "Sign-in failed";
        setState({
          status: "unauthenticated",
          loginError: message,
          busy: false,
        });
        return false;
      }
    },
    [],
  );

  const logout = useCallback(async (): Promise<void> => {
    await logoutRequest();
    setAccessToken(null);
    queryClient.clear();
    setState({
      status: "unauthenticated",
      loginError: null,
      busy: false,
    });
  }, []);

  const changePassword = useCallback(
    async (current: string, next: string): Promise<boolean> => {
      setState((s) => ({ ...s, busy: true, loginError: null }));
      try {
        await changePasswordRequest({
          currentPassword: current,
          newPassword: next,
        });
        await queryClient.invalidateQueries({ queryKey: meQueryKey });
        setState((s) => ({ ...s, busy: false }));
        return true;
      } catch (err) {
        const message =
          err instanceof AuthRequestError
            ? err.apiError.message
            : "Change password failed";
        setState((s) => ({ ...s, busy: false, loginError: message }));
        return false;
      }
    },
    [],
  );

  return (
    <AuthContext.Provider value={{ ...state, login, logout, changePassword }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

import type { ReactNode } from "react";
import { Navigate, useLocation } from "react-router-dom";
import { Center, Loader } from "@mantine/core";
import { useAuth } from "./AuthContext";
import { useMe } from "./useMe";

export function ProtectedRoute({ children }: { children: ReactNode }) {
  const { status } = useAuth();
  const me = useMe();
  const location = useLocation();

  if (status === "unauthenticated") {
    return <Navigate to="/login" replace state={{ from: location.pathname }} />;
  }
  if (me.isLoading) {
    return (
      <Center h="50vh">
        <Loader />
      </Center>
    );
  }
  if (me.isError || !me.data) {
    return <Navigate to="/login" replace />;
  }
  if (
    me.data.must_change_password &&
    location.pathname !== "/change-password"
  ) {
    return <Navigate to="/change-password" replace />;
  }
  return <>{children}</>;
}

export function buildNav(permissions: string[]): NavItem[] {
  const set = new Set(permissions);
  return PERMISSION_NAV.filter((entry) => set.has(entry.permission)).map(
    (entry) => entry.item,

import { useState } from "react";
import { Navigate, useNavigate } from "react-router-dom";
import {
  Alert,
  Button,
  Center,
  Container,
  Loader,
  PasswordInput,
  Stack,
  Title,
} from "@mantine/core";
import { useAuth } from "../auth/AuthContext";
import { useMe } from "../auth/useMe";
import { strings } from "../lib/strings";

export function ForcePasswordChangePage() {
  const { loginError, busy, changePassword } = useAuth();
  const me = useMe();
  const navigate = useNavigate();
  const [current, setCurrent] = useState("");
  const [next, setNext] = useState("");
  const [confirm, setConfirm] = useState("");
  const [localError, setLocalError] = useState<string | null>(null);

  if (me.isLoading) {
    return (
      <Center h="50vh">
        <Loader />
      </Center>
    );
  }
  if (me.isError || !me.data) {
    return <Navigate to="/login" replace />;
  }
  if (!me.data.must_change_password) {
    return <Navigate to="/" replace />;
  }

  const onSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLocalError(null);
    if (next.length < 8) {
      setLocalError(strings.changePassword.tooShort);
      return;
    }
    if (next !== confirm) {
      setLocalError(strings.changePassword.mismatch);
      return;
    }
    const ok = await changePassword(current, next);
    if (ok) navigate("/", { replace: true });
  };

  return (
    <Container size="xs" py="xl">
      <form onSubmit={onSubmit}>
        <Stack>
          <Title order={2}>{strings.changePassword.title}</Title>
          <Alert color="blue">{strings.changePassword.intro}</Alert>
          {loginError ? <Alert color="red">{loginError}</Alert> : null}
          {localError ? <Alert color="red">{localError}</Alert> : null}
          <PasswordInput
            label={strings.changePassword.currentLabel}
            value={current}
            onChange={(e) => setCurrent(e.currentTarget.value)}
            required
            autoComplete="current-password"
          />
          <PasswordInput
            label={strings.changePassword.newLabel}
            value={next}
            onChange={(e) => setNext(e.currentTarget.value)}
            required
            autoComplete="new-password"
          />
          <PasswordInput
            label={strings.changePassword.confirmLabel}
            value={confirm}
            onChange={(e) => setConfirm(e.currentTarget.value)}
            required
            autoComplete="new-password"
          />
          <Button type="submit" loading={busy}>
            {strings.changePassword.submit}
          </Button>
        </Stack>
      </form>
    </Container>
  );
}
