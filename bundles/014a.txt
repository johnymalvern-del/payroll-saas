"""Payroll refusals (Chapter 4).

Raised by services when a computation cannot proceed because a statutory
table, a tenant setting, or a required employee field is missing. Endpoints
translate these into HTTP responses with the code and status supplied here.
"""
from __future__ import annotations


class PayrollRefusal(Exception):
    def __init__(self, code: str, message: str, http_status: int = 409) -> None:
        self.code = code
        self.message = message
        self.http_status = http_status
        super().__init__(message)


# Statutory data
STATUTORY_TABLE_MISSING = "statutory_table_missing"
STATUTORY_TABLE_EXPIRED = "statutory_table_expired"
STATUTORY_TABLE_INVALID = "statutory_table_invalid"

# Employee field / setting
AGE_REFERENCE_DATE_UNSET = "age_reference_date_unset"
MISSING_DATE_OF_BIRTH = "missing_date_of_birth"
MISSING_CONTRACTED_HOURS = "missing_contracted_hours"

# Tenant settings
SDL_APPLIES_UNSET = "sdl_applies_unset"

# Specific statutory tables
UIF_CEILING_MISSING = "uif_ceiling_missing"
RETIREMENT_LIMIT_MISSING = "retirement_limit_missing"

# Method gate
TAX_METHOD_UNSUPPORTED = "tax_method_unsupported"

# F7: legal-order deductions never deferred
DEDUCTION_EXCEEDS_FLOOR_LEGAL_ORDER = "deduction_exceeds_floor_legal_order"

=== FILE: backend/app/services/payroll_uif_sdl.py ===
"""UIF, SDL, and retirement-limit calculators (Chapter 4, Bundle 004a-iii part 1).

Pure functions. No DB, no clock, no side effects. Callers supply the
effective-dated statutory tables (from payroll_tax.load_tax_tables or the
equivalent row-dict loaders) as plain dicts. Values are NOT baked in — no
statutory number appears anywhere in this file.

Rules encoded (E4, E7, D10):
  UIF: ceiling applies to the base BEFORE the rate (per SARS practice); rate
       then applied; result rounded 2dp half-up.
  UIF employer: identical mechanics with employer_rate.
  SDL: employer cost only, not an employee deduction. Refuses when the tenant
       setting `sdl_applies` is None (deliberately unset).
  Retirement 11F cap: min(percentage_of_taxable_income * period_gross,
       annual_cap / periods_per_year). The employee's intended contribution is
       then capped at this value (excess carried forward is deferred — see
       004a-iii part 2).
"""
from __future__ import annotations

from decimal import Decimal, ROUND_HALF_UP
from typing import Any

from app.services import payroll_refusals as R


def _q(amount: Decimal) -> Decimal:
    return amount.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


def _dec(v: Any) -> Decimal:
    return Decimal(str(v))


def _value_of(row: dict) -> Decimal:
    if "amount" in row:
        return _dec(row["amount"])
    if "value" in row:
        return _dec(row["value"])
    raise R.PayrollRefusal(
        R.STATUTORY_TABLE_INVALID,
        f"statutory row {row.get('row_key')!r} has neither 'amount' nor 'value'",
        500,
    )


def _as_dict(rows: list[dict]) -> dict[str, Decimal]:
    """Turn statutory_table rows into {row_key: decimal}."""
    return {row["row_key"]: _value_of(row) for row in rows}


# ---------------------------------------------------------------------------
# UIF
# ---------------------------------------------------------------------------

def compute_uif_employee(
    period_gross_for_uif: Decimal,
    uif_rows: list[dict],
    uif_exempt: bool = False,
) -> Decimal:
    if uif_exempt:
        return Decimal("0.00")
    t = _as_dict(uif_rows)
    rate = t.get("employee_rate")
    ceiling = t.get("monthly_ceiling")
    if rate is None or ceiling is None:
        raise R.PayrollRefusal(
            R.UIF_CEILING_MISSING,
            "uif table is missing employee_rate and/or monthly_ceiling",
            409,
        )
    base = min(period_gross_for_uif, ceiling)
    return _q(base * rate)


def compute_uif_employer(
    period_gross_for_uif: Decimal,
    uif_rows: list[dict],
    uif_exempt: bool = False,
) -> Decimal:
    if uif_exempt:
        return Decimal("0.00")
    t = _as_dict(uif_rows)
    rate = t.get("employer_rate")
    ceiling = t.get("monthly_ceiling")
    if rate is None or ceiling is None:
        raise R.PayrollRefusal(
            R.UIF_CEILING_MISSING,
            "uif table is missing employer_rate and/or monthly_ceiling",
            409,
        )
    base = min(period_gross_for_uif, ceiling)
    return _q(base * rate)


# ---------------------------------------------------------------------------
# SDL (employer cost only)
# ---------------------------------------------------------------------------

def compute_sdl(
    period_gross_for_sdl: Decimal,
    sdl_rows: list[dict],
    sdl_applies: bool | None,
) -> Decimal:
    if sdl_applies is None:
        raise R.PayrollRefusal(
            R.SDL_APPLIES_UNSET,
            "tenant setting payroll.sdl_applies is unset; set it before first finalise",
            409,
        )
    if not sdl_applies:
        return Decimal("0.00")
    t = _as_dict(sdl_rows)
    rate = t.get("rate")
    if rate is None:
        raise R.PayrollRefusal(
            R.STATUTORY_TABLE_MISSING,
            "sdl table is missing rate",
            409,
        )
    return _q(period_gross_for_sdl * rate)


# ---------------------------------------------------------------------------
# Retirement 11F cap
# ---------------------------------------------------------------------------

def compute_retirement_limit(
    period_gross: Decimal,
    retirement_rows: list[dict],
    periods_per_year: int,
) -> Decimal:
    """Maximum retirement-fund deduction for one pay period.

    Returns min(percentage_of_taxable_income * period_gross,
               annual_cap / periods_per_year), rounded 2dp.
    The caller then takes min(intended_deduction, limit) to determine how much
    actually reduces taxable income this period.
    """
    if periods_per_year <= 0:
        raise R.PayrollRefusal(
            R.STATUTORY_TABLE_INVALID,
            "periods_per_year must be positive",
            500,
        )
    t = _as_dict(retirement_rows)
    pct = t.get("percentage_of_taxable_income")
    cap = t.get("annual_cap")
    if pct is None or cap is None:
        raise R.PayrollRefusal(
            R.RETIREMENT_LIMIT_MISSING,
            "retirement_limit table is missing percentage_of_taxable_income and/or annual_cap",
            409,
        )
    per_period_cap = cap / Decimal(periods_per_year)
    by_pct = period_gross * pct
    return _q(min(by_pct, per_period_cap))


def apply_retirement_limit(
    intended_contribution: Decimal,
    period_gross: Decimal,
    retirement_rows: list[dict],
    periods_per_year: int,
) -> tuple[Decimal, Decimal]:
    """Return (allowed_deduction, excess_carried_forward)."""
    if intended_contribution < 0:
        raise R.PayrollRefusal(
            R.STATUTORY_TABLE_INVALID,
            "intended_contribution must be non-negative",
            500,
        )
    limit = compute_retirement_limit(period_gross, retirement_rows, periods_per_year)
    allowed = min(intended_contribution, limit)
    excess = max(Decimal("0.00"), intended_contribution - allowed)
    return allowed, excess

"""UIF / SDL / retirement-limit calculators (Chapter 4, Bundle 004a-iii part 1).

Every expected figure is hand-worked in the comment. Fixtures are synthetic —
no real statutory number appears anywhere.
"""
from __future__ import annotations

from decimal import Decimal

import pytest

from app.services import payroll_refusals as R
from app.services import payroll_uif_sdl as U


# ---------------------------------------------------------------------------
# Synthetic fixtures (obviously fake)
# ---------------------------------------------------------------------------

UIF_ROWS = [
    {"row_key": "employee_rate", "value": "0.01"},
    {"row_key": "employer_rate", "value": "0.01"},
    {"row_key": "monthly_ceiling", "value": "500"},
]

SDL_ROWS = [
    {"row_key": "rate", "value": "0.01"},
    {"row_key": "exemption_threshold_annual", "value": "50000"},
]

RETIREMENT_ROWS = [
    {"row_key": "percentage_of_taxable_income", "value": "0.10"},
    {"row_key": "annual_cap", "value": "1000"},
]


# ---------------------------------------------------------------------------
# UIF employee
# ---------------------------------------------------------------------------

def test_uif_employee_under_ceiling():
    # base 300, 1% = 3.00
    assert U.compute_uif_employee(Decimal(300), UIF_ROWS) == Decimal("3.00")


def test_uif_employee_at_ceiling():
    # base 500, 1% = 5.00
    assert U.compute_uif_employee(Decimal(500), UIF_ROWS) == Decimal("5.00")


def test_uif_employee_above_ceiling_capped():
    # base 1000 but ceiling 500, so 500*1% = 5.00
    assert U.compute_uif_employee(Decimal(1000), UIF_ROWS) == Decimal("5.00")


def test_uif_employee_rounds_half_up():
    # base 333, 1% = 3.33 (exactly)
    # base 355, 1% = 3.55
    # base 355.5 would need half-up; use 355.5 * 0.01 = 3.555 -> 3.56
    rows = [
        {"row_key": "employee_rate", "value": "0.01"},
        {"row_key": "employer_rate", "value": "0.01"},
        {"row_key": "monthly_ceiling", "value": "10000"},
    ]
    assert U.compute_uif_employee(Decimal("355.50"), rows) == Decimal("3.56")


def test_uif_employee_exempt_returns_zero():
    assert U.compute_uif_employee(Decimal(1000), UIF_ROWS, uif_exempt=True) == Decimal("0.00")


def test_uif_employee_missing_rate_raises():
    rows = [{"row_key": "monthly_ceiling", "value": "500"}]
    with pytest.raises(R.PayrollRefusal) as exc_info:
        U.compute_uif_employee(Decimal(100), rows)
    assert exc_info.value.code == R.UIF_CEILING_MISSING


# ---------------------------------------------------------------------------
# UIF employer
# ---------------------------------------------------------------------------

def test_uif_employer_under_ceiling():
    # base 300, 1% = 3.00
    assert U.compute_uif_employer(Decimal(300), UIF_ROWS) == Decimal("3.00")


def test_uif_employer_above_ceiling_capped():
    assert U.compute_uif_employer(Decimal(5000), UIF_ROWS) == Decimal("5.00")


def test_uif_employer_exempt_returns_zero():
    assert U.compute_uif_employer(Decimal(1000), UIF_ROWS, uif_exempt=True) == Decimal("0.00")


# ---------------------------------------------------------------------------
# SDL
# ---------------------------------------------------------------------------

def test_sdl_applies_true():
    # base 1000, 1% = 10.00
    assert U.compute_sdl(Decimal(1000), SDL_ROWS, sdl_applies=True) == Decimal("10.00")


def test_sdl_applies_false_returns_zero():
    assert U.compute_sdl(Decimal(1000), SDL_ROWS, sdl_applies=False) == Decimal("0.00")


def test_sdl_unset_raises():
    with pytest.raises(R.PayrollRefusal) as exc_info:
        U.compute_sdl(Decimal(1000), SDL_ROWS, sdl_applies=None)
    assert exc_info.value.code == R.SDL_APPLIES_UNSET


def test_sdl_missing_rate_raises():
    rows = [{"row_key": "exemption_threshold_annual", "value": "50000"}]
    with pytest.raises(R.PayrollRefusal) as exc_info:
        U.compute_sdl(Decimal(1000), rows, sdl_applies=True)
    assert exc_info.value.code == R.STATUTORY_TABLE_MISSING


# ---------------------------------------------------------------------------
# Retirement 11F cap
# ---------------------------------------------------------------------------

def test_retirement_limit_pct_binds():
    # 1000 * 10% = 100; cap/12 = 1000/12 = 83.33
    # min(100, 83.33) = 83.33
    assert U.compute_retirement_limit(
        Decimal(1000), RETIREMENT_ROWS, periods_per_year=12,
    ) == Decimal("83.33")


def test_retirement_limit_cap_binds():
    # 10000 * 10% = 1000; cap/12 = 83.33
    # min(1000, 83.33) = 83.33
    assert U.compute_retirement_limit(
        Decimal(10000), RETIREMENT_ROWS, periods_per_year=12,
    ) == Decimal("83.33")


def test_retirement_limit_pct_binds_low_income():
    # 100 * 10% = 10; cap/12 = 83.33
    # min(10, 83.33) = 10.00
    assert U.compute_retirement_limit(
        Decimal(100), RETIREMENT_ROWS, periods_per_year=12,
    ) == Decimal("10.00")


def test_retirement_limit_zero_periods_raises():
    with pytest.raises(R.PayrollRefusal):
        U.compute_retirement_limit(Decimal(100), RETIREMENT_ROWS, periods_per_year=0)


def test_retirement_limit_missing_pct_raises():
    rows = [{"row_key": "annual_cap", "value": "1000"}]
    with pytest.raises(R.PayrollRefusal) as exc_info:
        U.compute_retirement_limit(Decimal(100), rows, periods_per_year=12)
    assert exc_info.value.code == R.RETIREMENT_LIMIT_MISSING


def test_apply_retirement_limit_under():
    # intended 50, limit 83.33 -> allowed 50, excess 0
    allowed, excess = U.apply_retirement_limit(
        Decimal(50), Decimal(1000), RETIREMENT_ROWS, periods_per_year=12,
    )
    assert allowed == Decimal(50)
    assert excess == Decimal("0.00")


def test_apply_retirement_limit_over():
    # intended 200, limit 83.33 -> allowed 83.33, excess 116.67
    allowed, excess = U.apply_retirement_limit(
        Decimal(200), Decimal(1000), RETIREMENT_ROWS, periods_per_year=12,
    )
    assert allowed == Decimal("83.33")
    assert excess == Decimal("116.67")


def test_apply_retirement_limit_negative_raises():
    with pytest.raises(R.PayrollRefusal):
        U.apply_retirement_limit(
            Decimal(-1), Decimal(1000), RETIREMENT_ROWS, periods_per_year=12,
        )
