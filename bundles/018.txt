"""Current-user endpoint (B-1).

GET /auth/me returns the authenticated user's id, username, role,
employee_id, and permissions from the live ROLE_PERMISSIONS matrix
(app/core/permissions.py), never a client-side hard-coded copy.

must_change_password reflects User.password_changed_at: NULL means the
user has never changed their password and must do so before continuing.
If the column is absent (unexpected) the endpoint returns False rather
than blocking the user.
"""
from __future__ import annotations

import uuid
from typing import Annotated, Any

from fastapi import APIRouter, Depends
from pydantic import BaseModel, ConfigDict

from app.core.auth_deps import get_current_user
from app.core.permissions import ROLE_PERMISSIONS

router = APIRouter(prefix="/auth/me", tags=["auth"])


class UserMeOut(BaseModel):
    model_config = ConfigDict(from_attributes=True)

    id: uuid.UUID
    username: str
    role: str
    employee_id: uuid.UUID | None
    permissions: list[str]
    must_change_password: bool


@router.get("", response_model=UserMeOut)
def get_me(
    user: Annotated[Any, Depends(get_current_user)],
) -> UserMeOut:
    role = getattr(user, "role", None) or "unknown"
    permissions = sorted(
        p.value for p in ROLE_PERMISSIONS.get(role, frozenset())
    )
    if hasattr(user, "password_changed_at"):
        must_change = user.password_changed_at is None
    else:
        must_change = False
    return UserMeOut(
        id=user.id,
        username=user.username,
        role=role,
        employee_id=getattr(user, "employee_id", None),
        permissions=permissions,
        must_change_password=must_change,
    )

@dataclass
class _DisposableTenant:
    slug: str
    tenant_db_name: str
    subscriber_id: uuid.UUID


@pytest.fixture()
def disposable_tenant() -> Iterator[_DisposableTenant]:
    from app.db.control.models import Subscriber
    from app.db.control.session import control_session
    from app.services.provisioning import (
        _drop_tenant_database,
        _tenant_db_name,
        provision_tenant,
    )
    slug = f"auth-me-{uuid.uuid4().hex[:8]}"
    result = provision_tenant(
        slug=slug,
        name="Auth Me Test",
        admin_username="admin",
        admin_password_hash=hash_password("TestPassword123!"),
    )
    db_name = _tenant_db_name(slug)
    try:
        yield _DisposableTenant(
            slug=result.slug,
            tenant_db_name=result.tenant_db_name,
            subscriber_id=result.subscriber_id,
        )
    finally:
        try:
            _drop_tenant_database(db_name)
        except Exception:
            pass
        with control_session() as session:
            row = (
                session.query(Subscriber)
                .filter(Subscriber.slug == slug)
                .one_or_none()
            )
            if row is not None:
                session.delete(row)


@pytest.fixture()
def tenant_session(disposable_tenant):
    engine = get_tenant_engine(disposable_tenant.tenant_db_name)
    session = Session(engine)
    try:
        yield session, engine
    finally:
        session.close()
        engine.dispose()


def _mk_user(
    session,
    *,
    username,
    role,
    password,
    employee_id=None,
    password_changed_at="set",
):
    if password_changed_at == "set":
        password_changed_at = datetime(2025, 1, 1, tzinfo=timezone.utc)
    u = User(
        id=uuid.uuid4(),
        username=username,
        password_hash=hash_password(password),
        role=role,
        employee_id=employee_id,
        password_changed_at=password_changed_at,
    )
    session.add(u)
    session.commit()
    return u


def _client() -> TestClient:
    from app.main import app
    return TestClient(app)


def _login(client, slug, username, password):
    r = client.post(
        "/api/v1/auth/login",
        headers={"X-Tenant-Slug": slug},
        json={"identifier": username, "password": password},
    )
    assert r.status_code == 200, r.text
    return r.json()["access_token"]

def test_auth_me_returns_payroll_admin_role_and_permissions(
    disposable_tenant, tenant_session,
):
    session, _ = tenant_session
    _mk_user(session, username="pa-1", role="payroll_admin", password="PaPass123!")
    client = _client()
    token = _login(client, disposable_tenant.slug, "pa-1", "PaPass123!")
    r = client.get(
        "/api/v1/auth/me",
        headers={
            "X-Tenant-Slug": disposable_tenant.slug,
            "Authorization": f"Bearer {token}",
        },
    )
    assert r.status_code == 200, r.text
    body = r.json()
    assert body["role"] == "payroll_admin"
    assert body["username"] == "pa-1"
    expected = sorted(p.value for p in ROLE_PERMISSIONS["payroll_admin"])
    assert sorted(body["permissions"]) == expected
    assert body["must_change_password"] is False
