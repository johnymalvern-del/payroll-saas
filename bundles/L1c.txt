"""Leave engine helpers (Chapter 3, 3B).

compute_cycle_bounds: returns (cycle_start, cycle_end) for an employee
given a leave type and an as-of date.

available_balance: net available for a balance row. The full grant is
tracked separately as `entitled` (informational), but the amount actually
reachable in the current cycle is accrued + carried_in + adjusted - taken
- pending. This avoids the entitled/accrued double-count flagged in
amendment 4.

Accrual service (accrue_all_employees, carry-over) lands in L1d.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import date, timedelta
from decimal import Decimal

from app.core.errors import AppError
from app.db.tenant.models.leave import LeaveBalance, LeaveType
from app.db.tenant.models.payroll_settings import PayrollSettings
from app.db.tenant.models.people import Employee


_ZERO = Decimal("0")


def _d(x: Decimal | None) -> Decimal:
    return x if x is not None else _ZERO


@dataclass(frozen=True)
class CycleBounds:
    start: date
    end: date


def _calendar_bounds(as_of_date: date) -> CycleBounds:
    return CycleBounds(
        start=date(as_of_date.year, 1, 1),
        end=date(as_of_date.year, 12, 31),
    )


def _anniversary_bounds(hire_date: date, as_of_date: date) -> CycleBounds:
    month = hire_date.month
    day = hire_date.day
    try:
        this_year_anchor = date(as_of_date.year, month, day)
    except ValueError:
        this_year_anchor = date(as_of_date.year, month, 28)
    if this_year_anchor <= as_of_date:
        start = this_year_anchor
    else:
        try:
            start = date(as_of_date.year - 1, month, day)
        except ValueError:
            start = date(as_of_date.year - 1, month, 28)
    try:
        next_anchor = date(start.year + 1, start.month, start.day)
    except ValueError:
        next_anchor = date(start.year + 1, start.month, 28)
    return CycleBounds(start=start, end=next_anchor - timedelta(days=1))


def _fixed_bounds(anchor: date, as_of_date: date) -> CycleBounds:
    start = anchor
    while start <= as_of_date:
        try:
            next_start = date(start.year + 1, start.month, start.day)
        except ValueError:
            next_start = date(start.year + 1, start.month, 28)
        end = next_start - timedelta(days=1)
        if start <= as_of_date <= end:
            return CycleBounds(start=start, end=end)
        start = next_start
    raise AppError(
        "leave_cycle_out_of_range",
        "as_of_date is before the configured fixed cycle anchor",
        422,
    )


def compute_cycle_bounds(
    employee: Employee,
    leave_type: LeaveType,
    settings: PayrollSettings | None,
    as_of_date: date,
) -> CycleBounds:
    cycle = leave_type.cycle
    if cycle == "calendar":
        return _calendar_bounds(as_of_date)
    if cycle == "anniversary":
        return _anniversary_bounds(employee.hire_date, as_of_date)
    if cycle == "fixed":
        anchor = settings.leave_cycle_fixed_anchor if settings else None
        if anchor is None:
            raise AppError(
                "leave_cycle_anchor_not_configured",
                "leave_cycle_fixed_anchor is unset; cycle='fixed' requires it",
                409,
            )
        return _fixed_bounds(anchor, as_of_date)
    raise AppError("invalid_cycle", f"Unknown cycle {cycle!r}", 500)


def available_balance(balance: LeaveBalance) -> Decimal:
    return (
        _d(balance.accrued)
        + _d(balance.carried_in)
        + _d(balance.adjusted)
        - _d(balance.taken)
        - _d(balance.pending)
    )

"""Leave balance read + admin adjustment (Chapter 3, 3B.2).

Permissions:
  GET*  : LEAVE_APPROVE (all in scope) OR SELF_VIEW (own only)
  POST /adjust: LEAVE_APPROVE for a live cycle; PAYROLL_FINALISE for a
    closed cycle (cycle_end < today).

Adjustment:
  - Creates a LeaveBalanceLedger row (append-only) and increments the
    balance's `adjusted` field inside one transaction.
  - 422 insufficient_balance when the resulting available < 0 and the
    leave type's allow_negative is False.
  - Missing balance row for a given live cycle is auto-created with all
    fields zeroed before the delta is applied.
  - `entitled`, `accrued`, `taken`, `pending`, `carried_in` are NEVER
    touched by adjust — only `adjusted`.

Available = accrued + carried_in + adjusted - taken - pending.
`entitled` is informational only and is NOT added (avoids double-count,
see amendment 4 / test_available_balance_uses_accrued_not_entitled).
"""
from __future__ import annotations

import base64
import json
import uuid
from datetime import date, datetime, timezone
from typing import Annotated, Any

from fastapi import APIRouter, Depends, Query, status
from sqlalchemy import func, or_, select

from app.api.deps import TenantDbDep
from app.api.v1.leave_balances_schemas import (
    BalanceAdjustIn,
    BalanceAdjustOut,
    LeaveBalanceListOut,
    LeaveBalanceOut,
)
from app.core.auth_deps import get_current_user
from app.core.errors import AppError
from app.core.permissions import Permission, check_permission
from app.core.scope import ScopeViolation, apply_scope, assert_in_scope
from app.db.tenant.models.leave import (
    LeaveBalance,
    LeaveBalanceLedger,
    LeaveType,
)
from app.db.tenant.models.payroll_settings import PayrollSettings
from app.db.tenant.models.people import Employee
from app.db.tenant.models.platform import AuditLog
from app.payroll_engine.leave import available_balance, compute_cycle_bounds
from app.services.audit_redact import redact_for_audit

router = APIRouter(prefix="/leave/balances", tags=["leave-balances"])

_DEFAULT_LIMIT = 100
_MAX_LIMIT = 500


def _not_found() -> AppError:
    return AppError("not_found", "Leave balance not found", 404)


def _to_out(row: LeaveBalance) -> LeaveBalanceOut:
    out = LeaveBalanceOut.model_validate(row)
    out.available = available_balance(row)
    return out


def _encode_cursor(created_at: datetime, bid: uuid.UUID) -> str:
    payload = {"c": created_at.isoformat(), "i": str(bid)}
    raw = json.dumps(payload, separators=(",", ":")).encode("utf-8")
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")


def _decode_cursor(cursor: str) -> tuple[datetime, uuid.UUID]:
    padded = cursor + "=" * (-len(cursor) % 4)
    try:
        raw = base64.urlsafe_b64decode(padded.encode("ascii")).decode("utf-8")
        payload = json.loads(raw)
        ts = datetime.fromisoformat(payload["c"])
        if ts.tzinfo is None:
            ts = ts.replace(tzinfo=timezone.utc)
        return ts, uuid.UUID(payload["i"])
    except Exception:
        raise AppError(
            "invalid_cursor", "The pagination cursor is malformed", 400
        ) from None


def _forbidden() -> AppError:
    return AppError(
        "forbidden", "Not permitted to view leave balances", 403
    )


def _require_any_view(user: Any) -> None:
    if check_permission(user, Permission.LEAVE_APPROVE):
        return
    if check_permission(user, Permission.SELF_VIEW):
        return
    raise _forbidden()


def _is_self_only(user: Any) -> bool:
    return (
        check_permission(user, Permission.SELF_VIEW)
        and not check_permission(user, Permission.LEAVE_APPROVE)
    )


@router.get("", response_model=LeaveBalanceListOut)
def list_leave_balances(
    db: TenantDbDep,
    user: Annotated[Any, Depends(get_current_user)],
    employee_id: uuid.UUID | None = Query(None),
    leave_type_id: uuid.UUID | None = Query(None),
    cycle_start_gte: date | None = Query(None),
    cycle_start_lte: date | None = Query(None),
    limit: int = Query(_DEFAULT_LIMIT, ge=1, le=_MAX_LIMIT),
    cursor: str | None = Query(None),
) -> LeaveBalanceListOut:
    _require_any_view(user)

    q = select(LeaveBalance)

    if _is_self_only(user):
        own = getattr(user, "employee_id", None)
        if own is None:
            raise AppError(
                "no_employee_record",
                "User has no employee record",
                404,
            )
        q = q.where(LeaveBalance.employee_id == own)
    else:
        if employee_id is not None:
            q = q.where(LeaveBalance.employee_id == employee_id)
        q = apply_scope(q, user, db, model=LeaveBalance)

    if leave_type_id is not None:
        q = q.where(LeaveBalance.leave_type_id == leave_type_id)
    if cycle_start_gte is not None:
        q = q.where(LeaveBalance.cycle_start >= cycle_start_gte)
    if cycle_start_lte is not None:
        q = q.where(LeaveBalance.cycle_start <= cycle_start_lte)

    if cursor is not None:
        ts, cid = _decode_cursor(cursor)
        q = q.where(
            or_(
                LeaveBalance.created_at > ts,
                (LeaveBalance.created_at == ts) & (LeaveBalance.id > cid),
            )
        )

    q = q.order_by(
        LeaveBalance.created_at.asc(), LeaveBalance.id.asc()
    ).limit(limit + 1)
    rows = list(db.scalars(q).all())

    next_cursor = None
    if len(rows) > limit:
        rows = rows[:limit]
        last = rows[-1]
        next_cursor = _encode_cursor(last.created_at, last.id)

    return LeaveBalanceListOut(
        items=[_to_out(r) for r in rows],
        next_cursor=next_cursor,
    )


@router.get("/{balance_id}", response_model=LeaveBalanceOut)
def get_leave_balance(
    balance_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(get_current_user)],
) -> LeaveBalanceOut:
    _require_any_view(user)
    row = db.get(LeaveBalance, balance_id)
    if row is None:
        raise _not_found()
    if _is_self_only(user):
        own = getattr(user, "employee_id", None)
        if own is None or row.employee_id != own:
            raise _not_found()
    else:
        try:
            assert_in_scope(user, row.employee_id, db)
        except ScopeViolation:
            raise _not_found() from None
    return _to_out(row)


@router.post("/adjust", response_model=BalanceAdjustOut)
def adjust_leave_balance(
    payload: BalanceAdjustIn,
    db: TenantDbDep,
    user: Annotated[Any, Depends(get_current_user)],
) -> BalanceAdjustOut:
    can_approve = check_permission(user, Permission.LEAVE_APPROVE)
    can_finalise = check_permission(user, Permission.PAYROLL_FINALISE)
    if not (can_approve or can_finalise):
        raise AppError(
            "forbidden",
            "Not permitted to adjust leave balances",
            status.HTTP_403_FORBIDDEN,
        )

    employee = db.get(Employee, payload.employee_id)
    if employee is None:
        raise AppError("not_found", "Employee not found", 404)
    try:
        assert_in_scope(user, employee.id, db)
    except ScopeViolation:
        raise AppError("not_found", "Employee not found", 404) from None

    leave_type = db.get(LeaveType, payload.leave_type_id)
    if leave_type is None:
        raise AppError("not_found", "Leave type not found", 404)

    settings = db.scalar(select(PayrollSettings).limit(1))

    if payload.cycle_start is not None:
        balance = db.scalar(
            select(LeaveBalance).where(
                LeaveBalance.employee_id == employee.id,
                LeaveBalance.leave_type_id == leave_type.id,
                LeaveBalance.cycle_start == payload.cycle_start,
            )
        )
        if balance is None:
            raise AppError(
                "not_found",
                "Leave balance not found for the requested cycle",
                404,
            )
    else:
        bounds = compute_cycle_bounds(
            employee, leave_type, settings, date.today()
        )
        balance = db.scalar(
            select(LeaveBalance).where(
                LeaveBalance.employee_id == employee.id,
                LeaveBalance.leave_type_id == leave_type.id,
                LeaveBalance.cycle_start == bounds.start,
            )
        )
        if balance is None:
            balance = LeaveBalance(
                employee_id=employee.id,
                leave_type_id=leave_type.id,
                cycle_start=bounds.start,
                cycle_end=bounds.end,
                entitled=_zero(),
                accrued=_zero(),
                taken=_zero(),
                pending=_zero(),
                carried_in=_zero(),
                adjusted=_zero(),
                row_version=1,
            )
            db.add(balance)
            db.flush()

    closed = balance.cycle_end < date.today()
    if closed and not can_finalise:
        raise AppError(
            "forbidden",
            "Adjusting a closed cycle requires PAYROLL_FINALISE",
            status.HTTP_403_FORBIDDEN,
            {"cycle_end": balance.cycle_end.isoformat()},
        )

    old_snapshot = _to_out(balance).model_dump(mode="json")
    balance.adjusted = (balance.adjusted or _zero()) + payload.delta
    balance.row_version = balance.row_version + 1
    db.flush()

    if not leave_type.allow_negative:
        avail = available_balance(balance)
        if avail < _zero():
            db.rollback()
            raise AppError(
                "insufficient_balance",
                f"Resulting available balance would be {avail} (negative "
                f"not allowed for this leave type)",
                status.HTTP_422_UNPROCESSABLE_ENTITY,
                {"available": str(avail)},
            )

    ledger = LeaveBalanceLedger(
        employee_id=employee.id,
        leave_type_id=leave_type.id,
        delta=payload.delta,
        reason="adjustment",
        reference_id=None,
        created_by_user_id=getattr(user, "id", None),
    )
    db.add(ledger)
    db.flush()

    db.add(AuditLog(
        id=uuid.uuid4(),
        actor_user_id=getattr(user, "id", None),
        actor_role=getattr(user, "role", None),
        entity_type="leave_balance",
        entity_id=str(balance.id),
        action="leave_balance.adjust",
        old_value=redact_for_audit(old_snapshot),
        new_value=redact_for_audit({
            "delta": str(payload.delta),
            "reason": payload.reason,
            "ledger_entry_id": str(ledger.id),
            "balance": _to_out(balance).model_dump(mode="json"),
        }),
    ))
    db.commit()
    db.refresh(balance)
    return BalanceAdjustOut(
        balance=_to_out(balance),
        ledger_entry_id=ledger.id,
    )


def _zero():
    from decimal import Decimal
    return Decimal("0")
