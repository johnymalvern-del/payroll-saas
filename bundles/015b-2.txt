from __future__ import annotations

import uuid

from pydantic import BaseModel, ConfigDict, Field, model_validator


class EmployeeFieldsOut(BaseModel):
    model_config = ConfigDict(from_attributes=True)

    employee_id: uuid.UUID
    contracted_weekly_hours: float | None = None
    medical_scheme_members: int
    uif_exempt: bool
    row_version: int


class EmployeeFieldsChangeIn(BaseModel):
    """Payload for POST /employees/{id}/fields (F3 field-update request).

    At least one of the three mutable fields must be provided; row_version is
    captured at submission time and re-checked at approval time so a change
    that was queued against an older snapshot is refused with record_changed.
    """

    row_version: int
    contracted_weekly_hours: float | None = Field(None, ge=0)
    medical_scheme_members: int | None = Field(None, ge=0)
    uif_exempt: bool | None = None

    @model_validator(mode="after")
    def _require_one_field(self) -> "EmployeeFieldsChangeIn":
        if all(
            getattr(self, name) is None
            for name in ("contracted_weekly_hours", "medical_scheme_members", "uif_exempt")
        ):
            raise ValueError(
                "at least one of contracted_weekly_hours, medical_scheme_members, "
                "uif_exempt must be provided"
            )
        return self

def _resolve_employee_or_404(db, user, employee_id: uuid.UUID) -> Employee:
    try:
        assert_in_scope(user, employee_id, db)
    except ScopeViolation:
        raise AppError("not_found", "Employee not found", 404)
    emp = db.scalar(select(Employee).where(Employee.id == employee_id))
    if emp is None:
        raise AppError("not_found", "Employee not found", 404)
    return emp


@router.get("/employees/{employee_id}/fields", response_model=EmployeeFieldsOut)
def get_employee_fields(
    employee_id: uuid.UUID,
    db: TenantDbDep,
    user: Annotated[Any, Depends(require_permission(Permission.PAYROLL_PREVIEW))],
) -> EmployeeFieldsOut:
    emp = _resolve_employee_or_404(db, user, employee_id)
    return EmployeeFieldsOut(
        employee_id=emp.id,
        contracted_weekly_hours=(
            float(emp.contracted_weekly_hours)
            if emp.contracted_weekly_hours is not None
            else None
        ),
        medical_scheme_members=emp.medical_scheme_members,
        uif_exempt=emp.uif_exempt,
        row_version=emp.row_version,
    )


@router.post(
    "/employees/{employee_id}/fields",

def submit_employee_fields_change(

        kind="employee_field_update",

"""Payroll change request service (Chapter 4, Bundle 004a-iv-a + iv-b).

F3: HR submits, payroll_admin approves, approver != requester. Pending
requests never touch employee_earning / employee_deduction / employee fields.
Approval writes through in the same transaction as the status flip.

004a-iv-b hardening for employee_field_update:
  - payload requires row_version + at least one mutable field
  - _apply checks row_version at approval time (stale request -> record_changed)
  - _apply refuses if no field was actually written (defence in depth)
"""

def submit_change_request(
    db: Session,
    user: Any,
    *,
    kind: str,
    employee_id: uuid.UUID,
    payload: dict,
    note: str | None = None,
) -> PayrollChangeRequest:
    if kind not in KINDS:
        raise AppError(
            "unknown_kind", f"change request kind {kind!r} is not supported", 422,
        )
    schema = SUBMIT_SCHEMAS[kind]
    try:
        validated = schema(**payload)
    except ValidationError as exc:
        raise AppError(
            "invalid_payload",
            f"payload does not match {kind} schema: {exc.errors()}",
            422,
        ) from None

    try:
        assert_in_scope(user, employee_id, db)
    except ScopeViolation:
        raise AppError("not_found", "Employee not found", 404)

    emp = db.scalar(select(Employee).where(Employee.id == employee_id))
    if emp is None:
        raise AppError("not_found", "Employee not found", 404)

    request = PayrollChangeRequest(
        id=uuid.uuid4(),
        kind=kind,
        employee_id=employee_id,
        payload=validated.model_dump(mode="json"),
        status="pending",
        requested_by_user_id=getattr(user, "id", None),
        decision_note=note,
    )
    db.add(request)
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_change_request", entity_id=request.id, action="submit",
        new_value={
            "kind": kind,
            "employee_id": str(employee_id),
            "payload": validated.model_dump(mode="json"),
        },
    )
    return request


def approve_change_request(
    db: Session, user: Any, request_id: uuid.UUID,
) -> tuple[PayrollChangeRequest, uuid.UUID]:
    request = _load_pending(db, request_id)
    _reject_self_approval(request, user)

    applied_id = _apply(db, request, approver_user=user)
    request.status = "approved"
    request.decided_by_user_id = getattr(user, "id", None)
    request.decided_at = datetime.utcnow()
    request.applied_entity_id = applied_id
    request.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_change_request", entity_id=request.id, action="approve",
        old_value={"status": "pending"},
        new_value={"status": "approved", "applied_entity_id": str(applied_id)},
    )
    return request, applied_id


def reject_change_request(
    db: Session, user: Any, request_id: uuid.UUID, note: str,
) -> PayrollChangeRequest:
    request = _load_pending(db, request_id)
    _reject_self_approval(request, user)
    request.status = "rejected"
    request.decided_by_user_id = getattr(user, "id", None)
    request.decided_at = datetime.utcnow()
    request.decision_note = note
    request.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_change_request", entity_id=request.id, action="reject",
        old_value={"status": "pending"},
        new_value={"status": "rejected", "note": note},
    )
    return request


def cancel_change_request(
    db: Session, user: Any, request_id: uuid.UUID,
) -> PayrollChangeRequest:
    request = _load_pending(db, request_id)
    if request.requested_by_user_id != getattr(user, "id", None):
        raise AppError(
            "not_requester", "Only the requester may cancel a pending change", 403,
        )
    request.status = "cancelled"
    request.row_version += 1
    db.flush()
    write_audit(
        db, user,
        entity_type="payroll_change_request", entity_id=request.id, action="cancel",
        old_value={"status": "pending"}, new_value={"status": "cancelled"},
    )
    return request


# ---------------------------------------------------------------------------
# Internals
# ---------------------------------------------------------------------------

def _load_pending(db: Session, request_id: uuid.UUID) -> PayrollChangeRequest:
    request = db.scalar(
        select(PayrollChangeRequest).where(PayrollChangeRequest.id == request_id)
    )
    if request is None:
        raise AppError("not_found", "Change request not found", 404)
    if request.status != "pending":
        raise AppError(
            "not_pending",
            f"Change request is {request.status}, cannot transition",
            409,
        )
    return request


def _reject_self_approval(request: PayrollChangeRequest, user: Any) -> None:
    if request.requested_by_user_id == getattr(user, "id", None):
        raise AppError(
            "self_approval_forbidden",
            "Approver must differ from requester",
            409,
        )


def _apply(
    db: Session, request: PayrollChangeRequest, *, approver_user: Any,
) -> uuid.UUID:
    kind = request.kind
    p = request.payload

    if kind == "employee_earning_create":
        row = EmployeeEarning(
            id=uuid.uuid4(),
            employee_id=request.employee_id,
            earning_type_id=uuid.UUID(p["earning_type_id"]),
            amount=Decimal(p["amount"]) if p.get("amount") is not None else None,
            rate=Decimal(p["rate"]) if p.get("rate") is not None else None,
            effective_from=date.fromisoformat(p["effective_from"]),
            effective_to=date.fromisoformat(p["effective_to"]) if p.get("effective_to") else None,
            is_active=True,
        )
        _validate_amount_or_rate(row.amount, row.rate)
        db.add(row)
        db.flush()
        return row.id

    if kind == "employee_earning_update":
        row = _load_scoped(db, EmployeeEarning, request.employee_id, uuid.UUID(p["target_id"]))
        _check_row_version(row, p["row_version"])
        _apply_patch(row, p, {"amount", "rate", "effective_from", "effective_to", "is_active"})
        _validate_amount_or_rate(row.amount, row.rate)
        row.row_version += 1
        db.flush()
        return row.id

    if kind == "employee_earning_deactivate":
        row = _load_scoped(db, EmployeeEarning, request.employee_id, uuid.UUID(p["target_id"]))
        _check_row_version(row, p["row_version"])
        row.is_active = False
        row.row_version += 1
        db.flush()
        return row.id

    if kind == "employee_deduction_create":
        row = EmployeeDeduction(
            id=uuid.uuid4(),
            employee_id=request.employee_id,
            deduction_type_id=uuid.UUID(p["deduction_type_id"]),
            amount=Decimal(p["amount"]),
            balance=Decimal("0.00"),
            effective_from=date.fromisoformat(p["effective_from"]),
            effective_to=date.fromisoformat(p["effective_to"]) if p.get("effective_to") else None,
            is_active=True,
        )
        db.add(row)
        db.flush()
        return row.id

    if kind == "employee_deduction_update":
        row = _load_scoped(db, EmployeeDeduction, request.employee_id, uuid.UUID(p["target_id"]))
        _check_row_version(row, p["row_version"])
        _apply_patch(row, p, {"amount", "effective_from", "effective_to", "is_active"})
        row.row_version += 1
        db.flush()
        return row.id

    if kind == "employee_deduction_deactivate":
        row = _load_scoped(db, EmployeeDeduction, request.employee_id, uuid.UUID(p["target_id"]))
        _check_row_version(row, p["row_version"])
        row.is_active = False
        row.row_version += 1
        db.flush()
        return row.id

    if kind == "employee_field_update":
        emp = db.scalar(select(Employee).where(Employee.id == request.employee_id))
        if emp is None:
            raise AppError("not_found", "Employee not found", 404)
        expected_rv = p.get("row_version")
        if expected_rv is not None and emp.row_version != expected_rv:
            raise AppError(
                "record_changed",
                f"row_version mismatch: sent {expected_rv}, current {emp.row_version}",
                409,
            )
        applied_any = False
        for field_name in ("contracted_weekly_hours", "medical_scheme_members", "uif_exempt"):
            if field_name in p and p[field_name] is not None:
                setattr(emp, field_name, p[field_name])
                applied_any = True
        if not applied_any:
            raise AppError(
                "empty_update",
                "no recognised fields to update on employee",
                422,
            )
        emp.row_version += 1
        db.flush()
        return emp.id

    raise AppError("unknown_kind", f"no handler for kind {kind!r}", 500)


def _load_scoped(db: Session, model: Any, employee_id: uuid.UUID, target_id: uuid.UUID) -> Any:
    row = db.scalar(
        select(model).where(model.id == target_id, model.employee_id == employee_id)
    )
    if row is None:
        raise AppError("not_found", "Target row not found for this employee", 404)
    return row


def _check_row_version(row: Any, expected: int) -> None:
    if row.row_version != expected:
        raise AppError(
            "record_changed",
            f"row_version mismatch: sent {expected}, current {row.row_version}",
            409,
        )


def _apply_patch(row: Any, payload: dict, allowed: set[str]) -> None:
    for field_name in allowed:
        if field_name in payload and payload[field_name] is not None:
            value = payload[field_name]
            if field_name in ("effective_from", "effective_to") and value is not None:
                value = date.fromisoformat(value)
            setattr(row, field_name, value)


def _validate_amount_or_rate(amount: Decimal | None, rate: Decimal | None) -> None:
    if amount is None and rate is None:
        raise AppError("amount_or_rate_required", "Provide amount or rate", 422)
    if amount is not None and rate is not None:
        raise AppError("amount_xor_rate", "Provide amount or rate, not both", 422)

def _route_map():
    from fastapi.routing import APIRoute
    from app.main import app
    out: dict[tuple[str, str], APIRoute] = {}
    for r in app.routes:
        if not isinstance(r, APIRoute):
            continue
        for m in r.methods:
            if m in ("HEAD", "OPTIONS"):
                continue
            out[(m, r.path)] = r
    return out


def _extract_permissions(route) -> set[Permission]:
    found: set[Permission] = set()
    stack = [route.dependant]
    while stack:
        d = stack.pop()
        fn = getattr(d, "call", None)
        if fn is not None and getattr(fn, "__closure__", None):
            for cell in fn.__closure__:
                try:
                    v = cell.cell_contents
                except ValueError:
                    continue
                if isinstance(v, Permission):
                    found.add(v)
        stack.extend(getattr(d, "dependencies", []) or [])
    return found


_IDS = [f"{m}-{p}-{perm.value}" for m, p, perm in EXPECTED_ROUTES]


@pytest.mark.parametrize("method,path,expected_perm", EXPECTED_ROUTES, ids=_IDS)
def test_route_registered(method: str, path: str, expected_perm: Permission) -> None:
    routes = _route_map()
    key = (method, path)
    assert key in routes, (
        f"Route {method} {path} not registered. "
        f"Available payroll routes: {sorted(k for k in routes if 'payroll' in k[1])}"
    )


@pytest.mark.parametrize("method,path,expected_perm", EXPECTED_ROUTES, ids=_IDS)
def test_route_has_expected_permission(method: str, path: str, expected_perm: Permission) -> None:
    routes = _route_map()
    route = routes.get((method, path))
    assert route is not None, f"Route {method} {path} not found"
    perms = _extract_permissions(route)
    assert expected_perm in perms, (
        f"{method} {path}: expected permission {expected_perm.value!r}, "
        f"found {sorted(p.value for p in perms)}"
    )


def test_payroll_routes_all_have_a_permission() -> None:
    routes = _route_map()
    offenders = []
    for (method, path), route in routes.items():
        if not path.startswith("/api/v1/payroll"):
            continue
        if not _extract_permissions(route):
            offenders.append(f"{method} {path}")
    assert not offenders, f"payroll routes missing require_permission: {offenders}"
