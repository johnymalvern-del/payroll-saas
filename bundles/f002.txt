import createClient from "openapi-fetch";
import type { paths } from "./schema";

let accessToken: string | null = null;
let refreshPromise: Promise<boolean> | null = null;
let onUnauthorized: (() => void) | null = null;

export function getAccessToken(): string | null {
  return accessToken;
}

export function setAccessToken(token: string | null): void {
  accessToken = token;
}

export function setOnUnauthorized(fn: (() => void) | null): void {
  onUnauthorized = fn;
}

async function refreshOnce(): Promise<boolean> {
  if (refreshPromise) return refreshPromise;
  refreshPromise = (async () => {
    try {
      const res = await fetch("/api/v1/auth/refresh", {
        method: "POST",
        credentials: "include",
      });
      if (!res.ok) return false;
      const body = (await res.json().catch(() => null)) as
        | { access_token?: unknown }
        | null;
      const token = body?.access_token;
      if (typeof token === "string") {
        accessToken = token;
        return true;
      }
      return false;
    } catch {
      return false;
    }
  })();
  try {
    return await refreshPromise;
  } finally {
    refreshPromise = null;
  }
}

const wrappedFetch = async (req: Request): Promise<Response> => {
  const original = req.clone();
  const first = new Request(req);
  if (accessToken) first.headers.set("Authorization", `Bearer ${accessToken}`);
  let res = await fetch(first);
  if (res.status !== 401) return res;
  const ok = await refreshOnce();
  if (!ok) {
    onUnauthorized?.();
    return res;
  }
  const retry = new Request(original);
  if (accessToken) retry.headers.set("Authorization", `Bearer ${accessToken}`);
  res = await fetch(retry);
  if (res.status === 401) onUnauthorized?.();
  return res;
};

export const api = createClient<paths>({
  baseUrl: "/api/v1",
  credentials: "include",
  fetch: wrappedFetch,
});

export function mapErrorCode(code: string | undefined, fallback?: string): string {
  if (code) {
    const m = MESSAGES[code];
    if (m) return m;
  }
  return fallback ?? "Something went wrong. Please try again.";
}

export async function parseApiError(response: Response): Promise<ApiError> {
  try {
    const body = (await response.json()) as
      | { error?: { code?: unknown; message?: unknown; details?: unknown } }
      | null;
    const err = body?.error ?? {};
    const code = typeof err.code === "string" ? err.code : "http_error";
    const message =
      typeof err.message === "string" ? err.message : response.statusText;
    return { code, message, status: response.status, details: err.details };
  } catch {
    return {
      code: "http_error",
      message: response.statusText,
      status: response.status,
    };
  }
}

import { describe, expect, it } from "vitest";
import { mapErrorCode } from "./errors";

describe("mapErrorCode", () => {
  it("maps known codes to human messages", () => {
    expect(mapErrorCode("record_changed")).toContain("Reload");
  });

  it("returns the generic fallback for unknown codes", () => {
    expect(mapErrorCode("unknown_thing")).toBe(
      "Something went wrong. Please try again.",
    );
  });

  it("uses the supplied fallback message when provided", () => {
    expect(mapErrorCode("unknown_thing", "Try again")).toBe("Try again");
  });

  it("handles undefined code", () => {
    expect(mapErrorCode(undefined)).toBe(
      "Something went wrong. Please try again.",
    );
  });
});

import { StrictMode } from "react";
import { createRoot } from "react-dom/client";
import { MantineProvider } from "@mantine/core";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { BrowserRouter } from "react-router-dom";
import "@mantine/core/styles.css";
import App from "./App";
import { theme } from "./theme";

const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 30_000,
      refetchOnWindowFocus: false,
      retry: 1,
    },
  },
});

const rootEl = document.getElementById("root");
if (!rootEl) throw new Error("root element not found");

createRoot(rootEl).render(
  <StrictMode>
    <MantineProvider theme={theme} defaultColorScheme="auto">
      <QueryClientProvider client={queryClient}>
        <BrowserRouter>
          <App />
        </BrowserRouter>
      </QueryClientProvider>
    </MantineProvider>
  </StrictMode>,
);
