"""System payroll seed (Chapter 4, Bundle 004a-iv-c).

Idempotent seed of the four system earning types (normal_pay, overtime,
holiday_pay, leave_pay) required by the pipeline for standard SA payroll.
All four are classified `frequency="regular"` per F5 - only genuine one-offs
(bonus, commission, advances) use once_off/irregular and are refused under
the regular-earnings method.

Called from:
  - POST /payroll/earning-types/ensure-system  (this bundle)
  - import_employees.py on first import (future refinement, noted in STATE.md)
  - provision_tenant post-migration hook (future refinement, noted in STATE.md)

Never from tests' disposable-tenant fixture directly - they call the endpoint
or the service function explicitly so the seed state is deterministic.
"""
from __future__ import annotations

import uuid

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.db.tenant.models.pay import EarningType


SYSTEM_TYPES: tuple[dict, ...] = (
    {
        "code": "normal_pay",
        "name": "Normal Pay",
        "calc_method": "per_hour",
        "frequency": "regular",
        "is_taxable": True,
        "counts_for_uif": True,
        "counts_for_sdl": True,
        "proration_rule": "none",
    },
    {
        "code": "overtime",
        "name": "Overtime",
        "calc_method": "per_hour",
        "frequency": "regular",
        "is_taxable": True,
        "counts_for_uif": True,
        "counts_for_sdl": True,
        "proration_rule": "none",
    },
    {
        "code": "holiday_pay",
        "name": "Holiday Pay",
        "calc_method": "per_hour",
        "frequency": "regular",
        "is_taxable": True,
        "counts_for_uif": True,
        "counts_for_sdl": True,
        "proration_rule": "none",
    },
    {
        "code": "leave_pay",
        "name": "Leave Pay",
        "calc_method": "per_hour",
        "frequency": "regular",
        "is_taxable": True,
        "counts_for_uif": True,
        "counts_for_sdl": True,
        "proration_rule": "none",
    },
)


def ensure_system_earning_types(db: Session) -> dict:
    """Idempotent. Existing rows (by code) are left untouched even if fields differ -
    we do not silently rewrite system rows after they have been used."""
    created: list[str] = []
    existing: list[str] = []
    for spec in SYSTEM_TYPES:
        row = db.scalar(select(EarningType).where(EarningType.code == spec["code"]))
        if row is not None:
            existing.append(spec["code"])
            continue
        db.add(EarningType(
            id=uuid.uuid4(),
            name=spec["name"],
            code=spec["code"],
            calc_method=spec["calc_method"],
            frequency=spec["frequency"],
            is_taxable=spec["is_taxable"],
            counts_for_uif=spec["counts_for_uif"],
            counts_for_sdl=spec["counts_for_sdl"],
            proration_rule=spec["proration_rule"],
            is_system=True,
            is_active=True,
        ))
        created.append(spec["code"])
    db.flush()
    return {"created": created, "existing": existing, "total": len(SYSTEM_TYPES)}

"""Import-field helpers (Chapter 4, Bundle 004a-iv-c).

The three payroll-specific employee fields (contracted_weekly_hours,
medical_scheme_members, uif_exempt) must be set through the same approval
mechanism as pay-rate changes (F3). This module provides:

  - IMPORT_FIELDS: the three field names a CSV/JSON row may supply.
  - validate_import_row_fields(row_dict) -> list[str]: returns a list of
    validation errors (empty if OK). Caller is expected to fail the whole
    import batch on any error.
  - apply_import_row_fields(employee, row_dict) -> bool: writes the three
    fields to `employee` if present; returns True if any was written.

Wiring into import_employees.py is a documented follow-up (STATE.md at close).
The helper is deliberately side-effect-free on the Employee until the caller
commits, so no partial writes.

The three fields, once imported, still require payroll_admin approval to take
effect on payroll - imported values on Employee are only a stored snapshot.
Enforcing F3 on the payroll run is what the pipeline actually consumes.
"""
from __future__ import annotations

from typing import Any

from app.db.tenant.models.people import Employee


IMPORT_FIELDS: tuple[str, ...] = (
    "contracted_weekly_hours",
    "medical_scheme_members",
    "uif_exempt",
)


def validate_import_row_fields(row: dict[str, Any]) -> list[str]:
    errors: list[str] = []
    if "contracted_weekly_hours" in row and row["contracted_weekly_hours"] is not None:
        try:
            hours = float(row["contracted_weekly_hours"])
            if hours < 0 or hours > 168:
                errors.append(
                    f"contracted_weekly_hours out of range: {hours} (must be 0..168)"
                )
        except (TypeError, ValueError):
            errors.append(
                f"contracted_weekly_hours not numeric: {row['contracted_weekly_hours']!r}"
            )

    if "medical_scheme_members" in row and row["medical_scheme_members"] is not None:
        try:
            members = int(row["medical_scheme_members"])
            if members < 0:
                errors.append(f"medical_scheme_members negative: {members}")
        except (TypeError, ValueError):
            errors.append(
                f"medical_scheme_members not integer: {row['medical_scheme_members']!r}"
            )

    if "uif_exempt" in row and row["uif_exempt"] is not None:
        val = row["uif_exempt"]
        if not isinstance(val, bool) and val not in (0, 1, "0", "1", "true", "false", "TRUE", "FALSE"):
            errors.append(f"uif_exempt not boolean-like: {val!r}")

    return errors


def apply_import_row_fields(employee: Employee, row: dict[str, Any]) -> bool:
    """Mutate `employee` in place for the three payroll fields. Returns True if any
    field was set. Does NOT commit - caller commits within the batch transaction."""
    wrote = False

    hours = row.get("contracted_weekly_hours")
    if hours is not None:
        employee.contracted_weekly_hours = float(hours)
        wrote = True

    members = row.get("medical_scheme_members")
    if members is not None:
        employee.medical_scheme_members = int(members)
        wrote = True

    exempt = row.get("uif_exempt")
    if exempt is not None:
        if isinstance(exempt, bool):
            employee.uif_exempt = exempt
        elif exempt in (1, "1", "true", "TRUE"):
            employee.uif_exempt = True
        elif exempt in (0, "0", "false", "FALSE"):
            employee.uif_exempt = False
        wrote = True

    return wrote

def _route_map():
    from fastapi.routing import APIRoute
    from app.main import app
    out: dict[tuple[str, str], APIRoute] = {}
    for r in app.routes:
        if not isinstance(r, APIRoute):
            continue
        for m in r.methods:
            if m in ("HEAD", "OPTIONS"):
                continue
            out[(m, r.path)] = r
    return out


def _extract_permissions(route) -> set[Permission]:
    found: set[Permission] = set()
    stack = [route.dependant]
    while stack:
        d = stack.pop()
        fn = getattr(d, "call", None)
        if fn is not None and getattr(fn, "__closure__", None):
            for cell in fn.__closure__:
                try:
                    v = cell.cell_contents
                except ValueError:
                    continue
                if isinstance(v, Permission):
                    found.add(v)
        stack.extend(getattr(d, "dependencies", []) or [])
    return found


_IDS = [f"{m}-{p}-{perm.value}" for m, p, perm in EXPECTED_ROUTES]

def test_route_registered(method: str, path: str, expected_perm: Permission) -> None:

    key = (method, path)

        f"Route {method} {path} not registered. "
        f"Available payroll routes: {sorted(k for k in routes if 'payroll' in k[1])}"
    )


@pytest.mark.parametrize("method,path,expected_perm", EXPECTED_ROUTES, ids=_IDS)
def test_route_has_expected_permission(method: str, path: str, expected_perm: Permission) -> None:
    routes = _route_map()
    route = routes.get((method, path))
    assert route is not None, f"Route {method} {path} not found"
    perms = _extract_permissions(route)
    assert expected_perm in perms, (
        f"{method} {path}: expected permission {expected_perm.value!r}, "
        f"found {sorted(p.value for p in perms)}"
    )


def test_payroll_routes_all_have_a_permission() -> None:
    routes = _route_map()
    offenders = []
    for (method, path), route in routes.items():
        if not path.startswith("/api/v1/payroll"):
            continue
        if not _extract_permissions(route):
            offenders.append(f"{method} {path}")
    assert not offenders, f"payroll routes missing require_permission: {offenders}"
